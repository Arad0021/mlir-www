31 /// A streamer forwarding every reported remark to a C callback.
32 class CallbackRemarkStreamer : public remark::detail::MLIRRemarkStreamerBase {
33 public:
34  CallbackRemarkStreamer(MlirRemarkCallback callback, void *userData,
35  void (*deleteUserData)(void *))
36  : callback(callback), userData(userData), deleteUserData(deleteUserData) {
37  }
38 
39  ~CallbackRemarkStreamer() override {
40  if (deleteUserData)
41  deleteUserData(userData);
42  }
43 
44  void streamOptimizationRemark(const Remark &remark) override {
45  callback(wrap(&remark), userData);
46  }
47 
48 private:
49  MlirRemarkCallback callback;
50  void *userData;
51  void (*deleteUserData)(void *);
52 };

55 /// Every filter reaches the engine as a (possibly empty) string, as with
56 /// mlir-opt's options: a kind is reported when its own filter or `all` matches.
57 static remark::RemarkCategories unwrap(MlirRemarkCategories categories) {
58  return remark::RemarkCategories{
59  unwrap(categories.all).str(), unwrap(categories.passed).str(),
60  unwrap(categories.missed).str(), unwrap(categories.analysis).str(),
61  unwrap(categories.failed).str()};
62 }

66  switch (policy) {

85 static MlirRemarkKind wrap(remark::RemarkKind kind) {
86  switch (kind) {
87  case remark::RemarkKind::RemarkUnknown:
88  return MlirRemarkKindUnknown;
89  case remark::RemarkKind::RemarkPassed:
90  return MlirRemarkKindPassed;
91  case remark::RemarkKind::RemarkMissed:
92  return MlirRemarkKindMissed;
93  case remark::RemarkKind::RemarkFailure:
94  return MlirRemarkKindFailure;
95  case remark::RemarkKind::RemarkAnalysis:
96  return MlirRemarkKindAnalysis;
97  }
98  llvm_unreachable("unknown remark kind");
99 }

105 MlirLogicalResult mlirContextEnableOptimizationRemarks(
106  MlirContext context, MlirRemarkCategories categories,
107  MlirRemarkPolicy policy, bool printAsEmitRemarks) {
108  MLIRContext *ctx = unwrap(context);
109  if (ctx->getRemarkEngine())
110  return mlirLogicalResultFailure();
111  return wrap(remark::enableOptimizationRemarks(
112  *ctx, /*streamer=*/nullptr, createPolicy(policy), unwrap(categories),
113  printAsEmitRemarks));
114 }

128 MlirLogicalResult mlirContextEnableOptimizationRemarksWithCallback(
129  MlirContext context, MlirRemarkCategories categories,
130  MlirRemarkPolicy policy, MlirRemarkCallback callback, void *userData,
131  void (*deleteUserData)(void *), bool printAsEmitRemarks) {
132  assert(callback && "unexpected null remark callback");
133  MLIRContext *ctx = unwrap(context);
134  if (ctx->getRemarkEngine()) {
135  // The caller handed over `userData`; release it as on success.
136  if (deleteUserData)
137  deleteUserData(userData);
138  return mlirLogicalResultFailure();
139  }
140  return wrap(remark::enableOptimizationRemarks(
141  *ctx,
142  std::make_unique<CallbackRemarkStreamer>(callback, userData,
143  deleteUserData),
144  createPolicy(policy), unwrap(categories), printAsEmitRemarks));
145 }

217 bool mlirEmitOptimizationRemark(
218  MlirLocation location, MlirRemarkKind kind, MlirStringRef remarkName,
219  MlirStringRef categoryName, MlirStringRef subCategoryName,
220  MlirStringRef functionName, MlirStringRef message, intptr_t numArgs,
221  const MlirStringRef *argKeys, const MlirStringRef *argValues) {
222  assert((numArgs == 0 || (argKeys && argValues)) &&
223  "remark arguments require both keys and values");
224  Location loc = unwrap(location);
225  remark::RemarkOpts opts = remark::RemarkOpts::name(unwrap(remarkName))
226  .category(unwrap(categoryName))
227  .subCategory(unwrap(subCategoryName))
228  .function(unwrap(functionName));
229  remark::detail::InFlightRemark inFlight;
230  switch (kind) {
231  case MlirRemarkKindUnknown:
232  return false;
233  case MlirRemarkKindPassed:
234  inFlight = remark::passed(loc, opts);
235  break;
236  case MlirRemarkKindMissed:
237  inFlight = remark::missed(loc, opts);
238  break;
239  case MlirRemarkKindFailure:
240  inFlight = remark::failed(loc, opts);
241  break;
242  case MlirRemarkKindAnalysis:
243  inFlight = remark::analysis(loc, opts);
244  break;
245  }
246  if (!inFlight)
247  return false;
248  if (message.length != 0)
249  inFlight << unwrap(message);
250  for (intptr_t i = 0; i < numArgs; ++i)
251  inFlight << Remark::Arg(unwrap(argKeys[i]), unwrap(argValues[i]));
252  // The remark is reported when `inFlight` goes out of scope.
253  return true;
254 }
