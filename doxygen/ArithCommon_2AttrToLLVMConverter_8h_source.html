49 // Attribute converter that populates a NamedAttrList by removing the fastmath
50 // attribute from the source operation attributes, and replacing it with an
51 // equivalent LLVM fastmath attribute.
52 template <typename SourceOp, typename TargetOp>
53 class AttrConvertFastMathToLLVM {
54 public:
55  AttrConvertFastMathToLLVM(SourceOp srcOp)
56  : convertedAttr(srcOp->getDiscardableAttrDictionary()) {
57  srcOp->getName().populateInherentAttrs(srcOp, convertedAttr);
58  convertedAttr.erase(SourceOp::getFastMathAttrName());
59  auto arithFMFAttr = srcOp.getFastMathFlagsAttr();
60  if (arithFMFAttr) {
61  convertedAttr.set(TargetOp::getFastmathAttrName(),
62  convertArithFastMathAttrToLLVM(arithFMFAttr));
63  }
64  }
65  ArrayRef<NamedAttribute> getAttrs() const { return convertedAttr.getAttrs(); }
66  Attribute getPropAttr() const { return {}; }
67 
68 private:
69  NamedAttrList convertedAttr;
70 };

72 // Attribute converter that populates a NamedAttrList by removing the overflow
73 // attribute from the source operation attributes, and replacing it with an
74 // equivalent LLVM overflow attribute.
75 template <typename SourceOp, typename TargetOp>
76 class AttrConvertOverflowToLLVM {
77 public:
78  AttrConvertOverflowToLLVM(SourceOp srcOp)
79  : convertedAttr(srcOp->getDiscardableAttrDictionary()) {
80  using IntegerOverflowFlagsAttr = LLVM::IntegerOverflowFlagsAttr;
81 
82  if (auto arithAttr = srcOp.getOverflowAttr()) {
83  auto llvmFlag = convertArithOverflowFlagsToLLVM(arithAttr.getValue());
84  // Create a dictionary attribute holding the overflow flags property.
85  // (In the LLVM dialect, the overflow flags are a property, not an
86  // attribute.)
87  MLIRContext *ctx = srcOp.getOperation()->getContext();
88  Builder b(ctx);
89  auto llvmFlagAttr = IntegerOverflowFlagsAttr::get(ctx, llvmFlag);
90  StringRef llvmAttrName = TargetOp::getOverflowFlagsAttrName();
91  NamedAttribute attr{llvmAttrName, llvmFlagAttr};
92  // Set the properties attribute of the operation state so that the
93  // property can be updated when the operation is created.
94  propertiesAttr = b.getDictionaryAttr(ArrayRef(attr));
95  }
96  }
97  ArrayRef<NamedAttribute> getAttrs() const { return convertedAttr.getAttrs(); }
98  Attribute getPropAttr() const { return propertiesAttr; }
99 
100 private:
101  NamedAttrList convertedAttr;
102  DictionaryAttr propertiesAttr;
103 };

105 // Attribute converter that populates a NamedAttrList by removing the nonNeg
106 // attribute from the source operation attributes, and setting it as a property
107 // on the target LLVM operation.
108 template <typename SourceOp, typename TargetOp>
109 class AttrConvertNonNegToLLVM {
110 public:
111  AttrConvertNonNegToLLVM(SourceOp srcOp)
112  : convertedAttr(srcOp->getDiscardableAttrDictionary()) {
113  if (!srcOp.getNonNeg())
114  return;
115  MLIRContext *ctx = srcOp.getOperation()->getContext();
116  Builder b(ctx);
117  NamedAttribute attr{"nonNeg", b.getUnitAttr()};
118  propertiesAttr = b.getDictionaryAttr(ArrayRef(attr));
119  }
120  ArrayRef<NamedAttribute> getAttrs() const { return convertedAttr.getAttrs(); }
121  Attribute getPropAttr() const { return propertiesAttr; }
122 
123 private:
124  NamedAttrList convertedAttr;
125  DictionaryAttr propertiesAttr;
126 };

128 template <typename SourceOp, typename TargetOp>
129 class AttrConverterConstrainedFPToLLVM {
130  static_assert(TargetOp::template hasTrait<
131  LLVM::FPExceptionBehaviorOpInterface::Trait>(),
132  "Target constrained FP operations must implement "
133  "LLVM::FPExceptionBehaviorOpInterface");
134 
135 public:
136  AttrConverterConstrainedFPToLLVM(SourceOp srcOp)
137  : convertedAttr(srcOp->getDiscardableAttrDictionary()) {
138  if constexpr (TargetOp::template hasTrait<
139  LLVM::RoundingModeOpInterface::Trait>()) {
140  auto arithAttr = srcOp.getRoundingModeAttr();
141  convertedAttr.set(TargetOp::getRoundingModeAttrName(),
142  convertArithRoundingModeAttrToLLVM(arithAttr));
143  }
144  // Constrained intrinsics (llvm.intr.experimental.constrained.*) do not
145  // support fastmath flags, so do not copy them from the source operation.
146  convertedAttr.set(TargetOp::getFPExceptionBehaviorAttrName(),
147  getLLVMDefaultFPExceptionBehavior(*srcOp->getContext()));
148  }
149 
150  ArrayRef<NamedAttribute> getAttrs() const { return convertedAttr.getAttrs(); }
151  Attribute getPropAttr() const { return {}; }
152 
153 private:
154  NamedAttrList convertedAttr;
155 };
