65 /// Helper class for tracking the analysis state of an mlir value. For layout
66 /// propagation, the analysis state is simply the distribution layout of
67 /// each value. The distribution layout information is encapsulated using
68 /// xegpu::DistributeLayoutAttr class which can hold information about any type
69 /// of distribution layout that XeGPU dialect supports. Purpose of this analysis
70 /// to propagate some unique distribution layout for each value in the program
71 /// starting from a set of anchor operations (like DPAS, StoreNd, etc.). Note
72 /// that analysis will reach a fixed point when all values are reached some
73 /// layout and, analysis does not try to modify any already assigned layouts.
74 ///
75 /// Given this, LayoutInfo satisifies the following properties:
76 /// 1) A LayoutInfo value can be in one of two states - `assigned` or `not
77 /// assigned`.
78 /// 2) Two LayoutInfo values are equal if they are both not assigned, or both
79 /// assigned with the same layout.
80 /// 3) The meet operator works as follows:
81 /// - If only one side is assigned, return that side.
82 /// - If both sides are assigned, prefer the layout demanded by the user
83 /// that is nearer to the producer in program order (smaller
84 /// `programOrder`); on a tie keep lhs.
85 ///
86 /// The `programOrder` field records the program-order index of the consumer op
87 /// that demanded the layout (stamped via
88 /// `LayoutInfoPropagation::makeLayoutInfo` from
89 /// `LayoutInfoPropagation::currentProgramOrder`). During this backward analysis
90 /// a value can be demanded by several users; keeping the nearest one tends to
91 /// preserve a consumer's layout as far up the def chain as possible, minimizing
92 /// layout conversions. This is a hint, not an optimum. `programOrder` is never
93 /// propagated up the chain - each visited op stamps its own index - so it is
94 /// excluded from `operator==`.
95 
96 struct LayoutInfo {
97 private:
98  xegpu::DistributeLayoutAttr storage = nullptr;
99  // Program-order index of the consumer op that demanded this layout. Smaller
100  // means nearer to the producer. Unassigned/unknown demands sort last.
101  int64_t programOrder = std::numeric_limits<int64_t>::max();
102 
103 public:
104  LayoutInfo() = default;
105  LayoutInfo(const xegpu::DistributeLayoutAttr &layout, int64_t programOrder)
106  : storage(layout), programOrder(programOrder) {}
107 
108  // Equality by assignment state and, when both assigned, by the layout:
109  // - one assigned, the other not -> not equal;
110  // - both unassigned -> equal;
111  // - both assigned -> equal iff the layouts match.
112  bool operator==(const LayoutInfo &other) const {
113  if (isAssigned() != other.isAssigned())
114  return false;
115  if (!isAssigned())
116  return true;
117  return storage.isEqualTo(other.storage);
118  }

164 LayoutInfo LayoutInfo::meet(const LayoutInfo &lhs, const LayoutInfo &rhs) {
165  if (!lhs.isAssigned())
166  return rhs;
167  if (!rhs.isAssigned())
168  return lhs;
169  // Prefer the demand from the user nearer to the producer in program order.
170  // Distinct users always have distinct indices, so this decides every
171  // real conflict; on a tie (same op, or both unknown) keep lhs.
172  if (rhs.programOrder < lhs.programOrder)
173  return rhs;
174  return lhs;
175 }

354 int64_t LayoutInfoPropagation::getProgramOrder(Operation *op) {
355  auto it = programOrder.find(op);
356  if (it != programOrder.end())
357  return it->second;
358  // First time we see this op's tree: number every op under its top-level
359  // ancestor in pre-order (i.e. printed-IR order). Nested ops (e.g. inside an
360  // scf.for body) get an index between their parent and the parent's next
361  // sibling, so a use inside a loop is "nearer" than a use after it.
362  Operation *root = op;
363  while (root->getParentOp())
364  root = root->getParentOp();
365  int64_t counter = 0;
366  root->walk<WalkOrder::PreOrder>(
367  [&](Operation *o) { programOrder[o] = counter++; });
368  return programOrder.lookup(op);
369 }

371 LogicalResult LayoutInfoPropagation::visitOperation(
372  Operation *op, ArrayRef<LayoutInfoLattice *> operands,
373  ArrayRef<const LayoutInfoLattice *> results) {
374  // Stamp demands pushed by this op with its program-order index so `meet` can
375  // prefer the nearest consumer.
376  currentProgramOrder = getProgramOrder(op);
377  TypeSwitch<Operation *>(op)
378  .Case(
379  [&](xegpu::DpasOp dpasOp) { visitDpasOp(dpasOp, operands, results); })
380  .Case([&](xegpu::DpasMxOp dpasMxOp) {
381  visitDpasMxOp(dpasMxOp, operands, results);
382  })
383  .Case([&](xegpu::StoreNdOp storeNdOp) {
384  visitStoreNdOp(storeNdOp, operands, results);
385  })
386  .Case([&](xegpu::StoreScatterOp storeScatterOp) {
387  visitStoreScatterOp(storeScatterOp, operands, results);
388  })
389  .Case([&](xegpu::LoadNdOp loadNdOp) {
390  visitLoadNdOp(loadNdOp, operands, results);
391  })
392  .Case([&](xegpu::LoadGatherOp loadGatherOp) {
393  visitLoadGatherOp(loadGatherOp, operands, results);
394  })
395  .Case([&](xegpu::PrefetchNdOp prefetchNdOp) {
396  visitPrefetchNdOp(prefetchNdOp, operands, results);
397  })
398  .Case([&](vector::TransposeOp transposeOp) {
399  visitTransposeOp(transposeOp, operands, results);
400  })
401  .Case([&](vector::BitCastOp bitcastOp) {
402  visitVectorBitcastOp(bitcastOp, operands, results);
403  })
404  .Case([&](vector::InterleaveOp interleaveOp) {
405  visitVectorInterleaveOp(interleaveOp, operands, results);
406  })
407  .Case([&](vector::DeinterleaveOp deinterleaveOp) {
408  visitVectorDeinterleaveOp(deinterleaveOp, operands, results);
409  })
410  .Case([&](vector::MultiDimReductionOp reductionOp) {
411  visitVectorMultiReductionOp(reductionOp, operands, results);
412  })
413  .Case([&](vector::ReductionOp reductionOp) {
414  visitVectorReductionOp(reductionOp, operands, results);
415  })
416  .Case([&](vector::BroadcastOp broadcastOp) {
417  visitVectorBroadCastOp(broadcastOp, operands, results);
418  })
419  .Case([&](vector::ShapeCastOp shapeCastOp) {
420  visitShapeCastOp(shapeCastOp, operands, results);
421  })
422  .Case([&](vector::InsertStridedSliceOp insertStridedSliceOp) {
423  visitInsertStridedSliceOp(insertStridedSliceOp, operands, results);
424  })
425  .Case([&](xegpu::LoadMatrixOp loadMatrixOp) {
426  visitLoadMatrixOp(loadMatrixOp, operands, results);
427  })
428  .Case([&](xegpu::StoreMatrixOp storeMatrixOp) {
429  visitStoreMatrixOp(storeMatrixOp, operands, results);
430  })
431  .Case([&](xegpu::ConvertLayoutOp convertLayoutOp) {
432  visitConvertLayoutOp(convertLayoutOp, operands, results);
433  })
434  // All other ops.
435  .Default([&](Operation *op) {
436  for (const LayoutInfoLattice *resultInfo : results) {
437  if (!resultInfo->getValue().isAssigned())
438  continue;
439  for (auto [operandInfo, operand] :
440  llvm::zip(operands, op->getOpOperands())) {
441  // If the operand type is not a vector or tensor descriptor, skip
442  // it.
443  if (!isa<xegpu::TensorDescType, VectorType>(
444  operand.get().getType()))
445  continue;
446  // Propagate the result layout to the operand.
447  meet(operandInfo, *resultInfo);
448  }
449  }
450  });
451 
452  return success();
453 }

455 bool LayoutInfoPropagation::hasParamsOfLayoutKind(
456  xegpu::DistributeLayoutAttr anchorLayout) {
457  if (anchorLayout == nullptr) {
458  return false;
459  }
460  if (layoutKind == xegpu::LayoutKind::InstData) {
461  return !(anchorLayout.getEffectiveInstDataAsInt().empty());
462  }
463  if (layoutKind == xegpu::LayoutKind::Lane) {
464  return !(anchorLayout.getEffectiveLaneLayoutAsInt().empty() ||
465  anchorLayout.getEffectiveLaneDataAsInt().empty());
466  }
467  if (layoutKind == xegpu::LayoutKind::Subgroup) {
468  return !(anchorLayout.getEffectiveSgLayoutAsInt().empty() ||
469  anchorLayout.getEffectiveSgDataAsInt().empty());
470  }
471  return false;
472 }

474 FailureOr<int64_t> LayoutInfoPropagation::getNumSgOrFail(
475  Operation *op, int sgSize, xegpu::DistributeLayoutAttr consumerLayout) {
476  // The consumer's sg_layout, when present, dictates the count.
477  if (consumerLayout) {
478  auto sgLayout = consumerLayout.getEffectiveSgLayoutAsInt();
479  if (!sgLayout.empty())
480  return llvm::product_of(sgLayout);
481  }
482  // Otherwise fall back to the kernel's known_block_size.
483  if (auto gpuFunc = op->getParentOfType<gpu::GPUFuncOp>()) {
484  std::optional<ArrayRef<int32_t>> knownBlockSize =
485  gpuFunc.getKnownBlockSize();
486  if (knownBlockSize) {
487  bool isPowerOf2Block = llvm::all_of(*knownBlockSize, [](int32_t dim) {
488  return dim > 0 && llvm::isPowerOf2_32(dim);
489  });
490  int64_t numSg = llvm::product_of(*knownBlockSize) / sgSize;
491  if (isPowerOf2Block && numSg > 0)
492  return numSg;
493  }
494  }
495  // Only subgroup mode needs the count; elsewhere a missing one is benign.
496  if (layoutKind == xegpu::LayoutKind::Subgroup) {
497  markFailure(op, "Unable to determine the number of subgroups for the "
498  "operation. Please check @known_block_size is properly "
499  "attached as kernel attributes, with power-of-two "
500  "dimensions covering at least one subgroup.");
501  return failure();
502  }
503  return int64_t{0};
504 }

506 void LayoutInfoPropagation::visitPrefetchNdOp(
507  xegpu::PrefetchNdOp prefetch, ArrayRef<LayoutInfoLattice *> operands,
508  ArrayRef<const LayoutInfoLattice *> results) {
509 
510  LayoutInfo prefetchLayout;
511  const auto *uArch = xegpu::uArch::getUArch(getChipStr(prefetch).value_or(""));
512  if (!uArch)
513  return;
514  xegpu::DistributeLayoutAttr anchorLayout = prefetch.getLayoutAttr();
515  if (hasParamsOfLayoutKind(anchorLayout)) {
516  prefetchLayout = makeLayoutInfo(anchorLayout);
517  if (layoutKind == xegpu::LayoutKind::InstData) {
518  const auto *uArchInstruction =
519  dyn_cast<xegpu::uArch::Subgroup2DBlockPrefetchInstruction>(
520  uArch->getInstruction(
521  xegpu::uArch::InstructionKind::Subgroup2DBlockPrefetch));
522  if (!uArchInstruction)
523  return;
524  auto completed = xegpu::completeBlockStoreLaneLayoutFromInstData(
525  anchorLayout, prefetch.getTensorDescType().getElementType(),
526  uArchInstruction, uArch->getSubgroupSize());
527  if (!completed) {
528  prefetch.emitWarning(
529  "Failed to identify lane layouts for the specified inst_data.");
530  return;
531  }
532  prefetch.setLayoutAttr(*completed);
533  prefetchLayout = makeLayoutInfo(*completed);
534  }
535  } else {
536  auto tdescTy = prefetch.getTensorDescType();
537  auto numSgOrErr =
538  getNumSgOrFail(prefetch, uArch->getSubgroupSize(), nullptr);
539  if (failed(numSgOrErr))
540  return;
541 
542  auto layoutAttr = xegpu::setupPrefetchNdAnchorLayout(
543  layoutKind, tdescTy, numSgOrErr.value_or(0), uArch);
544  if (!layoutAttr) {
545  prefetch.emitWarning(
546  "Failed to determine required layout for prefetch_nd.");
547  return;
548  }
549  prefetchLayout = makeLayoutInfo(layoutAttr);
550  prefetch.setLayoutAttr(layoutAttr);
551  }
552  // Propagate the layout to the source tensor descriptor.
553  propagateIfChanged(operands[0], operands[0]->meet(prefetchLayout));
554 }

556 void LayoutInfoPropagation::visitVectorMultiReductionOp(
557  vector::MultiDimReductionOp reduction,
558  ArrayRef<LayoutInfoLattice *> operands,
559  ArrayRef<const LayoutInfoLattice *> results) {
560  Type resultTy = reduction.getDestType();
561  // The layout of the result must be present.
562  LayoutInfo resLayoutInfo = results[0]->getValue();
563 
564  xegpu::DistributeLayoutAttr consumerLayoutAttr;
565  if (!resultTy.isIntOrFloat()) {
566  if (!resLayoutInfo.isAssigned())
567  return;
568  consumerLayoutAttr =
569  dyn_cast<xegpu::DistributeLayoutAttr>(resLayoutInfo.get());
570  }
571 
572  VectorType sourceTy = reduction.getSourceVectorType();
573  SmallVector<int64_t> reductionDims(reduction.getReductionDims());
574 
575  const auto *uArch =
576  xegpu::uArch::getUArch(xegpu::getChipStr(reduction).value_or(""));
577  if (!uArch)
578  return;
579 
580  auto numSgOrErr =
581  getNumSgOrFail(reduction, uArch->getSubgroupSize(), consumerLayoutAttr);
582  if (failed(numSgOrErr))
583  return;
584 
585  // The result layout represents the layout requirements of the operation.
586  // it is recorded to anchor layout or temporary layout.
587  // it must be honored for current op and may conflict with the layout
588  // propagated from consumer op, the conflict is resolved in later phase by
589  // converting the required result layout to the consumer layout
590  auto requiredResLayoutAttr = xegpu::setupMultiReductionResultLayout(
591  layoutKind, sourceTy, consumerLayoutAttr, reductionDims,
592  numSgOrErr.value_or(0), uArch);

908  }

910  if (dpasMxBScaleLayout.isAssigned())
911  propagateIfChanged(operands[idx],
912  operands[idx]->meet(dpasMxBScaleLayout));
913  ++idx;

1025 /// ConvertLayoutOp.
1026 void LayoutInfoPropagation::visitConvertLayoutOp(
1027  xegpu::ConvertLayoutOp convert, ArrayRef<LayoutInfoLattice *> operands,
1028  ArrayRef<const LayoutInfoLattice *> results) {
1029 
1030  LayoutInfo resultLayout = results[0]->getValue();
1031 
1032  // TODO: fix if one of the layouts is a slice layout
1033  auto targetLayoutAttr =
1034  dyn_cast<xegpu::LayoutAttr>(convert.getTargetLayoutAttr());

1036  auto inputLayoutAttr =
1037  dyn_cast_if_present<xegpu::LayoutAttr>(convert.getInputLayoutAttr());

1040  // Fill the lane_layout / lane_data / order parameters the target_layout is
1041  // missing from it (sg_layout / sg_data / inst_data are left as-is), so the
1042  // target stays consistent with what is actually propagated downstream.
1043  auto resultLayoutAttr = resultLayout.isAssigned()
1044  ? dyn_cast<xegpu::LayoutAttr>(resultLayout.get())
1045  : nullptr;

1053  resultLayoutAttr.getOrder());

1055  }
1056  }

1060  // as-is), so the producer side receives a fully-populated lane layout.

1062  if (layoutKind == xegpu::LayoutKind::InstData &&
1063  !inputLayoutAttr.getLaneLayout()) {

1067  targetLayoutAttr.getLaneLayout(), targetLayoutAttr.getLaneData(),
1068  targetLayoutAttr.getOrder());

1070  }
1071  }
1072 
1073  xegpu::DistributeLayoutAttr anchorLayout = convert.getEffectiveInputLayout();
1074  LayoutInfo convertLayout = makeLayoutInfo(anchorLayout);
1075  // Propagate the new layout to the tensor descriptor operand.
1076  propagateIfChanged(operands[0], operands[0]->meet(convertLayout));

1574 LogicalResult ResolveLayoutConflicts::run() {
1575  // Scan all operations in the parent op and resolve layout conflicts at
1576  // tensor descriptor and vector use points.
1577  auto r = parentOp->walk([&](Operation *op) -> WalkResult {
1578  for (OpResult result : op->getResults()) {
1579  // if the operation inputs vector and output scalar, like multi-reduction
1580  // we need to check if the result has layout and add a convert_layout to
1581  // serve as anchor op for the reduction op's layout.
1582  if (result.getType().isIntOrFloat() &&
1583  (isa<vector::MultiDimReductionOp>(op) ||
1584  isa<vector::ReductionOp>(op))) {
1585  auto res = assignResultLayout(result);
1586  if (failed(res)) {
1587  DBGS() << "Failed to assign layout for scalar consumer of reduction "
1588  << *op << "\n";
1589  return WalkResult::interrupt();
1590  }
1591  }
1592  // If the op is a region branch op with a vector result that has no uses,
1593  // we need to add a convert_layout to serve as an anchor op for the
1594  // result's layout.
1595  if (isa<VectorType>(result.getType()) && result.use_empty() &&
1596  isa<RegionBranchOpInterface>(op)) {
1597  auto res = assignResultLayout(result);
1598  if (failed(res)) {
1599  DBGS() << "Failed to assign layout for vector consumer of region op "
1600  << *op << "\n";
1601  return WalkResult::interrupt();
1602  }
1603  }
1604  }
1605  for (OpOperand &operand : op->getOpOperands()) {
1606  // Handle conflicts in tensor descriptor operands.
1607  Type operandType = operand.get().getType();
1608  if (isa<xegpu::AnchorLayoutInterface>(op) &&
1609  isa<xegpu::TensorDescType>(operandType)) {
1610  auto res = resolveTensorDescConsumer(operand);
1611  if (failed(res)) {
1612  DBGS() << "Failed to resolve tensor descriptor consumer: " << *op
1613  << "\n";
1614  return WalkResult::interrupt();
1615  }
1616  }
1617  // Handle conflicts in vector operands.
1618  if (isa<VectorType>(operandType)) {
1619  auto res = resolveVectorConsumer(operand);
1620  if (failed(res)) {
1621  DBGS() << "Failed to resolve vector consumer: " << *op << "\n";
1622  return WalkResult::interrupt();
1623  }
1624  }
1625  }
1626  return WalkResult::advance();
1627  });
1628 
1629  LLVM_DEBUG({
1630  DBGS() << "IR after resolving layout conflicts:\n";
1631  parentOp->dump();
1632  });
1633 
1634  return r.wasInterrupted() ? failure() : success();
1635 }

1649 LogicalResult
1650 ResolveLayoutConflicts::resolveVectorConsumer(OpOperand &operand) {
1651  Value vectorValue = operand.get();
1652  Operation *consumerOp = operand.getOwner();
1653  // Get the current layout of the vector value.
1654  auto producerLayout = xegpu::getDistributeLayoutAttr(vectorValue);
1655  if (!producerLayout) {
1656  if (auto vectorTy = dyn_cast<VectorType>(vectorValue.getType());
1657  vectorTy && vectorTy.getRank() > 1)
1658  consumerOp->emitWarning("Expected layout for non-1D vectors.");
1659  return success(); // uniform non-tensor-data vector does not require
1660  // layout
1661  }
1662  // getConsumerLayoutAt also covers region-carried operands (loop init and
1663  // yield operands), so a layout conflict there is reconciled below rather than
1664  // silently trusted to region forwarding.
1665  auto consumerLayout = xegpu::getConsumerLayoutAt(operand);
1666  if (!consumerLayout) {
1667  if (isa<func::ReturnOp>(consumerOp) || isa<gpu::ReturnOp>(consumerOp))
1668  return success();
1669  return consumerOp->emitError(
1670  "No consumer layout found for vector operand.");
1671  }
1672 
1673  // If layouts are same, no conflict exists, return success.
1674  if (consumerLayout.isEqualTo(producerLayout))
1675  return success();
1676 
1677  // Consumer is a convert_layout: retarget its input_layout to the producer
1678  // instead of chaining a second convert. Always safe (single source
1679  // operand).
1680  if (auto consumerConvert = dyn_cast<xegpu::ConvertLayoutOp>(consumerOp)) {
1681  consumerConvert.setInputLayoutAttr(producerLayout);
1682  return success();
1683  }
1684 
1685  // Producer is a convert_layout feeding only this use: retarget its
1686  // target_layout to the consumer instead of appending another convert.
1687  if (auto producerConvert =
1688  vectorValue.getDefiningOp<xegpu::ConvertLayoutOp>();
1689  producerConvert && vectorValue.hasOneUse()) {
1690  // Pin the effective input before retargeting target, else an omitted
1691  // input_layout would follow target and make the conversion a no-op.
1692  producerConvert.setInputLayoutAttr(
1693  producerConvert.getEffectiveInputLayout());
1694  producerConvert.setTargetLayoutAttr(consumerLayout);
1695  return success();
1696  }
1697 
1698  // If the producer is trivially rematerializable (e.g. `vector.step`, splat
1699  // `arith.constant`), clone it and stamp the consumer's expected layout on
1700  // the clone instead of inserting a `xegpu.convert_layout`. The convert
1701  // would otherwise lower to a cross-subgroup data movement through SLM at
1702  // WG-to-SG distribution time, which is more expensive than
1703  // recomputing a pure value generator.
1704  if (auto *producerOp = vectorValue.getDefiningOp();
1705  producerOp && producerOp->getNumResults() == 1 &&
1706  isa<OpResult>(vectorValue) &&
1707  xegpu::isTriviallyRematerializable(producerOp)) {
1708  builder.setInsertionPointAfter(producerOp);
1709  Operation *clone = builder.clone(*producerOp);
1710  OpResult cloneResult = clone->getResult(0);
1711  // Drop the inherited producer layout so the new layout takes effect
1712  xegpu::removeLayoutAttr(cloneResult);
1713  xegpu::setDistributeLayoutAttr(cloneResult, consumerLayout);
1714  operand.set(cloneResult);
1715  return success();
1716  }
1717 
1718  // Insert a convert_layout op to resolve the conflict.
1719  builder.setInsertionPointAfterValue(vectorValue);
1720  auto convertOp = xegpu::ConvertLayoutOp::create(
1721  builder, consumerOp->getLoc(), vectorValue.getType(), vectorValue,
1722  producerLayout, consumerLayout);

1769 /// Update an operation with the layout of its results. For a vector result a
1770 /// temporary layout attribute is added to the op; for a tensor descriptor
1771 /// result the layout is written into its type.
1772 ///
1773 /// If the global propagation left a result without a layout, forward-fill it
1774 /// locally from the operand layouts.
1775 static LogicalResult updateOpWithForwardFill(mlir::OpBuilder &builder,
1776  mlir::Operation *op,
1777  GetLayoutFnTy getLayoutOfValue) {
1778  // Iterate over all the results.
1779  for (OpResult result : op->getResults()) {
1780  Type resultType = result.getType();
1781  // Layouts are needed only for vector and tensor descriptor types.
1782  if (!isa<VectorType, xegpu::TensorDescType>(resultType))
1783  continue;
1784  // If the result has no layout but has users, emit a warning and continue.
1785  xegpu::DistributeLayoutAttr layout = getLayoutOfValue(result);
1786  if (!layout) {
1787  // Gather operand layouts, indexed by operand number.
1788  SmallVector<xegpu::DistributeLayoutAttr> srcLayouts;
1789  srcLayouts.reserve(op->getNumOperands());
1790  bool anyAssigned = false;
1791  for (Value operand : op->getOperands()) {
1792  auto srclayout = xegpu::getDistributeLayoutAttr(operand);
1793  srcLayouts.push_back(srclayout);
1794  anyAssigned |= (srclayout != nullptr);
1795  }
1796  if (anyAssigned) {
1797  layout =
1798  xegpu::inferResultLayoutFromSourceForNonAnchorOp(op, srcLayouts);
1799  }
1800  }
1801  if (!layout && result.getNumUses() > 0) {
1802  op->emitWarning("op has users but no layout assigned for its result");
1803  }
1804  // If the result is a tensor descriptor type, update the tensor desc type
1805  // with layout.
1806  if (auto tensorDescTy = dyn_cast<xegpu::TensorDescType>(resultType)) {
1807  auto typeWithLayout = xegpu::TensorDescType::get(
1808  tensorDescTy.getContext(), tensorDescTy.getShape(),
1809  tensorDescTy.getElementType(), tensorDescTy.getEncoding(), layout);
1810  result.setType(typeWithLayout);
1811  continue;
1812  }
1813  // If the result is a vector type, add a temporary layout attribute to the
1814  // op.
1815  xegpu::setDistributeLayoutAttr(result, layout);
1816  }
1817  return success();
1818 }

1820 /// Update the function arguments and results with the layouts.
1821 static LogicalResult updateFunctionOpInterface(mlir::OpBuilder &builder,
1822  mlir::FunctionOpInterface funcOp,
1823  GetLayoutFnTy getLayoutOfValue) {
1824  // Only process functions whose type is a standard MLIR FunctionType.
1825  // Functions using a different type representation (e.g. llvm.func with
1826  // LLVMFunctionType) are not targets for XeGPU layout propagation, and
1827  // calling setType(FunctionType{}) on them would corrupt their type.
1828  if (!isa<FunctionType>(funcOp.getFunctionType()))
1829  return success();
1830  SmallVector<Type> newArgTypes;
1831  // Update the function arguments.
1832  for (BlockArgument arg : funcOp.getArguments()) {
1833  Type argType = arg.getType();
1834  newArgTypes.push_back(argType);
1835  if (!isa<VectorType, xegpu::TensorDescType>(argType))
1836  continue;
1837  xegpu::DistributeLayoutAttr layout = getLayoutOfValue(arg);
1838  if (!layout) {
1839  LLVM_DEBUG(DBGS() << "Expecting layout for function argument: " << arg
1840  << " but got none.\n");
1841  return failure();
1842  }
1843  if (auto tensorDescTy = dyn_cast<xegpu::TensorDescType>(argType)) {
1844  auto newTdescTy = xegpu::TensorDescType::get(
1845  tensorDescTy.getContext(), tensorDescTy.getShape(),
1846  tensorDescTy.getElementType(), tensorDescTy.getEncoding(), layout);
1847  arg.setType(newTdescTy);
1848  newArgTypes.back() = newTdescTy;
1849  }
1850  }
1851  // Update the function type with the new argument types.
1852  // NOTE: We assume that function results are not expected to have layouts.
1853  funcOp.setType(FunctionType::get(funcOp.getContext(), newArgTypes,
1854  funcOp.getResultTypes()));
1855  return success();
1856 }

1870 LogicalResult xegpu::propagateLayouts(OpBuilder &builder, Operation *target,
1871  LayoutKind layoutKind,
1872  unsigned indexBitWidth, bool printOnly) {
1873  RunLayoutInfoPropagation analysis(target, layoutKind, indexBitWidth);
1874  // Print the analysis result and exit. (for debugging purposes)
1875  if (printOnly) {
1876  auto &os = llvm::outs();
1877  analysis.printAnalysisResult(os);
1878  return success();
1879  }
1880  // An op with no determinable layout cannot be lowered; stop before the update
1881  // walk fabricates degenerate layouts from the unlabeled values.
1882  if (analysis.hasFailed())
1883  return failure();
1884  // Helper to convert LayoutInfo to xegpu::LayoutAttr.
1885  auto getLayoutFromPropagation =
1886  [&](Value val) -> xegpu::DistributeLayoutAttr {
1887  LayoutInfo layout = analysis.getLayoutInfo(val);
1888  if (auto opResult = dyn_cast<OpResult>(val)) {
1889  Operation *defOp = opResult.getDefiningOp();
1890  if (auto anchorOp = dyn_cast<xegpu::AnchorLayoutInterface>(defOp)) {
1891  auto anchorLayout = anchorOp.getAnchorLayout();
1892  if (anchorLayout != nullptr)
1893  return anchorLayout;
1894  }
1895  xegpu::DistributeLayoutAttr requiredResLayoutAttr =
1896  xegpu::getTemporaryLayout(opResult);
1897  if (requiredResLayoutAttr != nullptr)
1898  return requiredResLayoutAttr;
1899  }
1900  if (!layout.isAssigned())
1901  return {};
1902  xegpu::DistributeLayoutAttr layoutAttr =
1903  cast<xegpu::DistributeLayoutAttr>(layout.get());
1904  if (layout.isSliceLayout())
1905  return cast<xegpu::SliceAttr>(layoutAttr);
1906 
1907  return cast<xegpu::LayoutAttr>(layoutAttr);
1908  };
1909 
1910  Operation *op = target;
1911  auto walkResult = op->walk([&](mlir::Block *block) -> WalkResult {
1912  for (mlir::Operation &op : block->getOperations()) {
1913  LogicalResult r = success();
1914  TypeSwitch<Operation *>(&op)
1915  .Case([&](mlir::RegionBranchTerminatorOpInterface branchTermOp) {
1916  r = xegpu::propagateYieldOperandsToRegionResults(
1917  branchTermOp, getLayoutFromPropagation);
1918  })
1919  .Case([&](mlir::RegionBranchOpInterface branchOp) {
1920  r = xegpu::propagateRegionArgsToInits(branchOp,
1921  getLayoutFromPropagation);
1922  })
1923  .Case([&](mlir::FunctionOpInterface funcOp) {
1924  r = updateFunctionOpInterface(builder, funcOp,
1925  getLayoutFromPropagation);
1926  })
1927  .Default([&](Operation *op) {
1928  r = updateOpWithForwardFill(builder, op, getLayoutFromPropagation);
1929  });
1930  if (failed(r)) {
1931  op.emitError("Failed to update operation with the layout.");
1932  return WalkResult::interrupt();
1933  }
1934  }
1935  return WalkResult::advance();
1936  });
1937  if (walkResult.wasInterrupted())
1938  return failure();
1939 
1940  return success();
1941 }
