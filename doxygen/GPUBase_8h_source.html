51 /// MMAMatrixType storage and uniquing. Array is uniqued based on its shape
52 /// and type.
53 struct MMAMatrixStorageType : public TypeStorage {
54  MMAMatrixStorageType(unsigned numDims, const int64_t *dimShapes,
55  Type elementType, StringRef operand)
56  : dimShapes(dimShapes), numDims(numDims), elementType(elementType),
57  operand(operand) {}
58 
59  /// The hash key for uniquing.
60  using KeyTy = std::tuple<ArrayRef<int64_t>, Type, StringRef>;
61  bool operator==(const KeyTy &key) const {
62  return key == KeyTy(getShape(), elementType, operand);
63  }
64 
65  /// Construction.
66  static MMAMatrixStorageType *construct(TypeStorageAllocator &allocator,
67  const KeyTy &key) {
68  ArrayRef<int64_t> shape = allocator.copyInto(std::get<0>(key));
69  StringRef operand = allocator.copyInto(std::get<2>(key));
70 
71  return new (allocator.allocate<MMAMatrixStorageType>())
72  MMAMatrixStorageType(shape.size(), shape.data(), std::get<1>(key),
73  operand);
74  }
75 
76  ArrayRef<int64_t> getShape() const {
77  return ArrayRef<int64_t>(dimShapes, numDims);
78  }
79 
80  StringRef getOperand() const { return operand; }
81 
82  /// Reference to the shape of the MMA matrix.
83  const int64_t *dimShapes;
84 
85  /// Number of dimensions in the MMA matrix.
86  unsigned numDims;
87 
88  /// Element type of elements held in the MMA matrix.
89  Type elementType;
90 
91  /// MMA operand that this MMAMatrix holds. The general form of operation this
92  /// type supports is given by the equation C += A*B. This field specifies
93  /// which operand in the given equation is held by this type. The valid values
94  /// are "AOp", "BOp" and "COp".
95  StringRef operand;
96 };

98 /// MMAMatrix represents a matrix held by a subgroup for matrix-matrix multiply
99 /// accumulate operations. MMAMatrices are taken as direct operands by these
100 /// operations and are also produced as results. These matrices are meant to
101 /// reside in the registers. A limited number of pointwise operations can be
102 /// performed on these matrices, i.e., operations which operate uniformly on
103 /// all the elements in the matrix and do not change the order of matrix
104 /// elements. The above conditions exist because the layout of matrix elements
105 /// inside the matrix is opaque i.e., the elements may be present in the
106 /// matrix in any order. The general usage of this type is shown as follows:-
107 ///
108 /// %0 = gpu.subgroup_mma_load_matrix %arg0[%c0, %c0] {leadDimension = 16 :
109 /// index} : memref<16x16xf16> -> !gpu.mma_matrix<16x16xf16, "AOp">
110 ///
111 /// The MMAMatrixType describes the shape of the matrix being loaded and the
112 /// operand being loaded too. The operand needs to be specified to aid the
113 /// lowering of this type to dialects such as NVVM where each workitem may
114 /// hold different amount of elements depending on the elementType of the
115 /// matrix. For e.g., Each workitem holds 4 vector<2xf16>s for f16 data type
116 /// and 8 f32s for f32 data type of MMAMatrix. Some other instances of usage
117 /// are:-
118 ///
119 /// %3 = gpu.subgroup_mma_compute %0, %1, %2 :
120 /// !gpu.mma_matrix<16x16xf16, "AOp">, !gpu.mma_matrix<16x16xf16, "BOp">
121 /// -> !gpu.mma_matrix<16x16xf32, "COp">
122 ///
123 ///
124 /// gpu.subgroup_mma_store_matrix %3, %arg22[%c0, %c0] {leadDimension = 16
125 /// : index}: !gpu.mma_matrix<16x16xf32, "COp">, memref<16x16xf32>
126 ///
127 /// MMA matrices may be memref elements on targets that support storing them,
128 /// for example SPIRV can represent arrays of MMA matrices.
129 // TODO: consider moving this to ODS.
