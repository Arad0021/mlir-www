64 ParseResult AllocaOp::parse(OpAsmParser &parser, OperationState &result) {
65  OpAsmParser::UnresolvedOperand arraySize;
66  Type type, elemType;
67  SMLoc trailingTypeLoc;
68 
69  if (succeeded(parser.parseOptionalKeyword("inalloca")))
70  result.addAttribute(getInallocaAttrName(result.name),
71  UnitAttr::get(parser.getContext()));
72 
73  if (parser.parseOperand(arraySize) || parser.parseKeyword("x") ||
74  parser.parseType(elemType) ||
75  parser.parseOptionalAttrDict(result.attributes) || parser.parseColon() ||
76  parser.getCurrentLocation(&trailingTypeLoc) || parser.parseType(type))
77  return failure();
78 
79  std::optional<NamedAttribute> alignmentAttr =
80  result.attributes.getNamed("alignment");
81  if (alignmentAttr.has_value()) {
82  auto alignmentInt = llvm::dyn_cast<IntegerAttr>(alignmentAttr->getValue());
83  if (!alignmentInt)
84  return parser.emitError(parser.getNameLoc(),
85  "expected integer alignment");
86  if (alignmentInt.getValue().isZero())
87  result.attributes.erase("alignment");
88  }
89 
90  // Extract the result type from the trailing function type.
91  auto funcType = llvm::dyn_cast<FunctionType>(type);
92  if (!funcType || funcType.getNumInputs() != 1 ||
93  funcType.getNumResults() != 1)
94  return parser.emitError(
95  trailingTypeLoc,
96  "expected trailing function type with one argument and one result");
97 
98  if (parser.resolveOperand(arraySize, funcType.getInput(0), result.operands))
99  return failure();
100 
101  Type resultType = funcType.getResult(0);
102  if (auto ptrResultType = llvm::dyn_cast<LLVMPointerType>(resultType))
103  result.addAttribute(kElemTypeAttrName, TypeAttr::get(elemType));
104 
105  result.addTypes({funcType.getResult(0)});
106  return success();
107 }

119 LogicalResult AllocaOp::canonicalize(AllocaOp op, PatternRewriter &rewriter) {
120  // Convert `alloca Ty, C` to the canonical `alloca [C x Ty], 1` form.
121  APInt numElements;
122  if (!matchPattern(op.getArraySize(), m_ConstantInt(&numElements)) ||
123  numElements.isOne() || numElements.getActiveBits() > 64)
124  return failure();
125 
126  auto arrayType =
127  LLVMArrayType::get(op.getElemType(), numElements.getZExtValue());
128  Value one = ConstantOp::create(rewriter, op.getLoc(), rewriter.getI32Type(),
129  /*value=*/1);
130  auto newAlloca =
131  AllocaOp::create(rewriter, op.getLoc(), op.getType(), one,
132  op.getAlignmentAttr(), arrayType, op.getInalloca());
133  newAlloca->setDiscardableAttrs(op->getDiscardableAttrDictionary());
134  rewriter.replaceOp(op, newAlloca);
135  return success();
136 }

154 /// Destructures the 'indices' parameter into 'rawConstantIndices' and
155 /// 'dynamicIndices', encoding the former in the process. In the process,
156 /// dynamic indices which are used to index into a structure type are converted
157 /// to constant indices when possible. To do this, the GEPs element type should
158 /// be passed as first parameter.
159 static void destructureIndices(Type currType, ArrayRef<GEPArg> indices,
160  SmallVectorImpl<int32_t> &rawConstantIndices,
161  SmallVectorImpl<Value> &dynamicIndices) {
162  for (const GEPArg &iter : indices) {
163  // If the thing we are currently indexing into is a struct we must turn
164  // any integer constants into constant indices. If this is not possible
165  // we don't do anything here. The verifier will catch it and emit a proper
166  // error. All other canonicalization is done in the fold method.
167  bool requiresConst = !rawConstantIndices.empty() &&
168  isa_and_nonnull<LLVMStructType>(currType);
169  if (Value val = llvm::dyn_cast_if_present<Value>(iter)) {
170  APInt intC;
171  if (requiresConst && matchPattern(val, m_ConstantInt(&intC)) &&
172  intC.isSignedIntN(kGEPConstantBitWidth)) {
173  rawConstantIndices.push_back(intC.getSExtValue());
174  } else {
175  rawConstantIndices.push_back(GEPOp::kDynamicIndex);
176  dynamicIndices.push_back(val);
177  }
178  } else {
179  rawConstantIndices.push_back(cast<GEPConstantIndex>(iter));
180  }
181 
182  // Skip for very first iteration of this loop. First index does not index
183  // within the aggregates, but is just a pointer offset.
184  if (rawConstantIndices.size() == 1 || !currType)
185  continue;
186 
187  currType = TypeSwitch<Type, Type>(currType)
188  .Case<VectorType, LLVMArrayType>([](auto containerType) {
189  return containerType.getElementType();
190  })
191  .Case([&](LLVMStructType structType) -> Type {
192  int64_t memberIndex = rawConstantIndices.back();
193  if (memberIndex >= 0 && static_cast<size_t>(memberIndex) <
194  structType.getBody().size())
195  return structType.getBody()[memberIndex];
196  return nullptr;
197  })
198  .Default(nullptr);
199  }
200 }

230 ParseResult mlir::LLVM::parseGEPIndices(
231  OpAsmParser &parser,
232  SmallVectorImpl<OpAsmParser::UnresolvedOperand> &indices,
233  DenseI32ArrayAttr &rawConstantIndices) {
234  SmallVector<int32_t> constantIndices;
235 
236  auto idxParser = [&]() -> ParseResult {
237  int32_t constantIndex;
238  OptionalParseResult parsedInteger =
239  parser.parseOptionalInteger(constantIndex);
240  if (parsedInteger.has_value()) {
241  if (failed(parsedInteger.value()))
242  return failure();
243  constantIndices.push_back(constantIndex);
244  return success();
245  }
246 
247  constantIndices.push_back(LLVM::GEPOp::kDynamicIndex);
248  return parser.parseOperand(indices.emplace_back());
249  };
250  if (parser.parseCommaSeparatedList(idxParser))
251  return failure();
252 
253  rawConstantIndices =
254  DenseI32ArrayAttr::get(parser.getContext(), constantIndices);
255  return success();
256 }

271 /// For the given `indices`, check if they comply with `baseGEPType`,
272 /// especially check against LLVMStructTypes nested within.
273 static LogicalResult
274 verifyStructIndices(Type baseGEPType, unsigned indexPos,
275  GEPIndicesAdaptor<ValueRange> indices,
276  function_ref<InFlightDiagnostic()> emitOpError) {
277  if (indexPos >= indices.size())
278  // Stop searching
279  return success();
280 
281  return TypeSwitch<Type, LogicalResult>(baseGEPType)
282  .Case([&](LLVMStructType structType) -> LogicalResult {
283  auto attr = dyn_cast<IntegerAttr>(indices[indexPos]);
284  if (!attr)
285  return emitOpError() << "expected index " << indexPos
286  << " indexing a struct to be constant";
287 
288  int32_t gepIndex = attr.getInt();
289  ArrayRef<Type> elementTypes = structType.getBody();
290  if (gepIndex < 0 ||
291  static_cast<size_t>(gepIndex) >= elementTypes.size())
292  return emitOpError() << "index " << indexPos
293  << " indexing a struct is out of bounds";
294 
295  // Instead of recursively going into every children types, we only
296  // dive into the one indexed by gepIndex.
297  return verifyStructIndices(elementTypes[gepIndex], indexPos + 1,
298  indices, emitOpError);
299  })
300  .Case<VectorType, LLVMArrayType>(
301  [&](auto containerType) -> LogicalResult {
302  return verifyStructIndices(containerType.getElementType(),
303  indexPos + 1, indices, emitOpError);
304  })
305  .Default([&](auto otherType) -> LogicalResult {
306  return emitOpError()
307  << "type " << otherType << " cannot be indexed (index #"
308  << indexPos << ")";
309  });
310 }

319 LogicalResult LLVM::GEPOp::verify() {
320  if (static_cast<size_t>(
321  llvm::count(getRawConstantIndices(), kDynamicIndex)) !=
322  getDynamicIndices().size())
323  return emitOpError("expected as many dynamic indices as specified in '")
324  << getRawConstantIndicesAttrName().getValue() << "'";
325 
326  if (getNoWrapFlags() == GEPNoWrapFlags::inboundsFlag)
327  return emitOpError("'inbounds_flag' cannot be used directly.");
328 
329  // LLVM stores `inrange` only on GetElementPtrConstantExpr, whose operands
330  // are already Constants. This dialect has one GEP operation, so proving the
331  // base and indices are constant expressions means walking SSA. A local check
332  // rejects valid cases such as a nested constant GEP base or a `ptrtoint`
333  // index.
334  if (auto inrange = getInrangeAttr()) {
335  auto pointerType =
336  cast<LLVMPointerType>(extractVectorElementType(getBase().getType()));
337  DataLayout dataLayout = DataLayout::closest(*this);
338  std::optional<uint64_t> indexWidth =
339  dataLayout.getTypeIndexBitwidth(pointerType);
340  assert(indexWidth && "pointers always return an index bitwidth");
341  if (inrange.getLower().getBitWidth() != *indexWidth)
342  return emitOpError("'inrange' bitwidth ")
343  << inrange.getLower().getBitWidth()
344  << " must match the pointer index bitwidth (" << *indexWidth
345  << ") specified in the datalayout";
346  if (inrange.getLower().sge(inrange.getUpper()))
347  return emitOpError("expected 'inrange' end to be larger than start");
348  }
349 
350  return verifyStructIndices(getElemType(), getIndices(),
351  [&] { return emitOpError(); });
352 }

374 /// Returns true if the given type is supported by atomic operations. All
375 /// integer, float, and pointer types with a power-of-two bitsize and a minimal
376 /// size of 8 bits are supported.
377 bool LLVM::isTypeCompatibleWithAtomicOp(Type type,
378  const DataLayout &dataLayout) {
379  if (!isa<IntegerType, LLVMPointerType>(type))
380  if (!isCompatibleFloatingPointType(type))
381  return false;
382 
383  llvm::TypeSize bitWidth = dataLayout.getTypeSizeInBits(type);
384  if (bitWidth.isScalable())
385  return false;
386  // Needs to be at least 8 bits and a power of two.
387  return bitWidth >= 8 && (bitWidth & (bitWidth - 1)) == 0;
388 }

390 /// Verifies the attributes and the type of atomic memory access operations.
391 template <typename OpTy>
392 static LogicalResult
393 verifyAtomicMemOp(OpTy memOp, Type valueType,
394  ArrayRef<AtomicOrdering> unsupportedOrderings) {
395  if (memOp.getOrdering() != AtomicOrdering::not_atomic) {
396  DataLayout dataLayout = DataLayout::closest(memOp);
397  if (!isTypeCompatibleWithAtomicOp(valueType, dataLayout))
398  return memOp.emitOpError("unsupported type ")

401  return memOp.emitOpError("unsupported ordering '")
402  << stringifyAtomicOrdering(memOp.getOrdering()) << "'";
403  if (!memOp.getAlignment())
404  return memOp.emitOpError("expected alignment for atomic access");
405  return success();
406  }
407  if (memOp.getSyncscope())
408  return memOp.emitOpError(
409  "expected syncscope to be null for non-atomic access");
410  return success();
411 }

520  return emitOpError() << "expected at least '"

752 /// Checks if `type` can be used in any kind of conversion sequences.
753 static bool isSupportedTypeForConversion(Type type) {
754  // Aggregate types are not bitcastable.
755  if (isa<LLVM::LLVMStructType, LLVM::LLVMArrayType>(type))
756  return false;
757 
758  if (auto vectorType = dyn_cast<VectorType>(type)) {
759  // Vectors of pointers cannot be casted.
760  if (isa<LLVM::LLVMPointerType>(vectorType.getElementType()))
761  return false;
762  // Scalable types are not supported.
763  return !vectorType.isScalable();
764  }
765  return true;
766 }

768 /// Checks that `rhs` can be converted to `lhs` by a sequence of casts and
769 /// truncations. Checks for narrowing or widening conversion compatibility
770 /// depending on `narrowingConversion`.
771 static bool areConversionCompatible(const DataLayout &layout, Type targetType,
772  Type srcType, bool narrowingConversion) {
773  if (targetType == srcType)
774  return true;
775 
776  if (!isSupportedTypeForConversion(targetType) ||
777  !isSupportedTypeForConversion(srcType))
778  return false;
779 
780  uint64_t targetSize = layout.getTypeSize(targetType);
781  uint64_t srcSize = layout.getTypeSize(srcType);
782 
783  // Pointer casts will only be sane when the bitsize of both pointer types is
784  // the same.
785  if (isa<LLVM::LLVMPointerType>(targetType) &&
786  isa<LLVM::LLVMPointerType>(srcType))
787  return targetSize == srcSize;
788 
789  if (narrowingConversion)
790  return targetSize <= srcSize;
791  return targetSize >= srcSize;
792 }

800 /// Converts a value to an integer type of the same size.
801 /// Assumes that the type can be converted.
802 static Value castToSameSizedInt(OpBuilder &builder, Location loc, Value val,
803  const DataLayout &dataLayout) {

833 /// Constructs operations that convert `srcValue` into a new value of type
834 /// `targetType`. Assumes the types have the same bitsize.
835 static Value castSameSizedTypes(OpBuilder &builder, Location loc,
836  Value srcValue, Type targetType,
837  const DataLayout &dataLayout) {
838  Type srcType = srcValue.getType();
839  assert(areConversionCompatible(dataLayout, targetType, srcType,
840  /*narrowingConversion=*/true) &&
841  "expected that the compatibility was checked before");
842 
843  // Nothing has to be done if the types are already the same.
844  if (srcType == targetType)
845  return srcValue;
846 
847  // In the special case of casting one pointer to another, we want to generate
848  // an address space cast. Bitcasts of pointers are not allowed and using
849  // pointer to integer conversions are not equivalent due to the loss of
850  // provenance.
851  if (isa<LLVM::LLVMPointerType>(targetType) &&
852  isa<LLVM::LLVMPointerType>(srcType))
853  return builder.createOrFold<LLVM::AddrSpaceCastOp>(loc, targetType,
854  srcValue);
855 
856  // For all other castable types, casting through integers is necessary.
857  Value replacement = castToSameSizedInt(builder, loc, srcValue, dataLayout);
858  return castIntValueToSameSizedType(builder, loc, replacement, targetType);
859 }

861 /// Constructs operations that convert `srcValue` into a new value of type
862 /// `targetType`. Performs bit-level extraction if the source type is larger
863 /// than the target type. Assumes that this conversion is possible.
864 static Value createExtractAndCast(OpBuilder &builder, Location loc,
865  Value srcValue, Type targetType,
866  const DataLayout &dataLayout) {
867  // Get the types of the source and target values.
868  Type srcType = srcValue.getType();
869  assert(areConversionCompatible(dataLayout, targetType, srcType,
870  /*narrowingConversion=*/true) &&
871  "expected that the compatibility was checked before");
872 
873  // Nothing has to be done if the types are already the same. This also
874  // avoids querying the bit size of scalable vector types below.
875  if (srcType == targetType)
876  return srcValue;
877 
878  uint64_t srcTypeSize = dataLayout.getTypeSizeInBits(srcType);
879  uint64_t targetTypeSize = dataLayout.getTypeSizeInBits(targetType);
880  if (srcTypeSize == targetTypeSize)
881  return castSameSizedTypes(builder, loc, srcValue, targetType, dataLayout);
882 
883  // First, cast the value to a same-sized integer type.
884  Value replacement = castToSameSizedInt(builder, loc, srcValue, dataLayout);
885 
886  // Truncate the integer if the size of the target is less than the value.
887  if (isBigEndian(dataLayout)) {
888  uint64_t shiftAmount = srcTypeSize - targetTypeSize;
889  auto shiftConstant = LLVM::ConstantOp::create(
890  builder, loc, builder.getIntegerAttr(srcType, shiftAmount));
891  replacement =
892  builder.createOrFold<LLVM::LShrOp>(loc, srcValue, shiftConstant);
893  }
894 
895  replacement = LLVM::TruncOp::create(
896  builder, loc, builder.getIntegerType(targetTypeSize), replacement);
897 
898  // Now cast the integer to the actual target type if required.
899  return castIntValueToSameSizedType(builder, loc, replacement, targetType);
900 }

902 /// Constructs operations that insert the bits of `srcValue` into the
903 /// "beginning" of `reachingDef` (beginning is endianness dependent).
904 /// Assumes that this conversion is possible.
905 static Value createInsertAndCast(OpBuilder &builder, Location loc,
906  Value srcValue, Value reachingDef,
907  const DataLayout &dataLayout) {
908 
909  assert(areConversionCompatible(dataLayout, reachingDef.getType(),
910  srcValue.getType(),
911  /*narrowingConversion=*/false) &&
912  "expected that the compatibility was checked before");
913 
914  // Nothing has to be done if the types are already the same. This also
915  // avoids querying the bit size of scalable vector types below.
916  if (srcValue.getType() == reachingDef.getType())
917  return srcValue;
918 
919  uint64_t valueTypeSize = dataLayout.getTypeSizeInBits(srcValue.getType());
920  uint64_t slotTypeSize = dataLayout.getTypeSizeInBits(reachingDef.getType());
921  if (slotTypeSize == valueTypeSize)
922  return castSameSizedTypes(builder, loc, srcValue, reachingDef.getType(),
923  dataLayout);
924 
925  // In the case where the store only overwrites parts of the memory,
926  // bit fiddling is required to construct the new value.
927 
928  // First convert both values to integers of the same size.
929  Value defAsInt = castToSameSizedInt(builder, loc, reachingDef, dataLayout);
930  Value valueAsInt = castToSameSizedInt(builder, loc, srcValue, dataLayout);
931  // Extend the value to the size of the reaching definition.
932  valueAsInt =
933  builder.createOrFold<LLVM::ZExtOp>(loc, defAsInt.getType(), valueAsInt);
934  uint64_t sizeDifference = slotTypeSize - valueTypeSize;
935  if (isBigEndian(dataLayout)) {
936  // On big endian systems, a store to the base pointer overwrites the most
937  // significant bits. To accomodate for this, the stored value needs to be
938  // shifted into the according position.
939  Value bigEndianShift = LLVM::ConstantOp::create(
940  builder, loc,
941  builder.getIntegerAttr(defAsInt.getType(), sizeDifference));
942  valueAsInt =
943  builder.createOrFold<LLVM::ShlOp>(loc, valueAsInt, bigEndianShift);
944  }
945 
946  // Construct the mask that is used to erase the bits that are overwritten by
947  // the store.
948  APInt maskValue;
949  if (isBigEndian(dataLayout)) {
950  // Build a mask that has the most significant bits set to zero.
951  // Note: This is the same as 2^sizeDifference - 1
952  maskValue = APInt::getAllOnes(sizeDifference).zext(slotTypeSize);
953  } else {
954  // Build a mask that has the least significant bits set to zero.
955  // Note: This is the same as -(2^valueTypeSize)
956  maskValue = APInt::getAllOnes(valueTypeSize).zext(slotTypeSize);
957  maskValue.flipAllBits();
958  }
959 
960  // Mask out the affected bits ...
961  Value mask = LLVM::ConstantOp::create(
962  builder, loc, builder.getIntegerAttr(defAsInt.getType(), maskValue));
963  Value masked = builder.createOrFold<LLVM::AndOp>(loc, defAsInt, mask);
964 
965  // ... and combine the result with the new value.
966  Value combined = builder.createOrFold<LLVM::OrOp>(loc, masked, valueAsInt);
967 
968  return castIntValueToSameSizedType(builder, loc, combined,
969  reachingDef.getType());
970 }

1182 /// pointer by. Returns nullopt if no constant offset could be computed.
1183 static std::optional<uint64_t> gepToByteOffset(const DataLayout &dataLayout,
1184  LLVM::GEPOp gep) {

1186  SmallVector<uint64_t> indices;
1187  for (auto index : gep.getIndices()) {
1188  auto constIndex = dyn_cast<IntegerAttr>(index);
1189  if (!constIndex)

1193  if (gepIndex < 0)
1194  return {};
1195  indices.push_back(gepIndex);

1205  offset +=

1211  offset += structType.getElementOffset(dataLayout, index);

1214  })

1239 /// Computes subslot access information for an access into `slot` with the given
1240 /// offset.
1241 /// Returns nullopt when the offset is out-of-bounds or when the access is into
1242 /// the padding of `slot`.
1243 static std::optional<SubslotAccessInfo>
1244 getSubslotAccessInfo(const DestructurableMemorySlot &slot,
1245  const DataLayout &dataLayout, LLVM::GEPOp gep) {
1246  std::optional<uint64_t> offset = gepToByteOffset(dataLayout, gep);
1247  if (!offset)
1248  return {};
1249 
1250  // Helper to check that a constant index is in the bounds of the GEP index
1251  // representation. LLVM dialects's GEP arguments have a limited bitwidth, thus
1252  // this additional check is necessary.
1253  auto isOutOfBoundsGEPIndex = [](uint64_t index) {
1254  return index >= (1 << LLVM::kGEPConstantBitWidth);
1255  };
1256 
1257  Type type = slot.valueType;
1258  if (*offset >= dataLayout.getTypeSize(type))
1259  return {};
1260  return TypeSwitch<Type, std::optional<SubslotAccessInfo>>(type)
1261  .Case([&](LLVM::LLVMArrayType arrayType)
1262  -> std::optional<SubslotAccessInfo> {
1263  // Find which element of the array contains the offset.
1264  uint64_t elemSize = dataLayout.getTypeSize(arrayType.getElementType());
1265  uint64_t index = *offset / elemSize;
1266  if (isOutOfBoundsGEPIndex(index))
1267  return {};
1268  return SubslotAccessInfo{static_cast<uint32_t>(index),
1269  *offset - (index * elemSize)};
1270  })
1271  .Case([&](LLVM::LLVMStructType structType)
1272  -> std::optional<SubslotAccessInfo> {
1273  // Walk over the elements of the struct to find in which of
1274  // them the offset is.
1275  for (auto [index, elem] : llvm::enumerate(structType.getBody())) {
1276  uint64_t elemOffset = structType.getElementOffset(dataLayout, index);
1277  // If the offset is in padding, cancel the rewrite.
1278  if (*offset < elemOffset)
1279  return {};
1280 
1281  if (*offset < elemOffset + dataLayout.getTypeSize(elem)) {
1282  if (isOutOfBoundsGEPIndex(index))
1283  return {};
1284  // The offset is within this element, stop iterating the
1285  // struct and return the index.
1286  return SubslotAccessInfo{static_cast<uint32_t>(index),
1287  *offset - elemOffset};
1288  }
1289  }
1290 
1291  return {};
1292  });
1293 }
