26 /// Infer the result ranges for the NVVM SpecialRangeableRegisterOp that might
27 /// have ConstantRangeAttr.
28 static void nvvmInferResultRanges(std::optional<LLVM::ConstantRangeAttr> range,
29  Value result,
30  ArrayRef<::mlir::ConstantIntRanges> argRanges,
31  SetIntRangeFn setResultRanges) {
32  if (range) {
33  // ConstantRange uses a half-open interval of bit patterns, whereas
34  // ConstantIntRanges uses inclusive signed and unsigned bounds. Derive
35  // each interpretation separately to handle wrapping and sign boundaries.
36  llvm::ConstantRange valueRange(range->getLower(), range->getUpper());
37  if (!valueRange.isEmptySet()) {
38  setResultRanges(result,
39  {valueRange.getUnsignedMin(), valueRange.getUnsignedMax(),
40  valueRange.getSignedMin(), valueRange.getSignedMax()});
41  return;
42  }
43  }
44  // ConstantIntRanges cannot represent an empty set. Conservatively fall back
45  // to the full range rather than deriving invalid inclusive bounds.
46  setResultRanges(result, IntegerValueRange::getMaxRange(result).getValue());
47 }

49 /// Verify the range attribute satisfies LLVM ConstantRange constructor
50 /// requirements for NVVM SpecialRangeableRegisterOp.
51 static LogicalResult
52 verifyConstantRangeAttr(Operation *op,
53  std::optional<LLVM::ConstantRangeAttr> rangeAttr) {
54  if (!rangeAttr)
55  return success();
56 
57  const llvm::APInt &lower = rangeAttr->getLower();
58  const llvm::APInt &upper = rangeAttr->getUpper();
59 
60  // Check LLVM ConstantRange constructor condition
61  if (lower == upper && !lower.isMaxValue() && !lower.isMinValue()) {
62  unsigned bitWidth = lower.getBitWidth();
63  llvm::APInt minVal = llvm::APInt::getMinValue(bitWidth);
64  llvm::APInt maxVal = llvm::APInt::getMaxValue(bitWidth);
65  return op->emitOpError(
66  "invalid range attribute: Lower == Upper, but they aren't min (")
67  << llvm::toString(minVal, 10, false) << ") or max ("
68  << llvm::toString(maxVal, 10, false)
69  << ") value! This is an invalid constant range.";
70  }
71 
72  return success();
73 }
