61 LogicalResult NVVM::PrefetchOp::verify() {
62  using MemSpace = NVVM::NVVMMemorySpace;
63  using CacheLevel = NVVM::PrefetchCacheLevel;
64 
65  unsigned addressSpace =
66  llvm::cast<LLVM::LLVMPointerType>(getAddr().getType()).getAddressSpace();
67  std::optional<NVVM::CacheEvictionPriority> evictPriority = getEvictPriority();
68  std::optional<NVVM::PrefetchCacheLevel> cacheLevel = getCacheLevel();
69 
70  if (getTensormap() && cacheLevel)
71  return emitOpError("cannot specify both tensormap and cache level");
72 
73  if (getTensormap()) {
74  if (addressSpace != MemSpace::Generic &&
75  addressSpace != MemSpace::Constant) {
76  return emitOpError(
77  "prefetch tensormap requires a generic or constant pointer");
78  }
79 
80  if (evictPriority) {
81  return emitOpError(
82  "prefetch tensormap does not support eviction priority");
83  }
84 
85  if (getInParamSpace() && addressSpace != MemSpace::Generic) {
86  return emitOpError(
87  "in_param_space can only be specified for a generic pointer");
88  }
89 
90  } else if (cacheLevel) {
91  if (addressSpace != MemSpace::Generic && addressSpace != MemSpace::Global &&
92  addressSpace != MemSpace::Local) {
93  return emitOpError("prefetch to cache level requires a generic, global, "
94  "or local pointer");
95  }
96 
97  if (getUniform()) {
98  if (*cacheLevel != CacheLevel::L1) {
99  return emitOpError(
100  "unsupported cache level, the only supported uniform "
101  "cache level is L1");
102  }
103 
104  if (addressSpace != MemSpace::Generic) {
105  return emitOpError(
106  "prefetch to uniform cache requires a generic pointer");
107  }
108  }
109 
110  if (evictPriority) {
111  if (*cacheLevel != CacheLevel::L2)
112  return emitOpError(
113  "cache eviction priority supported only for cache level L2");
114 
115  if (addressSpace != MemSpace::Global)
116  return emitOpError("cache eviction priority requires a global pointer");
117 
118  if (*evictPriority != NVVM::CacheEvictionPriority::EvictNormal &&
119  *evictPriority != NVVM::CacheEvictionPriority::EvictLast)
120  return emitOpError(
121  "unsupported cache eviction priority, only evict_last and "
122  "evict_normal are supported");
123  }
124 
125  if (getPredicate())
126  return emitOpError("predicate supported only on prefetch tensormap");
127 
128  } else {
129  return emitOpError(
130  "requires specification of either cache level or tensormap");
131  }
132 
133  return success();
134 }

136 LogicalResult ShflOp::verify() {
137  auto returnStructType = llvm::dyn_cast<LLVM::LLVMStructType>(getType());
138 
139  auto verifyTypeError = [&](Twine desc, Type expectedType,
140  Type actualType) -> LogicalResult {
141  return emitOpError("expected " + desc + " to be of type ")
142  << expectedType << " but got " << actualType << " instead";
143  };
144 
145  if (returnStructType) {
146  if (!getReturnValueAndIsValid())
147  return emitOpError("\"return_value_and_is_valid\" attribute must be "
148  "specified when the return type is a struct type");
149 
150  if (returnStructType.getBody().size() != 2)
151  return emitOpError("expected return type to be a two-element struct");
152 
153  llvm::ArrayRef<Type> returnStruct = returnStructType.getBody();
154  auto resultType = returnStruct[0];
155  if (resultType != getVal().getType())
156  return verifyTypeError("first element in the returned struct",
157  getVal().getType(), resultType);
158 
159  auto predicateType = returnStruct[1];
160  if (!predicateType.isInteger(1))
161  return verifyTypeError("second element in the returned struct",
162  mlir::IntegerType::get(getContext(), 1),
163  predicateType);
164  } else {
165  if (getReturnValueAndIsValid())
166  return emitOpError("expected return type to be a two-element struct");
167 
168  if (getType() != getVal().getType())
169  return verifyTypeError("return type", getVal().getType(), getType());
170  }
171  return success();
172 }

215 mlir::NVVM::IDArgPair AsyncStoreGlobalOp::getIntrinsicIDAndArgs(
216  Operation &op, LLVM::ModuleTranslation &mt, llvm::IRBuilderBase &builder) {
217  using IDArgPair = mlir::NVVM::IDArgPair;
218  auto thisOp = cast<NVVM::AsyncStoreGlobalOp>(op);
219  mlir::NVVM::MemScopeKind scope = thisOp.getScope();
220  bool isMmio = thisOp.getMmio();
221 
222  llvm::Value *addr = mt.lookupValue(thisOp.getAddr());
223  llvm::Value *value = mt.lookupValue(thisOp.getValue());
224  llvm::Value *isMultimem = builder.getInt1(thisOp.getMultimem());
225 
226  if (scope == MemScopeKind::SYS) {
227  return isMmio ? IDArgPair(llvm::Intrinsic::nvvm_st_async_mmio_sys,
228  {addr, value})
229  : IDArgPair(llvm::Intrinsic::nvvm_st_async_sys,
230  {addr, value, isMultimem});
231  } else if (scope == MemScopeKind::GPU) {
232  return IDArgPair(llvm::Intrinsic::nvvm_st_async_gpu,
233  {addr, value, isMultimem});
234  }
235  llvm_unreachable("unsupported scope for AsyncStoreGlobalOp");
236 }

244 NVVM::IDArgPair
245 PrefetchOp::getIntrinsicIDAndArgs(NVVM::PrefetchOp &op,
246  LLVM::ModuleTranslation &mt,
247  llvm::IRBuilderBase &builder) {
248  using MemSpace = NVVM::NVVMMemorySpace;
249  using CacheLevel = NVVM::PrefetchCacheLevel;
250 
251  std::optional<NVVM::PrefetchCacheLevel> cacheLevel = op.getCacheLevel();
252  std::optional<NVVM::CacheEvictionPriority> evictPriority =
253  op.getEvictPriority();
254  unsigned addressSpace =
255  llvm::cast<LLVM::LLVMPointerType>(op.getAddr().getType())
256  .getAddressSpace();
257 
258  llvm::SmallVector<llvm::Value *> args;
259  llvm::Value *addr = mt.lookupValue(op.getAddr());
260  args.push_back(op.getInParamSpace() ? getParamCastedAddr(addr, builder)
261  : addr);
262 
263  if (op.getTensormap())
264  return {llvm::Intrinsic::nvvm_prefetch_tensormap, args};
265 
266  assert(cacheLevel && "expected cache level for non-tensormap prefetch");
267 
268  if (op.getUniform() && *cacheLevel == CacheLevel::L1)
269  return {llvm::Intrinsic::nvvm_prefetchu_L1, args};
270 
271  if (evictPriority && *cacheLevel == CacheLevel::L2) {
272  switch (*evictPriority) {
273  case NVVM::CacheEvictionPriority::EvictLast:
274  return {llvm::Intrinsic::nvvm_prefetch_global_L2_evict_last, args};
275  case NVVM::CacheEvictionPriority::EvictNormal:
276  return {llvm::Intrinsic::nvvm_prefetch_global_L2_evict_normal, args};
277  default:
278  llvm_unreachable("Invalid cache eviction priority");
279  }
280  }
281 
282  switch (static_cast<MemSpace>(addressSpace)) {
283  case MemSpace::Generic:
284  return *cacheLevel == CacheLevel::L1
285  ? NVVM::IDArgPair({llvm::Intrinsic::nvvm_prefetch_L1, args})
286  : NVVM::IDArgPair({llvm::Intrinsic::nvvm_prefetch_L2, args});
287  case MemSpace::Global:
288  return *cacheLevel == CacheLevel::L1
289  ? NVVM::IDArgPair(
290  {llvm::Intrinsic::nvvm_prefetch_global_L1, args})
291  : NVVM::IDArgPair(
292  {llvm::Intrinsic::nvvm_prefetch_global_L2, args});
293  case MemSpace::Local:
294  return *cacheLevel == CacheLevel::L1
295  ? NVVM::IDArgPair(
296  {llvm::Intrinsic::nvvm_prefetch_local_L1, args})
297  : NVVM::IDArgPair(
298  {llvm::Intrinsic::nvvm_prefetch_local_L2, args});
299  default:
300  llvm_unreachable("Invalid pointer address space");
301  }
302 }

305 PermuteOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,
306  llvm::IRBuilderBase &builder) {
307  auto thisOp = cast<NVVM::PermuteOp>(op);
308  NVVM::PermuteMode mode = thisOp.getMode();
309 
310  static constexpr llvm::Intrinsic::ID IDs[] = {
311  llvm::Intrinsic::nvvm_prmt, llvm::Intrinsic::nvvm_prmt_f4e,
312  llvm::Intrinsic::nvvm_prmt_b4e, llvm::Intrinsic::nvvm_prmt_rc8,
313  llvm::Intrinsic::nvvm_prmt_ecl, llvm::Intrinsic::nvvm_prmt_ecr,
314  llvm::Intrinsic::nvvm_prmt_rc16};
315 
316  unsigned modeIndex = static_cast<unsigned>(mode);
317  llvm::SmallVector<llvm::Value *> args;
318  args.push_back(mt.lookupValue(thisOp.getLo()));
319 
320  // Only first 3 modes (Default, f4e, b4e) need the hi operand.
321  if (modeIndex < 3)
322  args.push_back(mt.lookupValue(thisOp.getHi()));
323 
324  args.push_back(mt.lookupValue(thisOp.getSelector()));
325 
326  return {IDs[modeIndex], args};
327 }

346 LogicalResult SparseCompressOp::verify() {
347  auto dataTy = cast<VectorType>(getData().getType());
348  auto compressedTy = cast<VectorType>(getCompressedData().getType());
349  if (dataTy.getElementType() != compressedTy.getElementType())
350  return emitOpError(
351  "expects data and compressed data to have the same element type");
352 
353  // data occupies 2 * repFactor 32-bit registers.
354  unsigned elemSize = dataTy.getElementTypeBitWidth();
355  unsigned dataBits = dataTy.getNumElements() * elemSize;
356  unsigned repFactor = dataBits / 64;
357  if (dataBits % 64 != 0 || !isValidSPRepFactor(repFactor))
358  return emitOpError("expects data to occupy 2 * repFactor 32-bit "
359  "registers, where repFactor is a power of two up to 64");
360 
361  // spcompress only supports the 2:4 sparsity factor.
362  if (2 * compressedTy.getNumElements() != dataTy.getNumElements())
363  return emitOpError(
364  "expects compressed data to have half as many elements as data");
365 
366  unsigned numMetadataRegs =
367  llvm::divideCeil(repFactor * getIdxSize(), elemSize);
368  if (getNumSPRegisters(getMetadata().getType()) != numMetadataRegs)
369  return emitOpError("expects metadata to occupy ")
370  << numMetadataRegs << " 32-bit register(s)";
371 
372  return success();
373 }

375 LogicalResult SparseDecompressOp::verify() {
376  auto dataTy = cast<VectorType>(getData().getType());
377  auto compressedTy = cast<VectorType>(getCompressedData().getType());
378  if (dataTy.getElementType() != compressedTy.getElementType())
379  return emitOpError(
380  "expects data and compressed data to have the same element type");
381 
382  // data holds repFactor groups of numTgt elements, and compressed data holds
383  // numSrc elements for each of those groups.
384  unsigned idxSize = getIdxSize();
385  unsigned numTgt = getNumTgt();
386  unsigned numData = dataTy.getNumElements();
387  unsigned numCompressed = compressedTy.getNumElements();
388  unsigned repFactor = numData / numTgt;
389  if (numData % numTgt != 0 || !isValidSPRepFactor(repFactor))
390  return emitOpError("expects data to have repFactor * num_tgt elements, "
391  "where repFactor is a power of two up to 64");
392 
393  unsigned numSrc = numCompressed / repFactor;
394  if (numCompressed % repFactor != 0 ||
395  !llvm::is_contained({1u, 2u, 4u}, numSrc) || numSrc >= numTgt)
396  return emitOpError("unsupported sparsity factor: compressed data has ")
397  << numCompressed << " elements for " << repFactor << " group(s) of "
398  << numTgt << " elements";
399 
400  unsigned elemSize = dataTy.getElementTypeBitWidth();
401  unsigned dataBits = numData * elemSize;
402  if (dataBits < 32 || dataBits > 4096)
403  return emitOpError("expects data to be between 32 and 4096 bits");
404 
405  if (numSrc * elemSize > 32)
406  return emitOpError("expects at most 32 bits of compressed data per group");
407 
408  if (idxSize == 2 && numTgt > 4)
409  return emitOpError("expects num_tgt to be at most 4 if idx_size is 2");
410 
411  unsigned numMetadataRegs = llvm::divideCeil(numSrc * idxSize * repFactor, 32);
412  if (getNumSPRegisters(getMetadata().getType()) != numMetadataRegs)
413  return emitOpError("expects metadata to occupy ")
414  << numMetadataRegs << " 32-bit register(s)";
415 
416  return success();
417 }
