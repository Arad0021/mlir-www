108 FailureOr<APInt> convertIntegerConstant(IntegerType targetType,
109  const APInt &value,
110  bool allowLossyConversion) {
111  const unsigned targetWidth = targetType.getWidth();
112  if (!allowLossyConversion && !value.isSignedIntN(targetWidth))
113  return failure();
114 
115  if (allowLossyConversion)
116  return value.truncSSat(targetWidth);
117  return value.sextOrTrunc(targetWidth);
118 }

120 FailureOr<APFloat> convertFloatConstant(FloatType targetType,
121  const APFloat &value,
122  bool allowLossyConversion) {
123  APFloat converted(value);
124  bool losesInfo = false;
125  converted.convert(targetType.getFloatSemantics(),
126  APFloat::rmNearestTiesToEven, &losesInfo);
127  if (!allowLossyConversion && losesInfo)
128  return failure();
129  return converted;
130 }

132 // Narrows scalar constant attributes so they keep matching the converted
133 // element types.
134 template <TosaNarrowKind Kind>
135 FailureOr<Attribute> tryConvertScalarAttribute(Attribute attribute,
136  bool allowLossyConversion) {
137  if constexpr (Kind == TosaNarrowKind::Int64ToInt32) {
138  if (const auto intAttr = dyn_cast<IntegerAttr>(attribute)) {
139  if (const auto intType = dyn_cast<IntegerType>(intAttr.getType());
140  intType && isSourceInteger<Kind>(intType)) {
141  const auto convertedType =
142  cast<IntegerType>(convertInteger<Kind>(intType));
143  FailureOr<APInt> convertedValue = convertIntegerConstant(
144  convertedType, intAttr.getValue(), allowLossyConversion);
145  if (failed(convertedValue))
146  return failure();
147  return IntegerAttr::get(convertedType, convertedValue.value());
148  }
149  }
150  } else if constexpr (Kind == TosaNarrowKind::Float64ToFloat32) {
151  if (const auto floatAttr = dyn_cast<FloatAttr>(attribute)) {
152  if (const auto floatType = dyn_cast<FloatType>(floatAttr.getType());
153  floatType && isSourceFloat<Kind>(floatType)) {
154  const auto convertedType =
155  cast<FloatType>(convertFloat<Kind>(floatType));
156  FailureOr<APFloat> convertedValue = convertFloatConstant(
157  convertedType, floatAttr.getValue(), allowLossyConversion);
158  if (failed(convertedValue))
159  return failure();
160  return FloatAttr::get(convertedType, convertedValue.value());
161  }
162  }
163  }
164 
165  return attribute;
166 }

168 template <TosaNarrowKind Kind>
169 FailureOr<Attribute>
170 convertDenseIntElementsAttr(ShapedType type, DenseIntElementsAttr attr,
171  const TypeConverter &typeConverter,
172  bool allowLossyConversion) {
173  if constexpr (Kind != TosaNarrowKind::Int64ToInt32)
174  return attr;
175 
176  const auto oldElementType = dyn_cast<IntegerType>(type.getElementType());
177  if (!oldElementType || !isSourceInteger<Kind>(oldElementType))
178  return attr;
179 
180  const auto newType =
181  dyn_cast_or_null<ShapedType>(typeConverter.convertType(type));
182  if (!newType)
183  return failure();
184 
185  const auto newElementType = dyn_cast<IntegerType>(newType.getElementType());
186  if (!newElementType)
187  return failure();
188 
189  if (!allowLossyConversion) {
190  for (APInt value : attr.getValues<APInt>())
191  if (failed(convertIntegerConstant(newElementType, value,
192  /*allowLossyConversion=*/false)))
193  return failure();
194  }
195 
196  Attribute convertedAttr =
197  attr.mapValues(newElementType, [&](const APInt &value) -> APInt {
198  return convertIntegerConstant(newElementType, value,
199  /*allowLossyConversion=*/true)
200  .value();
201  });
202  return convertedAttr;
203 }

208  const TypeConverter &typeConverter,

245 template <TosaNarrowKind Kind>
246 FailureOr<Attribute> convertDenseResourceElementsAttr(
247  ShapedType type, DenseResourceElementsAttr attr,
248  const TypeConverter &typeConverter, bool allowLossyConversion) {
249  static_assert(Kind == TosaNarrowKind::Int64ToInt32 ||
250  Kind == TosaNarrowKind::Float64ToFloat32);
251  using From =
252  std::conditional_t<Kind == TosaNarrowKind::Int64ToInt32, int64_t, double>;
253  using To =
254  std::conditional_t<Kind == TosaNarrowKind::Int64ToInt32, int32_t, float>;
255 
256  if (Kind == TosaNarrowKind::Int64ToInt32 &&
257  !isa<DenseI64ResourceElementsAttr>(attr)) {
258  return attr;
259  }
260 
261  if (Kind == TosaNarrowKind::Float64ToFloat32 &&
262  !isa<DenseF64ResourceElementsAttr>(attr)) {
263  return attr;
264  }
265 
266  auto narrow = [](From value) {
267  if constexpr (Kind == TosaNarrowKind::Int64ToInt32) {
268  value = std::clamp<From>(value, std::numeric_limits<To>::min(),
269  std::numeric_limits<To>::max());
270  }
271 
272  return static_cast<To>(value);
273  };
274 
275  const auto newType =
276  dyn_cast_or_null<ShapedType>(typeConverter.convertType(type));
277  if (!newType) {
278  return failure();
279  }
280 
281  const std::optional<ArrayRef<From>> values =
282  tryGetDenseResourceValues<From>(attr);
283  if (!values) {
284  return failure();
285  }
286 
287  SmallVector<To> newValues;
288  newValues.reserve(values->size());
289  for (From value : *values) {
290  const To convertedValue = narrow(value);
291  if (!allowLossyConversion && convertedValue != value) {
292  return failure();
293  }
294 
295  newValues.push_back(convertedValue);
296  }
297 
298  AsmResourceBlob blob = HeapAsmResourceBlob::allocateAndCopyInferAlign(
299  ArrayRef<To>(newValues.data(), newValues.size()));
300 
301  auto resourceManager =
302  DenseResourceElementsHandle::getManagerInterface(attr.getContext());
303  resourceManager.getBlobManager().update(attr.getRawHandle().getKey(),
304  std::move(blob));
305 
306  return DenseResourceElementsAttr::get(newType, attr.getRawHandle());
307 }

324 // Rejects cast rewrites that would lose precision (unless aggressive mode is
325 // enabled).
326 template <TosaNarrowKind Kind>
327 LogicalResult
328 verifyCastDoesNotLosePrecision(Operation *op, ShapedType inputType,
329  ShapedType resultType,
330  ConversionPatternRewriter &rewriter) {
331  if constexpr (Kind == TosaNarrowKind::Int64ToInt32) {
332  const auto elementInputIntType =
333  dyn_cast<IntegerType>(inputType.getElementType());
334  const auto elementResultIntType =
335  dyn_cast<IntegerType>(resultType.getElementType());
336  if (elementInputIntType && elementResultIntType &&
337  elementInputIntType.getWidth() > elementResultIntType.getWidth())
338  return rewriter.notifyMatchFailure(
339  op, "Narrowing cast may lead to data loss.");
340  } else if constexpr (Kind == TosaNarrowKind::Float64ToFloat32) {
341  const auto elementInputFloatType =
342  dyn_cast<FloatType>(inputType.getElementType());
343  const auto elementResultFloatType =
344  dyn_cast<FloatType>(resultType.getElementType());
345  if (elementInputFloatType && elementResultFloatType &&
346  elementInputFloatType.getIntOrFloatBitWidth() >
347  elementResultFloatType.getIntOrFloatBitWidth())
348  return rewriter.notifyMatchFailure(
349  op, "Narrowing cast may lead to data loss.");
350  }
351 
352  return success();
353 }

359 // Applies the narrowing TypeConverter to a single TOSA op, including its
360 // attributes and nested regions.
361 template <TosaNarrowKind Kind>
362 LogicalResult convertGenericOp(Operation *op, ValueRange operands,
363  ConversionPatternRewriter &rewriter,
364  const TypeConverter *typeConverter,
365  bool allowLossyConversion) {
366  SmallVector<Type, 4> newResults;
367  if (failed(typeConverter->convertTypes(op->getResultTypes(), newResults)))
368  return failure();
369 
370  OperationState state(op->getLoc(), op->getName().getStringRef(), operands,
371  newResults, {}, op->getSuccessors());
372 
373  // Keep attribute payloads consistent with the converted element types.
374  NamedAttrList sourceAttrs(op->getDiscardableAttrDictionary().getValue());
375  op->getName().walkInherentAttrs(op, [&](StringRef name, Attribute &attr) {
376  sourceAttrs.append(name, attr);
377  });
378  for (const NamedAttribute &namedAttribute : sourceAttrs) {
379  const Attribute attribute = namedAttribute.getValue();
380 
381  if (isa<IntegerAttr>(attribute) || isa<FloatAttr>(attribute)) {
382  FailureOr<Attribute> convertedAttr =
383  tryConvertScalarAttribute<Kind>(attribute, allowLossyConversion);
384  if (failed(convertedAttr))
385  return rewriter.notifyMatchFailure(
386  op, "Scalar attribute narrowing would lose precision; enable "
387  "aggressive rewrite to override.");
388  state.addAttribute(namedAttribute.getName(), convertedAttr.value());
389  continue;
390  }
391 
392  if (const auto typeAttr = dyn_cast<TypeAttr>(attribute)) {
393  FailureOr<Attribute> convertedAttr =
394  convertAttributeWithTypeConverter<Kind>(typeAttr, typeAttr.getValue(),
395  typeConverter);
396  if (failed(convertedAttr))
397  return rewriter.notifyMatchFailure(op,
398  "Failed to convert type attribute.");
399  state.addAttribute(namedAttribute.getName(), convertedAttr.value());
400  continue;
401  }
402 
403  if (const auto denseElementsAttr = dyn_cast<DenseElementsAttr>(attribute)) {
404  FailureOr<Attribute> convertedAttr =
405  convertAttributeWithTypeConverter<Kind>(
406  denseElementsAttr, denseElementsAttr.getType(), typeConverter);
407  if (failed(convertedAttr))
408  return rewriter.notifyMatchFailure(
409  op, "Failed to convert dense elements attribute without precision "
410  "loss; enable aggressive rewrite to override.");
411  state.addAttribute(namedAttribute.getName(), convertedAttr.value());
412  continue;
413  }
414 
415  if (const auto denseResourceElementsAttr =
416  dyn_cast<DenseResourceElementsAttr>(attribute)) {
417  FailureOr<Attribute> convertedAttr =
418  convertAttributeWithTypeConverter<Kind>(
419  denseResourceElementsAttr, denseResourceElementsAttr.getType(),
420  typeConverter);
421  if (failed(convertedAttr))
422  return rewriter.notifyMatchFailure(
423  op, "Failed to convert dense resource elements attribute without "
424  "precision loss; enable aggressive rewrite to override.");
425  state.addAttribute(namedAttribute.getName(), convertedAttr.value());
426  continue;
427  }
428 
429  state.addAttribute(namedAttribute.getName(), attribute);
430  }
431 
432  for (Region &region : op->getRegions()) {
433  if (failed(rewriter.convertRegionTypes(&region, *typeConverter)))
434  return failure();
435  Region *newRegion = state.addRegion();
436  rewriter.inlineRegionBefore(region, *newRegion, newRegion->begin());
437  }
438 
439  Operation *newOp = rewriter.create(state);
440  rewriter.replaceOp(op, newOp->getResults());
441  return success();
442 }

584 // Shared implementation for both narrowing passes; the mode decides which
585 // element types and attribute payloads participate.
586 template <TosaNarrowKind Kind>
587 LogicalResult runTosaNarrowing(Operation *op, bool aggressiveRewrite,
588  bool convertFunctionBoundaries) {
589  MLIRContext *context = op->getContext();
590  const bool allowLossyConversion = aggressiveRewrite;
591 
592  TypeConverter typeConverter;
593  typeConverter.addConversion([](Type type) -> Type { return type; });
594 
595  typeConverter.addConversion(
596  [](IntegerType type) -> Type { return convertInteger<Kind>(type); });
597  typeConverter.addConversion(
598  [](FloatType type) -> Type { return convertFloat<Kind>(type); });
599  typeConverter.addConversion([&typeConverter](RankedTensorType type) -> Type {
600  Type elementType = type.getElementType();
601  if (!isSourceElement<Kind>(elementType))
602  return type;
603  Type converted = typeConverter.convertType(elementType);
604  if (!converted || converted == elementType)
605  return type;
606  return RankedTensorType::get(type.getShape(), converted,
607  type.getEncoding());
608  });
609  typeConverter.addConversion(
610  [&typeConverter](UnrankedTensorType type) -> Type {
611  Type elementType = type.getElementType();
612  if (!isSourceElement<Kind>(elementType))
613  return type;
614  Type converted = typeConverter.convertType(elementType);
615  if (!converted || converted == elementType)
616  return type;
617  return UnrankedTensorType::get(converted);
618  });
619 
620  const auto materializeCast = [](OpBuilder &builder, Type resultType,
621  ValueRange inputs, Location loc) -> Value {
622  if (inputs.size() != 1)
623  return Value();
624  return tosa::CastOp::create(
625  builder, loc, resultType, inputs.front(),
626  getStorageElementTypeOrSelf(inputs.front().getType())
627  .isUnsignedInteger());
628  };
629  typeConverter.addSourceMaterialization(materializeCast);
630  typeConverter.addTargetMaterialization(materializeCast);
631 
632  typeConverter.addTypeAttributeConversion(
633  [&typeConverter, allowLossyConversion](ShapedType type,
634  DenseResourceElementsAttr attr)
635  -> TypeConverter::AttributeConversionResult {
636  FailureOr<Attribute> converted = convertDenseResourceElementsAttr<Kind>(
637  type, attr, typeConverter, allowLossyConversion);
638  if (failed(converted))
639  return TypeConverter::AttributeConversionResult::abort();
640  return TypeConverter::AttributeConversionResult::result(
641  converted.value());
642  });
643 
644  if constexpr (Kind == TosaNarrowKind::Int64ToInt32) {
645  typeConverter.addTypeAttributeConversion(
646  [allowLossyConversion](IntegerType /*type*/, IntegerAttr attribute)
647  -> TypeConverter::AttributeConversionResult {
648  FailureOr<Attribute> converted =
649  tryConvertScalarAttribute<Kind>(attribute, allowLossyConversion);
650  if (failed(converted))
651  return TypeConverter::AttributeConversionResult::abort();
652  return TypeConverter::AttributeConversionResult::result(
653  converted.value());
654  });
655  typeConverter.addTypeAttributeConversion(
656  [&typeConverter, allowLossyConversion](ShapedType type,
657  DenseIntElementsAttr attr)

690  ConversionTarget target(*context);
691  target.addDynamicallyLegalDialect<tosa::TosaDialect>(
692  [&typeConverter](Operation *op) {
693  return typeConverter.isLegal(op->getResultTypes()) &&
694  typeConverter.isLegal(op->getOperandTypes());
695  });
696  if (convertFunctionBoundaries) {
697  target.addDynamicallyLegalOp<func::FuncOp>(
698  [&typeConverter](func::FuncOp op) {
699  return typeConverter.isSignatureLegal(op.getFunctionType()) &&
700  typeConverter.isLegal(&op.getBody());
701  });
702  target.addDynamicallyLegalOp<func::ReturnOp>([](func::ReturnOp op) {
703  const FunctionType funcType =

705  return llvm::equal(op.getOperandTypes(), funcType.getResults());
706  });
707  } else {

709  [](func::FuncOp) { return true; });
710  target.addDynamicallyLegalOp<func::ReturnOp>(
711  [](func::ReturnOp) { return true; });
712  }

714  RewritePatternSet patterns(context);
715  if (convertFunctionBoundaries) {
716  populateFunctionOpInterfaceTypeConversionPattern<func::FuncOp>(
717  patterns, typeConverter);
718  populateReturnOpTypeConversionPattern(patterns, typeConverter);
719  }
720  if (aggressiveRewrite) {
721  patterns.add<ConvertGenericOp<Kind>>(typeConverter, context,
722  allowLossyConversion);
723  } else {
724  if constexpr (Kind == TosaNarrowKind::Int64ToInt32) {
725  patterns.add<ConvertArgMaxOpWithBoundsChecking>(typeConverter, context);
726  patterns.add<ConvertClampOpWithBoundsChecking<Kind>>(typeConverter,
727  context);
728  }
729  patterns.add<ConvertTypedOp<tosa::ConstOp, Kind>>(typeConverter, context);
730  patterns.add<ConvertTypedOp<tosa::ConcatOp, Kind>>(typeConverter, context);
731  patterns.add<ConvertTypedOp<tosa::PadOp, Kind>>(typeConverter, context);
732  patterns.add<ConvertTypedOp<tosa::ReshapeOp, Kind>>(typeConverter, context);
733  patterns.add<ConvertTypedOp<tosa::ReverseOp, Kind>>(typeConverter, context);
734  patterns.add<ConvertTypedOp<tosa::SliceOp, Kind>>(typeConverter, context);
735  patterns.add<ConvertTypedOp<tosa::TileOp, Kind>>(typeConverter, context);
736  patterns.add<ConvertTypedOp<tosa::TransposeOp, Kind>>(typeConverter,
737  context);
738  patterns.add<ConvertTypedOp<tosa::IdentityOp, Kind>>(typeConverter,
739  context);
740  patterns.add<ConvertCastOpWithBoundsChecking<Kind>>(typeConverter, context);
741  patterns.add<ConvertTypedOp<tosa::IfOp, Kind>>(typeConverter, context);
742  patterns.add<ConvertTypedOp<tosa::WhileOp, Kind>>(typeConverter, context);
743  patterns.add<ConvertTypedOp<tosa::YieldOp, Kind>>(typeConverter, context);
744  }
745 
746  if (failed(applyFullConversion(op, target, std::move(patterns))))
747  return failure();
748  return success();
