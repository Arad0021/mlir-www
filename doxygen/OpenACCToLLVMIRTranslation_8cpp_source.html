45 /// Returns the offloading information of a GPU target attribute, or
46 /// `std::nullopt` if the target is not supported.
47 static std::optional<OffloadTargetInfo> getOffloadTargetInfo(Attribute target) {
48  return llvm::TypeSwitch<Attribute, std::optional<OffloadTargetInfo>>(target)
49  .Case([](NVVM::NVVMTargetAttr nvvm) {
50  return OffloadTargetInfo{nvvm.getTriple(), nvvm.getChip()};
51  })
52  .Case([](ROCDL::ROCDLTargetAttr rocdl) {
53  return OffloadTargetInfo{rocdl.getTriple(), rocdl.getChip()};
54  })
55  .Case([](xevm::XeVMTargetAttr xevm) {
56  return OffloadTargetInfo{xevm.getTriple(), xevm.getChip()};
57  })
58  .Default([](Attribute) { return std::nullopt; });
59 }

81 LogicalResult OffloadPackagerAttrImpl::embedBinary(
82  Attribute attribute, Operation *operation, llvm::IRBuilderBase &builder,
83  LLVM::ModuleTranslation &moduleTranslation) const {
84  assert(operation && "The binary operation must be non null.");
85  auto op = mlir::dyn_cast<gpu::BinaryOp>(operation);
86  if (!op) {
87  operation->emitError("operation must be a GPU binary");
88  return failure();
89  }
90 
91  using namespace llvm::object;
92 
93  SmallVector<char> binaryData;
94  llvm::raw_svector_ostream os(binaryData);
95  for (Attribute attr : op.getObjectsAttr().getValue()) {
96  auto object = dyn_cast<gpu::ObjectAttr>(attr);
97  if (!object)
98  return op.emitError("expected a gpu.object");
99 
100  // The section is chosen by the offloading binary, so a requested one would
101  // be ignored.
102  if (DictionaryAttr properties = object.getProperties())
103  if (properties.get(gpu::elfSectionName))
104  return op.emitError("the offload packager does not support custom "
105  "sections");
106 
107  std::optional<OffloadTargetInfo> target =
108  getOffloadTargetInfo(object.getTarget());
109  if (!target)
110  return op.emitError("the offload packager does not support the target ")
111  << object.getTarget();
112 
113  OffloadBinary::OffloadingImage image;
114  bool requiresNullTerminator = false;
115  switch (object.getFormat()) {
116  case gpu::CompilationTarget::Offload:
117  image.TheImageKind = IMG_Bitcode;
118  requiresNullTerminator = true;
119  break;
120  case gpu::CompilationTarget::Binary:
121  image.TheImageKind = IMG_Object;
122  requiresNullTerminator = false;
123  break;
124  case gpu::CompilationTarget::Assembly:
125  case gpu::CompilationTarget::Fatbin:
126  return op.emitError("the offloading linker wrapper cannot link the "
127  "assembly or fatbin formats");
128  }
129 
130  StringRef serialized = object.getObject().getValue();
131  image.Image = llvm::MemoryBuffer::getMemBuffer(
132  serialized, /*BufferName=*/"", requiresNullTerminator);
133  image.TheOffloadKind = OFK_OpenACC;
134  image.StringData["arch"] = target->arch;
135  image.StringData["triple"] = target->triple;
136 
137  llvm::SmallString<0> buffer = OffloadBinary::write(image);
138  if (buffer.size() % OffloadBinary::getAlignment() != 0)
139  return op.emitError("offload binary has invalid size alignment");
140  os << buffer;
141  }
142 
143  llvm::Module &m = *moduleTranslation.getLLVMModule();
144  llvm::Constant *serializedCst = llvm::ConstantDataArray::getString(
145  m.getContext(), os.str(), /*AddNull=*/false);
146  auto *serializedObj =
147  new llvm::GlobalVariable(m, serializedCst->getType(), /*isConstant=*/true,
148  llvm::GlobalValue::InternalLinkage,
149  serializedCst, op.getName() + "_binary");
150  serializedObj->setUnnamedAddr(llvm::GlobalValue::UnnamedAddr::Global);
151  serializedObj->setSection(".llvm.offloading");
152  serializedObj->setAlignment(llvm::Align(OffloadBinary::getAlignment()));
153  llvm::appendToUsed(m, {serializedObj});
154  return success();
155 }

164 /// Registers the host function or variable `op` as an offloading entry of kind
165 /// OpenACC.
166 static LogicalResult
167 convertOffloadEntry(Operation *op, LLVM::ModuleTranslation &moduleTranslation) {
168  llvm::Module *m = moduleTranslation.getLLVMModule();
169  llvm::GlobalValue *global = nullptr;
170  uint64_t size = 0;
171  if (auto func = dyn_cast<LLVM::LLVMFuncOp>(op)) {
172  global = moduleTranslation.lookupFunction(func.getName());
173  assert(global && "could not find the translated host symbol");
174  } else if (auto variable = dyn_cast<LLVM::GlobalOp>(op)) {
175  global = moduleTranslation.lookupGlobal(variable);
176  assert(global && "could not find the translated host symbol");
177  auto *llvmVariable = cast<llvm::GlobalVariable>(global);
178  size = m->getDataLayout().getTypeStoreSize(llvmVariable->getValueType());
179  } else {
180  return op->emitError("unsupported operation for '")
181  << acc::getOffloadEntryAttrName() << "'";
182  }
183 
184  StringRef symbolName = global->getName();
185  llvm::offloading::emitOffloadingEntry(*m, llvm::object::OFK_OpenACC, global,
186  symbolName, size,
187  /*Flags=*/0, /*Data=*/0);
188  return success();
189 }
