19 const llvm::abi::Type *ABITypeMapper::map(mlir::Type type) {
20  if (auto intTy = dyn_cast<mlir::IntegerType>(type))
21  return mapIntegerType(intTy);
22 
23  if (auto floatTy = dyn_cast<mlir::FloatType>(type))
24  return mapFloatType(floatTy);
25 
26  if (auto indexTy = dyn_cast<mlir::IndexType>(type))
27  return mapIndexType(indexTy);
28 
29  if (auto vecTy = dyn_cast<mlir::VectorType>(type))
30  return mapVectorType(vecTy);
31 
32  if (auto memRefTy = dyn_cast<mlir::MemRefType>(type))
33  return mapMemRefType(memRefTy);
34 
35  if (auto noneTy = dyn_cast<mlir::NoneType>(type))
36  return mapNoneType(noneTy);
37 
38  // For dialect-specific types, fall back to DataLayout queries.
39  // The type must implement DataLayoutTypeInterface for this to work.
40  llvm::TypeSize sizeInBits = dl.getTypeSizeInBits(type);
41  uint64_t abiAlign = dl.getTypeABIAlignment(type);
42  return builder.getIntegerType(sizeInBits.getFixedValue(),
43  llvm::Align(abiAlign),
44  /*Signed=*/false);
45 }

68 const llvm::abi::Type *ABITypeMapper::mapVectorType(mlir::VectorType type) {
69  const llvm::abi::Type *elementTy = map(type.getElementType());
70  if (!elementTy)
71  return nullptr;
72 
73  // ElementCount supports at most one scalable dimension, hence
74  // vectors with more than one scalable dimension (e.g. vector<[2]x[4]xf32>)
75  // cannot be represented and mapVectorType must return nullptr in this case.
76  if (llvm::count(type.getScalableDims(), true) > 1)
77  return nullptr;
78 
79  auto shape = type.getShape();
80  uint64_t totalElements = 1;
81  for (int64_t dim : shape)
82  totalElements *= dim;
83 
84  llvm::ElementCount ec =
85  llvm::ElementCount::get(totalElements, type.isScalable());
86  uint64_t abiAlign = dl.getTypeABIAlignment(type);
87  return builder.getVectorType(elementTy, ec, llvm::Align(abiAlign));
88 }

90 const llvm::abi::Type *ABITypeMapper::mapMemRefType(mlir::MemRefType type) {
91  llvm::TypeSize sizeInBits = dl.getTypeSizeInBits(type);
92  uint64_t abiAlign = dl.getTypeABIAlignment(type);
93  unsigned addrSpace = 0;
94  if (auto as = type.getMemorySpace())
95  if (auto intAttr = dyn_cast<IntegerAttr>(as))
96  addrSpace = intAttr.getInt();
97  return builder.getPointerType(sizeInBits.getFixedValue(),
98  llvm::Align(abiAlign), addrSpace);
99 }
