61 // Convert a validated byte-aligned payload, retaining storage for every
62 // element. In particular, each i4 element occupies one byte, not half a byte.
63 SmallVector<char> convertRawValues(ArrayRef<char> rawData, IntegerType type,
64  IntegerType convertedType) {
65  const unsigned inputElementSizeInBytes = llvm::divideCeil(type.getWidth(), 8);
66  const unsigned outputElementSizeInBytes = convertedType.getWidth() / 8;
67  const size_t numElements = rawData.size() / inputElementSizeInBytes;
68  SmallVector<char> data(numElements * outputElementSizeInBytes);
69  const auto *input = reinterpret_cast<const uint8_t *>(rawData.data());
70  auto *output = reinterpret_cast<uint8_t *>(data.data());
71  for (size_t i = 0; i < numElements; ++i) {
72  APInt value(inputElementSizeInBytes * 8, 0);
73  llvm::LoadIntFromMemory(value, input + i * inputElementSizeInBytes,
74  inputElementSizeInBytes);
75  // Discard padding bits before interpreting the source sign bit.
76  value = value.trunc(type.getWidth());
77  llvm::StoreIntToMemory(value.sext(convertedType.getWidth()),
78  output + i * outputElementSizeInBytes,
79  outputElementSizeInBytes);
80  }
81  return data;
82 }

84 class IntegerTypeReplacer : public AttrTypeReplacer {
85 public:
86  // The mappings must outlive this replacer.
87  explicit IntegerTypeReplacer(const DenseMap<unsigned, unsigned> &mappings) {
88  addReplacement([&mappings](IntegerType type) -> std::optional<Type> {
89  auto mapping = mappings.find(type.getWidth());
90  if (mapping == mappings.end())
91  return std::nullopt;
92  return IntegerType::get(type.getContext(), mapping->second);
93  });
94  addReplacement(
95  [this](IntegerAttr attr) { return skip(convertInteger(attr)); });
96  addReplacement(
97  [this](DenseArrayAttr attr) { return skip(convertDenseArray(attr)); });
98  addReplacement([this](DenseTypedElementsAttr attr) {
99  return skip(convertDenseElements(attr));
100  });
101  addReplacement([this](DenseResourceElementsAttr attr) {
102  return skip(convertDenseResource(attr));
103  });
104  }
105 
106  bool hasFailed() const { return conversionFailed; }
107 
108 private:
109  Attribute convertInteger(IntegerAttr attr) {
110  Type type = attr.getType();
111  Type convertedType = replace(type);
112  if (type == convertedType)
113  return attr;
114  return IntegerAttr::get(
115  convertedType,
116  attr.getValue().sext(cast<IntegerType>(convertedType).getWidth()));
117  }
118 
119  Attribute convertDenseArray(DenseArrayAttr attr) {
120  Type type = attr.getElementType();
121  Type convertedType = replace(type);
122  if (type == convertedType)
123  return attr;
124  // Unlike dense elements, arrays always store every element, even splats.
125  const SmallVector<char> data =
126  convertRawValues(attr.getRawData(), cast<IntegerType>(type),
127  cast<IntegerType>(convertedType));
128  return DenseArrayAttr::get(convertedType, attr.size(), data);
129  }
130 
131  Attribute convertDenseElements(DenseTypedElementsAttr attr) {
132  auto convertedType = dyn_cast_or_null<ShapedType>(replace(attr.getType()));
133  if (!convertedType)
134  return nullptr;
135 
136  Type elementType = attr.getElementType();
137  Type convertedElementType = convertedType.getElementType();
138  if (elementType == convertedElementType) {
139  // reshape is still necessary to handle any encoding changes.
140  return attr.reshape(convertedType);
141  }
142 
143  auto convertedIntegerType = dyn_cast<IntegerType>(convertedElementType);
144  if (!convertedIntegerType)
145  return nullptr;
146  return attr
147  .mapValues(convertedElementType,
148  [convertedIntegerType](const APInt &value) {
149  return value.sext(convertedIntegerType.getWidth());
150  })
151  .reshape(convertedType);
152  }
153 
154  Attribute convertDenseResource(DenseResourceElementsAttr attr) {
155  ShapedType type = attr.getType();
156  Type elementType = type.getElementType();
157  auto convertedType = dyn_cast_or_null<ShapedType>(replace(type));
158  if (!convertedType)
159  return nullptr;
160  Type convertedElementType = convertedType.getElementType();
161  if (elementType == convertedElementType)
162  return DenseResourceElementsAttr::get(convertedType, attr.getRawHandle());
163  auto convertedIntegerType = dyn_cast<IntegerType>(convertedElementType);
164  if (!convertedIntegerType)
165  return nullptr;
166  auto integerType = cast<IntegerType>(elementType);
167  const AsmResourceBlob *blob = attr.getRawHandle().getBlob();
168  if (!blob)
169  return nullptr;
170  ArrayRef<char> rawData = blob->getData();
171  const unsigned elementSizeInBytes =
172  llvm::divideCeil(integerType.getWidth(), 8);
173  if (rawData.size() % elementSizeInBytes != 0)
174  return nullptr;
175  // Check resource is correct size.
176  if (type.hasStaticShape() &&
177  rawData.size() !=
178  elementSizeInBytes * static_cast<uint64_t>(type.getNumElements()))
179  return nullptr;
180  const SmallVector<char> data =
181  convertRawValues(rawData, integerType, convertedIntegerType);
182  const unsigned alignment = convertedIntegerType.getWidth() / 8;
183  return DenseResourceElementsAttr::get(
184  convertedType, attr.getRawHandle().getKey(),
185  HeapAsmResourceBlob::allocateAndCopyWithAlign(data, alignment));
186  }
187 
188  // Payload handlers already converted their subelements.
189  ReplaceFnResult<Attribute> skip(Attribute attr) {
190  // replaceElementsIn ignores null replacements, so track failures here.
191  conversionFailed |= !attr;
192  return std::make_pair(attr, WalkResult::skip());
193  }
194 
195  bool conversionFailed = false;
196 };

198 struct TosaConvertIntegerTypesBitwidth
199  : tosa::impl::TosaConvertIntegerTypesBitwidthPassBase<
200  TosaConvertIntegerTypesBitwidth> {
201  using Base::Base;
202 
203  void runOnOperation() override {
204  if (bitwidthMappings.empty()) {
205  getOperation().emitError(
206  "The 'bitwidth-mappings' option must be specified and non-empty.");
207  return signalPassFailure();
208  }
209  DenseMap<unsigned, unsigned> mappings;
210  for (auto [source, destination] : bitwidthMappings) {
211  if (!((source == 4 && destination == 8) ||
212  (source == 48 && destination == 64))) {
213  getOperation().emitError("unsupported integer bitwidth mapping ")
214  << source << ":" << destination << ": expected 4:8 or 48:64";
215  return signalPassFailure();
216  }
217  mappings.try_emplace(source, destination);
218  }
219  IntegerTypeReplacer replacer(mappings);
220  replacer.recursivelyReplaceElementsIn(getOperation(), /*replaceAttrs=*/true,
221  /*replaceLocs=*/false,
222  /*replaceTypes=*/true);
223  if (replacer.hasFailed()) {
224  getOperation().emitError("failed to convert integer types or attributes");
225  signalPassFailure();
226  }
227  }
228 };
