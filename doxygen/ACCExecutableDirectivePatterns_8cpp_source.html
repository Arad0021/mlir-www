45 struct WaitOpLowering : public ACCExecutableDirectivePattern<WaitOp> {
46  using ACCExecutableDirectivePattern<WaitOp>::ACCExecutableDirectivePattern;
47 
48  LogicalResult
49  matchAndRewrite(WaitOp op, WaitOp::Adaptor,
50  ConversionPatternRewriter &rewriter) const override {
51  Location loc = op->getLoc();
52 
53  auto emitWait = [&]() -> LogicalResult {
54  Value asyncOperand = op.getAsyncOperand()
55  ? rewriter.getRemappedValue(op.getAsyncOperand())
56  : Value();
57  Value asyncQueue =
58  getAsyncQueue(loc, asyncOperand, op.getAsync(), rewriter, config);
59  SmallVector<Value> waitValues;
60  for (Value operand : op.getWaitOperands())
61  waitValues.push_back(rewriter.getRemappedValue(operand));
62  return emitWaitCall(loc, waitValues, asyncQueue, rewriter,
63  globalSymbolRegion, symbolTable, config);
64  };
65 
66  if (failed(emitGuardedByIfCond(loc, op.getIfCond(), rewriter, emitWait)))
67  return failure();
68 
69  rewriter.eraseOp(op);
70  return success();
71  }
72 };

74 /// Emit a call to a runtime entry point taking
75 /// `(ident, flags, deviceType, deviceNum)`. A null `deviceNum` selects the
76 /// current device.
77 static LogicalResult
78 emitDeviceOperationCall(Location loc, RuntimeFunction fn, DeviceType deviceType,
79  Value deviceNum, StringRef functionName,
80  Region &globalSymbolRegion, SymbolTable &symbolTable,
81  ConversionPatternRewriter &rewriter,
82  const ACCRuntimeCallConfig &config) {
83  Type i64Ty = rewriter.getI64Type();
84  Value deviceTypeValue = LLVM::ConstantOp::create(
85  rewriter, loc, i64Ty, config.getDeviceTypeRuntimeValue(deviceType));
86  Value ident = createIdent(loc, functionName, rewriter, globalSymbolRegion,
87  config, &symbolTable);
88  Value flags = LLVM::ConstantOp::create(rewriter, loc, i64Ty, 0);
89  Value deviceNumValue =
90  deviceNum ? castToI64(loc, deviceNum, rewriter)
91  : LLVM::ConstantOp::create(rewriter, loc, i64Ty, -1);
92  return createRuntimeCall(loc, rewriter, globalSymbolRegion, symbolTable, fn,
93  config,
94  {ident, flags, deviceTypeValue, deviceNumValue});
95 }

97 static LogicalResult rewriteInitOrShutdown(Operation *op, Value deviceNum,
98  ArrayAttr deviceTypesAttr,
99  Value ifCond, bool isInit,
100  ConversionPatternRewriter &rewriter,
101  Region &globalSymbolRegion,
102  SymbolTable &symbolTable,
103  const ACCRuntimeCallConfig &config) {
104  Location loc = op->getLoc();
105 
106  auto emitCalls = [&]() -> LogicalResult {
107  StringRef functionName = deviceNum ? getParentFunctionName(deviceNum)
108  : getParentFunctionName(op);
109  RuntimeFunction fn = isInit ? RuntimeFunction::ACCRTL_tgt_acc_init
110  : RuntimeFunction::ACCRTL_tgt_acc_shutdown;
111 
112  auto emitOne = [&](DeviceType deviceType) {
113  return emitDeviceOperationCall(loc, fn, deviceType, deviceNum,
114  functionName, globalSymbolRegion,
115  symbolTable, rewriter, config);
116  };
117 
118  if (!deviceTypesAttr)
119  return emitOne(DeviceType::None);
120 
121  for (Attribute attr : deviceTypesAttr) {
122  if (auto typeAttr = dyn_cast<DeviceTypeAttr>(attr))
123  if (failed(emitOne(typeAttr.getValue())))
124  return failure();
125  }
126  return success();
127  };
128 
129  if (failed(emitGuardedByIfCond(loc, ifCond, rewriter, emitCalls)))
130  return failure();
131 
132  rewriter.eraseOp(op);
133  return success();
134 }

161 struct SetOpLowering : public ACCExecutableDirectivePattern<SetOp> {
162  using ACCExecutableDirectivePattern<SetOp>::ACCExecutableDirectivePattern;
163 
164  LogicalResult
165  matchAndRewrite(SetOp op, SetOp::Adaptor adaptor,
166  ConversionPatternRewriter &rewriter) const override {
167  Location loc = op.getLoc();
168  Type i64Ty = rewriter.getI64Type();
169 
170  auto emitSet = [&]() -> LogicalResult {
171  if (Value asyncValue = adaptor.getDefaultAsync()) {
172  asyncValue = castToI64(loc, asyncValue, rewriter);
173  Value ident =
174  createIdent(loc, getParentFunctionName(asyncValue), rewriter,
175  globalSymbolRegion, config, &symbolTable);
176  if (failed(createRuntimeCall(
177  loc, rewriter, globalSymbolRegion, symbolTable,
178  RuntimeFunction::ACCRTL_tgt_acc_set_default_async, config,
179  {ident, asyncValue})))
180  return failure();
181  }
182 
183  if (op.getDeviceNum()) {
184  Value deviceNum = adaptor.getDeviceNum();
185  DeviceType deviceType = DeviceType::None;
186  if (auto deviceTypeAttr = op.getDeviceTypeAttr())
187  deviceType = deviceTypeAttr.getValue();
188  return emitDeviceOperationCall(
189  loc, RuntimeFunction::ACCRTL_tgt_acc_set_device_num, deviceType,
190  deviceNum, getParentFunctionName(deviceNum), globalSymbolRegion,
191  symbolTable, rewriter, config);
192  }
193  if (auto deviceTypeAttr = op.getDeviceTypeAttr()) {
194  Value deviceTypeValue = LLVM::ConstantOp::create(
195  rewriter, loc, i64Ty,
196  config.getDeviceTypeRuntimeValue(deviceTypeAttr.getValue()));
197  Value ident = createIdent(loc, StringRef(), rewriter,
198  globalSymbolRegion, config, &symbolTable);
199  Value flags = LLVM::ConstantOp::create(rewriter, loc, i64Ty, 0);
200  return createRuntimeCall(
201  loc, rewriter, globalSymbolRegion, symbolTable,
202  RuntimeFunction::ACCRTL_tgt_acc_set_device_type, config,
203  {ident, flags, deviceTypeValue});
204  }
205  return success();
206  };
207 
208  if (failed(emitGuardedByIfCond(loc, op.getIfCond(), rewriter, emitSet)))
209  return failure();
210 
211  rewriter.eraseOp(op);
212  return success();
213  }
214 };
