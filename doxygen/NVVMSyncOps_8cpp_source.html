77 LogicalResult NVVM::MatchSyncOp::verify() {
78  if (getKind() == NVVM::MatchSyncKind::all) {
79  auto type = llvm::dyn_cast<LLVM::LLVMStructType>(getType());
80  if (!type || type.getBody().size() != 2 ||
81  !type.getBody()[0].isInteger(32) || !type.getBody()[1].isInteger(1)) {
82  return emitOpError("match.sync 'all' returns a two element struct with "
83  "first element as i32 and second element as i1");
84  }
85  } else {
86  if (!getType().isInteger(32)) {
87  return emitOpError("match.sync 'any' returns an i32");
88  }
89  }
90  return success();
91 }

126 LogicalResult NVVM::ReduxOp::verify() {
127  mlir::Type reduxType = getType();
128 
129  if (!reduxType.isF32()) {
130  if (getAbs())
131  return emitOpError("abs attribute is supported only for f32 type");
132  if (getNan())
133  return emitOpError("nan attribute is supported only for f32 type");
134  }
135 
136  NVVM::ReductionKind kind = getKind();
137  switch (kind) {
138  case NVVM::ReductionKind::ADD:
139  case NVVM::ReductionKind::AND:
140  case NVVM::ReductionKind::OR:
141  case NVVM::ReductionKind::XOR:
142  case NVVM::ReductionKind::MAX:
143  case NVVM::ReductionKind::MIN:
144  case NVVM::ReductionKind::UMAX:
145  case NVVM::ReductionKind::UMIN:
146  if (!reduxType.isInteger(32))
147  return emitOpError("'")
148  << kind << "' reduction kind unsupported with " << reduxType
149  << " type. Only supported type is 'i32'.";
150  break;
151  case NVVM::ReductionKind::FMIN:
152  case NVVM::ReductionKind::FMAX:
153  if (!reduxType.isF32())
154  return emitOpError("'")
155  << kind << "' reduction kind unsupported with " << reduxType
156  << " type. Only supported type is 'f32'.";
157  break;
158  }
159 
160  return success();
161 }

197 /// Maps the (aligned, hasCount) pair to the `@llvm.nvvm.barrier.cta.sync.*`
198 /// intrinsic ID.
199 static llvm::Intrinsic::ID getBarrierSyncIntrinsic(bool aligned,
200  bool hasCount) {
201  if (hasCount) {
202  return aligned ? llvm::Intrinsic::nvvm_barrier_cta_sync_aligned_count
203  : llvm::Intrinsic::nvvm_barrier_cta_sync_count;
204  }
205  return aligned ? llvm::Intrinsic::nvvm_barrier_cta_sync_aligned_all
206  : llvm::Intrinsic::nvvm_barrier_cta_sync_all;
207 }

209 /// Maps the (aligned, kind) pair to the `@llvm.nvvm.barrier.cta.red.*`
210 /// intrinsic ID.
211 static llvm::Intrinsic::ID
212 getBarrierReductionIntrinsic(bool aligned, NVVM::BarrierReduction kind) {
213  switch (kind) {
214  case NVVM::BarrierReduction::AND:
215  return aligned ? llvm::Intrinsic::nvvm_barrier_cta_red_and_aligned_all
216  : llvm::Intrinsic::nvvm_barrier_cta_red_and_all;
217  case NVVM::BarrierReduction::OR:
218  return aligned ? llvm::Intrinsic::nvvm_barrier_cta_red_or_aligned_all
219  : llvm::Intrinsic::nvvm_barrier_cta_red_or_all;
220  case NVVM::BarrierReduction::POPC:
221  return aligned ? llvm::Intrinsic::nvvm_barrier_cta_red_popc_aligned_all
222  : llvm::Intrinsic::nvvm_barrier_cta_red_popc_all;
223  }
224  llvm_unreachable("unknown BarrierReduction kind");
225 }

227 mlir::NVVM::IDArgPair NVVM::BarrierOp::getIntrinsicIDAndArgs(
228  Operation &op, LLVM::ModuleTranslation &mt, llvm::IRBuilderBase &builder) {
229  auto thisOp = cast<NVVM::BarrierOp>(op);
230  llvm::Value *barrierId = thisOp.getBarrierId()
231  ? mt.lookupValue(thisOp.getBarrierId())
232  : builder.getInt32(0);
233  bool hasCount = static_cast<bool>(thisOp.getNumberOfThreads());
234  llvm::Intrinsic::ID id =
235  getBarrierSyncIntrinsic(thisOp.getAligned(), hasCount);
236  llvm::SmallVector<llvm::Value *> args = {barrierId};
237  if (hasCount)
238  args.push_back(mt.lookupValue(thisOp.getNumberOfThreads()));
239  return {id, std::move(args)};
240 }
