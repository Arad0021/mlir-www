23 static ArrayAttr getLLVMAlignParamForCompressExpand(OpBuilder &builder,
24  bool isExpandLoad,
25  uint64_t alignment = 1) {
26  // From
27  // https://llvm.org/docs/LangRef.html#llvm-masked-expandload-intrinsics
28  // https://llvm.org/docs/LangRef.html#llvm-masked-compressstore-intrinsics
29  //
30  // The pointer alignment defaults to 1.
31  if (alignment == 1) {
32  return nullptr;
33  }
34 
35  auto emptyDictAttr = builder.getDictionaryAttr({});
36  auto alignmentAttr = builder.getI64IntegerAttr(alignment);
37  auto namedAttr =
38  builder.getNamedAttr(LLVMDialect::getAlignAttrName(), alignmentAttr);
39  SmallVector<mlir::NamedAttribute> attrs = {namedAttr};
40  auto alignDictAttr = builder.getDictionaryAttr(attrs);
41  // From
42  // https://llvm.org/docs/LangRef.html#llvm-masked-expandload-intrinsics
43  // https://llvm.org/docs/LangRef.html#llvm-masked-compressstore-intrinsics
44  //
45  // The align parameter attribute can be provided for [expandload]'s first
46  // argument. The align parameter attribute can be provided for
47  // [compressstore]'s second argument.
48  int pos = isExpandLoad ? 0 : 1;
49  return pos == 0 ? builder.getArrayAttr(
50  {alignDictAttr, emptyDictAttr, emptyDictAttr})
51  : builder.getArrayAttr(
52  {emptyDictAttr, alignDictAttr, emptyDictAttr});
53 }

59 LogicalResult LLVM::masked_gather::verify() {
60  auto ptrsVectorType = getPtrs().getType();
61  Type expectedPtrsVectorType =
62  LLVM::getVectorType(extractVectorElementType(ptrsVectorType),
63  LLVM::getVectorNumElements(getRes().getType()));
64  // Vector of pointers type should match result vector type, other than the
65  // element type.
66  if (ptrsVectorType != expectedPtrsVectorType)
67  return emitOpError("expected operand #1 type to be ")
68  << expectedPtrsVectorType;
69  return success();
70 }
