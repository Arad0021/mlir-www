34 /// Forwards a reported remark to the Python callable in `userData`; the
35 /// PyRemark is invalidated once the callable returns, like a PyDiagnostic.
36 static void pyRemarkCallback(MlirRemark remark, void *userData) {
37  auto *data = static_cast<PyRemarkCallbackData *>(userData);
38  // Since this can be called from arbitrary C++ contexts, always get the gil.
39  nb::gil_scoped_acquire gil;
40  // A remark postponed until the context's destruction cannot be delivered:
41  // the Python wrapper of the context is already gone.
42  if (!PyMlirContext::isLiveContext(data->context))
43  return;
44  PyRemark *pyRemark = new PyRemark(remark);
45  nb::object pyRemarkObject = nb::cast(pyRemark, nb::rv_policy::take_ownership);
46  try {
47  data->callback(pyRemarkObject);
48  } catch (std::exception &e) {
49  fprintf(stderr, "MLIR Python remark callback raised exception: %s\n",
50  e.what());
51  }
52  pyRemark->invalidate();
53 }
54 
55 static void pyRemarkCallbackDelete(void *userData) {
56  nb::gil_scoped_acquire gil;
57  delete static_cast<PyRemarkCallbackData *>(userData);
58 }

65 void PyMlirContext::enableRemarks(
66  PyRemarkPolicy policy, const std::optional<std::string> &outputFile,
67  PyRemarkFormat format, const std::string &allFilter,
68  const std::string &passedFilter, const std::string &missedFilter,
69  const std::string &analysisFilter, const std::string &failedFilter,
70  nb::object callback, std::optional<bool> printAsEmitRemarks) {
71  bool hasCallback = !callback.is_none();
72  if (outputFile && hasCallback)
73  throw nb::value_error(
74  "a remark callback cannot be combined with an output file");
75  if (mlirContextHasRemarkEngine(get())) {
76  throw nb::value_error("remarks are already enabled on this context; call "
77  "finalize_remarks() first");
78  }
79 
80  MlirRemarkCategories categories{
81  toMlirStringRef(allFilter), toMlirStringRef(passedFilter),
82  toMlirStringRef(missedFilter), toMlirStringRef(analysisFilter),
83  toMlirStringRef(failedFilter)};
84  auto remarkPolicy = static_cast<MlirRemarkPolicy>(policy);
85  // Without a sink of its own, the engine prints as MLIR remark diagnostics.
86  bool printAsEmit =
87  printAsEmitRemarks.value_or(!outputFile.has_value() && !hasCallback);
88 
89  MlirLogicalResult result;
90  if (outputFile) {
91  result = mlirContextEnableOptimizationRemarksToFile(
92  get(), categories, remarkPolicy,
93  static_cast<MlirRemarkFileFormat>(format), toMlirStringRef(*outputFile),
94  printAsEmit);
95  if (mlirLogicalResultIsFailure(result))
96  throw nb::value_error(
97  ("failed to enable remarks: cannot write '" + *outputFile + "'")
98  .c_str());
99  } else if (hasCallback) {
100  auto *data = new PyRemarkCallbackData{get(), std::move(callback)};
101  result = mlirContextEnableOptimizationRemarksWithCallback(
102  get(), categories, remarkPolicy, pyRemarkCallback, data,
103  pyRemarkCallbackDelete, printAsEmit);
104  } else {
105  result = mlirContextEnableOptimizationRemarks(get(), categories,
106  remarkPolicy, printAsEmit);
107  }
108  if (mlirLogicalResultIsFailure(result))
109  throw nb::value_error("failed to enable remarks");
110 }

162 nb::list PyRemark::getArgs() const {
163  checkValid();
164  nb::list args;
165  intptr_t numArgs = mlirRemarkGetNumArgs(remark);
166  for (intptr_t i = 0; i < numArgs; ++i) {
167  args.append(nb::make_tuple(remarkStr(mlirRemarkGetArgKey(remark, i)),
168  remarkStr(mlirRemarkGetArgValue(remark, i))));
169  }
170  return args;
171 }

173 nb::str PyRemark::getMessage() const {
174  checkValid();
175  std::string message;
176  auto callback = +[](MlirStringRef ref, void *userData) {
177  static_cast<std::string *>(userData)->append(ref.data, ref.length);
178  };
179  mlirRemarkPrint(remark, callback, &message);
180  return nb::str(message.c_str(), message.size());
181 }

183 void populateIRRemarks(nb::module_ &m) {
184  nb::enum_<PyRemarkKind>(m, "RemarkKind")
185  .value("UNKNOWN", PyRemarkKind::Unknown)
186  .value("PASSED", PyRemarkKind::Passed)
187  .value("MISSED", PyRemarkKind::Missed)
188  .value("FAILURE", PyRemarkKind::Failure)
189  .value("ANALYSIS", PyRemarkKind::Analysis);
190 
191  nb::enum_<PyRemarkPolicy>(m, "RemarkPolicy")
192  .value("ALL", PyRemarkPolicy::All)
193  .value("FINAL", PyRemarkPolicy::Final);
194 
195  nb::enum_<PyRemarkFormat>(m, "RemarkFormat")
196  .value("YAML", PyRemarkFormat::YAML)
197  .value("BITSTREAM", PyRemarkFormat::Bitstream);
198 
199  nb::class_<PyRemark>(m, "Remark")
200  .def_prop_ro("kind", &PyRemark::getKind,
201  "Returns the kind of the remark.")
202  .def_prop_ro("remark_name", &PyRemark::getRemarkName,
203  "Returns the name identifying the remark.")
204  .def_prop_ro("category_name", &PyRemark::getCategoryName,
205  "Returns the category of the remark (what the filters "
206  "match).")
207  .def_prop_ro("full_category_name", &PyRemark::getFullCategoryName,
208  "Returns the combined `category:subcategory` name.")
209  .def_prop_ro("function_name", &PyRemark::getFunctionName,
210  "Returns the name of the function the remark refers to.")
211  .def_prop_ro("location", &PyRemark::getLocation,
212  "Returns the location associated with the remark.")
213  .def_prop_ro("remark_id", &PyRemark::getId,
214  "Returns the id of the remark within its engine (0 when "
215  "unset).")
216  .def_prop_ro("args", &PyRemark::getArgs,
217  "Returns the key/value arguments as a list of tuples.")
218  .def_prop_ro("message", &PyRemark::getMessage,
219  "Returns the textual form of the remark, `[Kind] name | "
220  "Category:... | key=value, ...`.")
221  .def(
222  "__str__",
223  [](PyRemark &self) -> nb::str {
224  if (!self.isValid())
225  return nb::str("<Invalid Remark>");
226  return self.getMessage();
227  },
228  "Returns the remark message as a string.");
229 }
