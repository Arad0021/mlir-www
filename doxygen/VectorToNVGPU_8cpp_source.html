46 /// For a vector TransferOpType `xferOp`, an empty `indices` vector, and an
47 /// AffineMap representing offsets to apply to indices, the function fills
48 /// `indices` with the original indices plus the offsets. The offsets are
49 /// applied by taking into account the permutation map of the transfer op. If
50 /// the `offsetMap` has dimension placeholders, those should be provided in
51 /// `dimValues`.
52 template <typename TransferOpType>
53 static void getXferIndices(RewriterBase &rewriter, TransferOpType xferOp,
54  AffineMap offsetMap, ArrayRef<Value> dimValues,
55  SmallVector<Value, 4> &indices) {
56  indices.append(xferOp.getIndices().begin(), xferOp.getIndices().end());
57  Location loc = xferOp.getLoc();
58  unsigned offsetsIdx = 0;
59  for (auto expr : xferOp.getPermutationMap().getResults()) {
60  if (auto dim = dyn_cast<AffineDimExpr>(expr)) {
61  Value prevIdx = indices[dim.getPosition()];
62  SmallVector<OpFoldResult, 3> dims(dimValues);
63  dims.push_back(prevIdx);
64  AffineExpr d0 = rewriter.getAffineDimExpr(offsetMap.getNumDims());
65  indices[dim.getPosition()] = affine::makeComposedAffineApply(
66  rewriter, loc, d0 + offsetMap.getResult(offsetsIdx++), dims);
67  continue;
68  }
69  }
70 }

72 // Return true if the contract op can be convert to MMA matmul.
73 static bool contractSupportsMMAMatrixType(vector::ContractionOp contract) {
74  using MapList = ArrayRef<ArrayRef<AffineExpr>>;
75  auto infer = [&](MapList m) {
76  return AffineMap::inferFromExprList(m, contract.getContext());
77  };
78  AffineExpr m, n, k;
79  bindDims(contract.getContext(), m, n, k);
80  auto iteratorTypes = contract.getIteratorTypes().getValue();
81  if (!(vector::isParallelIterator(iteratorTypes[0]) &&
82  vector::isParallelIterator(iteratorTypes[1]) &&
83  vector::isReductionIterator(iteratorTypes[2])))
84  return false;
85 
86  // The contract needs to represent a matmul to be able to convert to
87  // MMAMatrix matmul.
88  if (contract.getIndexingMapsArray() != infer({{m, k}, {n, k}, {m, n}}))
89  return false;
90 
91  return true;
92 }

94 /// Returns true if the extract strided slice op is supported with `mma.sync`
95 /// path.
96 static bool
97 extractStridedSliceSupportsMMAMatrixType(vector::ExtractStridedSliceOp op) {
98 
99  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
100  nvgpu::getWarpMatrixInfo(op);
101  if (failed(warpMatrixInfo))
102  return false;
103 
104  FailureOr<vector::ContractionOp> contractOp = nvgpu::getUserContract(op);
105  if (failed(contractOp))
106  return false;
107 
108  // Handle vector.extract_strided_slice on registers containing
109  // matrixB and matrixC operands. vector.extract_strided_slice op
110  // is not supported on registers containing matrixA operands.
111  if (warpMatrixInfo->operandRole == nvgpu::MatMulOperandRole::B)
112  return (cast<VectorType>(op->getResult(0).getType()) ==
113  cast<VectorType>((*contractOp).getRhs().getType()));
114  if (warpMatrixInfo->operandRole == nvgpu::MatMulOperandRole::C)
115  return (cast<VectorType>(op->getResult(0).getType()) ==
116  cast<VectorType>((*contractOp).getAcc().getType()));
117 
118  return false;
119 }

121 static bool supportsMMaMatrixType(Operation *op) {
122  if (isa<scf::ForOp, scf::YieldOp>(op))
123  return true;
124  // MMA ops cannot honor a mask. An op masked by an enclosing vector.mask has
125  // no mask operand, so the checks below would miss it; rewriting it in place
126  // would leave non-maskable ops inside the vector.mask region.
127  auto maskableOp = dyn_cast<vector::MaskableOpInterface>(op);
128  if (maskableOp && maskableOp.isMasked())
129  return false;
130  if (auto transferRead = dyn_cast<vector::TransferReadOp>(op))
131  return nvgpu::canLowerToWarpMatrixOperation(transferRead);
132  if (auto transferWrite = dyn_cast<vector::TransferWriteOp>(op))
133  return nvgpu::canLowerToWarpMatrixOperation(transferWrite);
134  if (auto extractStridedSlice = dyn_cast<vector::ExtractStridedSliceOp>(op))
135  return extractStridedSliceSupportsMMAMatrixType(extractStridedSlice);
136  if (auto contract = dyn_cast<vector::ContractionOp>(op))
137  return contractSupportsMMAMatrixType(contract);
138  if (auto constant = dyn_cast<arith::ConstantOp>(op))
139  return constantSupportsMMAMatrixType(constant);
140  if (auto broadcast = dyn_cast<vector::BroadcastOp>(op))
141  return broadcastSupportsMMAMatrixType(broadcast);
142  if (auto signedExtend = dyn_cast<arith::ExtSIOp>(op))
143  return integerExtendSupportsMMAMatrixType<arith::ExtSIOp>(signedExtend);
144  if (auto unsignedExtend = dyn_cast<arith::ExtUIOp>(op))
145  return integerExtendSupportsMMAMatrixType<arith::ExtUIOp>(unsignedExtend);
146  return elementwiseSupportsMMAMatrixType(op);
147 }

149 // Analyze slice of operations based on convert op to figure out if the whole
150 // slice can be converted to MMA operations.
151 static SetVector<Operation *> getOpToConvert(mlir::Operation *op) {
152  auto hasVectorDest = [](Operation *op) {
153  return llvm::any_of(op->getResultTypes(), llvm::IsaPred<VectorType>);
154  };
155  BackwardSliceOptions backwardSliceOptions;
156  backwardSliceOptions.filter = hasVectorDest;
157 
158  auto hasVectorSrc = [](Operation *op) {
159  return llvm::any_of(op->getOperandTypes(), llvm::IsaPred<VectorType>);
160  };
161  ForwardSliceOptions forwardSliceOptions;
162  forwardSliceOptions.filter = hasVectorSrc;
163 
164  DenseMap<Operation *, SmallVector<Operation *>> backwardSliceCache;
165  DenseMap<Operation *, SmallVector<Operation *>> forwardSliceCache;
166  DenseMap<Operation *, bool> supportsMMAMatrixTypeCache;
167 
168  auto getCachedBackwardSlice =
169  [&](Operation *currentOp) -> ArrayRef<Operation *> {
170  auto [it, inserted] = backwardSliceCache.try_emplace(currentOp);
171  if (!inserted)
172  return it->second;
173 
174  SetVector<Operation *> backwardSlice;
175  LogicalResult result =
176  getBackwardSlice(currentOp, &backwardSlice, backwardSliceOptions);
177  assert(result.succeeded() && "expected a backward slice");
178  (void)result;
179  it->second = backwardSlice.takeVector();
180  return it->second;
181  };
182 
183  auto getCachedForwardSlice =
184  [&](Operation *currentOp) -> ArrayRef<Operation *> {
185  auto [it, inserted] = forwardSliceCache.try_emplace(currentOp);
186  if (!inserted)
187  return it->second;
188 
189  SetVector<Operation *> forwardSlice;
190  // Special case for ForOp, we don't want to include the whole region but
191  // only the value using the region arguments.
192  // TODO: We should refine this to only care about the region arguments being
193  // converted to matrix type.
194  if (auto forOp = dyn_cast<scf::ForOp>(currentOp)) {
195  for (Value forOpResult : forOp.getResults())
196  getForwardSlice(forOpResult, &forwardSlice, forwardSliceOptions);
197  for (BlockArgument &arg : forOp.getRegionIterArgs())
198  getForwardSlice(arg, &forwardSlice, forwardSliceOptions);
199  } else {
200  getForwardSlice(currentOp, &forwardSlice, forwardSliceOptions);
201  }
202  it->second = forwardSlice.takeVector();
203  return it->second;
204  };
205 
206  auto cachedSupportsMMAMatrixType = [&](Operation *currentOp) {
207  auto [it, inserted] =
208  supportsMMAMatrixTypeCache.try_emplace(currentOp, false);
209  if (inserted)
210  it->second = supportsMMaMatrixType(currentOp);
211  return it->second;
212  };
213 
214  SetVector<Operation *> opToConvert;
215  op->walk([&](Operation *nestedOp) {
216  if (!isa<vector::ContractionOp>(nestedOp) &&
217  !elementwiseSupportsMMAMatrixType(nestedOp))
218  return;
219  if (backwardSliceCache.contains(nestedOp))
220  return;
221 
222  SetVector<Operation *> dependentOps;
223  dependentOps.insert(nestedOp);
224  unsigned currentIndex = 0;
225  while (currentIndex != dependentOps.size()) {
226  Operation *currentOp = dependentOps[currentIndex++];
227  dependentOps.insert_range(getCachedBackwardSlice(currentOp));
228  dependentOps.insert_range(getCachedForwardSlice(currentOp));
229  }
230  // If any instruction cannot use MMA matrix type drop the whole
231  // chain. MMA matrix are stored in an opaque type so they cannot be used
232  // by all operations.
233  if (llvm::any_of(dependentOps, [&](Operation *op) {
234  if (!cachedSupportsMMAMatrixType(op)) {
235  LDBG() << "cannot convert op: " << *op;
236  return true;
237  }
238  return false;
239  }))
240  return;
241 
242  opToConvert.insert_range(dependentOps);
243  });
244  // Sort the operations so that we can convert them in topological order.
245  return topologicalSort(opToConvert);
246 }

294 /// Check if the loaded matrix operand requires transposed.
295 /// Transposed Map Example:
296 /// Example 1 : (..., d0, d1) -> (d1 * 1, d0 * 2)
297 /// Example 2 : (d0, d1, d2, d3) -> (d3, d2)
298 /// The code below checks if the output 2D is transposed using a generalized
299 /// version : (d0, d1, dn, ..., dm, ...) -> (dm, dn)
300 /// Returns : true; if m > n, false o.w.
301 static FailureOr<bool> isTransposed(vector::TransferReadOp op) {
302  mlir::AffineMap map = op.getPermutationMap();
303 
304  if (map.getNumResults() != 2) {
305  LDBG() << "Failed because the result of `vector.transfer_read` "
306  "is not a 2d operand";
307  return failure();
308  }
309 
310  // Output 2D matrix dimensions in the order of d0, d1.
311  mlir::AffineExpr dM = map.getResult(0);
312  mlir::AffineExpr dN = map.getResult(1);
313 
314  // Find the position of these expressions in the input.
315  auto exprM = dyn_cast<AffineDimExpr>(dM);
316  auto exprN = dyn_cast<AffineDimExpr>(dN);
317 
318  if (!exprM || !exprN) {
319  LDBG() << "Failed because expressions are not affine dim "
320  "expressions, then transpose cannot be determined.";
321  return failure();
322  }
323 
324  return exprM.getPosition() > exprN.getPosition();
325 }

327 static LogicalResult
328 creatLdMatrixCompatibleLoads(RewriterBase &rewriter, vector::TransferReadOp op,
329  llvm::DenseMap<Value, Value> &valueMapping) {
330  OpBuilder::InsertionGuard g(rewriter);
331  rewriter.setInsertionPoint(op);
332  Location loc = op->getLoc();
333 
334  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
335  nvgpu::getWarpMatrixInfo(op);
336  if (failed(warpMatrixInfo)) {
337  LDBG() << "no warpMatrixInfo";
338  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
339  }
340 
341  FailureOr<nvgpu::FragmentElementInfo> regInfo =
342  nvgpu::getMmaSyncRegisterType(*warpMatrixInfo);
343  if (failed(regInfo)) {
344  LDBG() << "not mma sync reg info";
345  return rewriter.notifyMatchFailure(op, "not mma sync reg info");
346  }
347 
348  FailureOr<bool> transpose = isTransposed(op);
349  if (failed(transpose)) {
350  LDBG() << "failed to determine the transpose";
351  return rewriter.notifyMatchFailure(
352  op, "Op should likely not be converted to a nvgpu.ldmatrix call.");
353  }
354 
355  FailureOr<nvgpu::LdMatrixParams> params =
356  nvgpu::getLdMatrixParams(*warpMatrixInfo, *transpose);
357 
358  if (failed(params)) {
359  LDBG() << "failed to convert vector.transfer_read to ldmatrix. "
360  << "Op should likely not be converted to a nvgpu.ldmatrix call.";
361  return rewriter.notifyMatchFailure(
362  op, "failed to convert vector.transfer_read to ldmatrix; this op "
363  "likely should not be converted to a nvgpu.ldmatrix call.");
364  }
365 
366  // Adjust the load offset.
367  auto laneId = gpu::LaneIdOp::create(rewriter, loc, /*upper_bound=*/nullptr);
368  FailureOr<AffineMap> offsets =
369  nvgpu::getLaneIdToLdMatrixMatrixCoord(rewriter, loc, *params);
370  if (failed(offsets)) {
371  LDBG() << "no offsets";
372  return rewriter.notifyMatchFailure(op, "no offsets");
373  }
374 
375  VectorType vectorType = getMmaSyncVectorOperandType(*regInfo);
376 
377  SmallVector<Value, 4> indices;
378  getXferIndices<vector::TransferReadOp>(rewriter, op, *offsets, {laneId},
379  indices);
380 
381  nvgpu::LdMatrixOp newOp =
382  nvgpu::LdMatrixOp::create(rewriter, loc, vectorType, op.getBase(),
383  indices, *transpose, params->numTiles);
384  valueMapping[op] = newOp->getResult(0);
385  return success();
386 }

388 static LogicalResult
389 createNonLdMatrixLoads(RewriterBase &rewriter, vector::TransferReadOp op,
390  llvm::DenseMap<Value, Value> &valueMapping) {
391  OpBuilder::InsertionGuard g(rewriter);
392  rewriter.setInsertionPoint(op);
393 
394  Location loc = op.getLoc();
395  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
396  nvgpu::getWarpMatrixInfo(op);
397  if (failed(warpMatrixInfo))
398  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
399  FailureOr<nvgpu::FragmentElementInfo> regInfo =
400  nvgpu::getMmaSyncRegisterType(*warpMatrixInfo);
401  if (failed(regInfo)) {
402  return rewriter.notifyMatchFailure(
403  op, "Failed to deduce register fragment type during "
404  "conversion to distributed non-ldmatrix compatible load");
405  }
406 
407  Value laneId = gpu::LaneIdOp::create(rewriter, loc, /*upper_bound=*/nullptr);
408 
409  // This is the individual element type.
410  Type loadedElType = regInfo->registerLLVMType;
411  VectorType vectorType = getMmaSyncVectorOperandType(*regInfo);
412 
413  Value fill = arith::ConstantOp::create(
414  rewriter, op.getLoc(), vectorType.getElementType(),
415  rewriter.getZeroAttr(vectorType.getElementType()));
416  Value result =
417  vector::BroadcastOp::create(rewriter, op.getLoc(), vectorType, fill);
418 
419  bool isTransposeLoad = !op.getPermutationMap().isMinorIdentity();
420 
421  // If we are not transposing, then we can use vectorized loads. Otherwise, we
422  // must load each element individually.
423  if (!isTransposeLoad) {
424  if (!isa<VectorType>(loadedElType)) {
425  loadedElType = VectorType::get({1}, loadedElType);
426  }
427 
428  for (int i = 0; i < vectorType.getShape()[0]; i++) {
429  FailureOr<AffineMap> coords = nvgpu::getLaneIdAndValueIdToOperandCoord(
430  rewriter, op.getLoc(), *warpMatrixInfo);
431  if (failed(coords))
432  return rewriter.notifyMatchFailure(op, "no coords");
433 
434  Value logicalValueId = arith::ConstantOp::create(
435  rewriter, loc, rewriter.getIndexType(),
436  rewriter.getIndexAttr(i * regInfo->elementsPerRegister));
437  SmallVector<Value, 4> newIndices;
438  getXferIndices<vector::TransferReadOp>(
439  rewriter, op, *coords, {laneId, logicalValueId}, newIndices);
440 
441  Value el = vector::LoadOp::create(rewriter, loc, loadedElType,
442  op.getBase(), newIndices);
443  result = vector::InsertOp::create(rewriter, loc, el, result, i);
444  }
445  } else {
446  if (auto vecType = dyn_cast<VectorType>(loadedElType)) {
447  loadedElType = vecType.getElementType();
448  }
449  for (int i = 0; i < vectorType.getShape()[0]; i++) {
450  for (unsigned innerIdx = 0; innerIdx < vectorType.getShape()[1];
451  innerIdx++) {
452 
453  Value logicalValueId = arith::ConstantOp::create(
454  rewriter, loc, rewriter.getIndexType(),
455  rewriter.getIndexAttr(i * regInfo->elementsPerRegister + innerIdx));
456  FailureOr<AffineMap> coords = nvgpu::getLaneIdAndValueIdToOperandCoord(
457  rewriter, op.getLoc(), *warpMatrixInfo);
458  if (failed(coords))
459  return rewriter.notifyMatchFailure(op, "no coords");
460 
461  SmallVector<Value, 4> newIndices;
462  getXferIndices<vector::TransferReadOp>(
463  rewriter, op, *coords, {laneId, logicalValueId}, newIndices);
464  Value el = memref::LoadOp::create(rewriter, op.getLoc(), loadedElType,
465  op.getBase(), newIndices);
466  result = vector::InsertOp::create(rewriter, op.getLoc(), el, result,
467  ArrayRef<int64_t>{i, innerIdx});
468  }
469  }
470  }
471 
472  valueMapping[op.getResult()] = result;
473  return success();
474 }

484 /// Converts a `vector.transfer_read` operation directly to either a
485 /// `vector.load` or a `nvgpu.ldmatrix` operation. This function should only be
486 /// used when converting to `nvgpu.mma.sync` operations.
487 static LogicalResult
488 convertTransferReadToLoads(RewriterBase &rewriter, vector::TransferReadOp op,
489  llvm::DenseMap<Value, Value> &valueMapping) {
490  OpBuilder::InsertionGuard g(rewriter);
491  rewriter.setInsertionPoint(op);
492 
493  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
494  nvgpu::getWarpMatrixInfo(op);
495  if (failed(warpMatrixInfo))
496  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
497 
498  bool isLdMatrixCompatible =
499  isSharedMemory(cast<MemRefType>(op.getBase().getType())) &&
500  nvgpu::inferTileWidthInBits(*warpMatrixInfo) == 128;
501 
502  VectorType vecTy = op.getVectorType();
503  int64_t bitWidth = vecTy.getElementType().getIntOrFloatBitWidth();
504 
505  // When we are transposing the B operand, ldmatrix will only work if we have
506  // at least 8 rows to read and the width to read for the transpose is 128
507  // bits.
508  if (!op.getPermutationMap().isMinorIdentity() &&
509  (bitWidth != 16 || vecTy.getDimSize(1) < 8 ||
510  vecTy.getDimSize(0) * bitWidth < 128))
511  isLdMatrixCompatible = false;
512 
513  if (!isLdMatrixCompatible)
514  return createNonLdMatrixLoads(rewriter, op, valueMapping);
515 
516  return creatLdMatrixCompatibleLoads(rewriter, op, valueMapping);
517 }

519 static LogicalResult
520 convertTransferWriteToStores(RewriterBase &rewriter, vector::TransferWriteOp op,
521  llvm::DenseMap<Value, Value> &valueMapping) {
522  OpBuilder::InsertionGuard g(rewriter);
523  rewriter.setInsertionPoint(op);
524 
525  Location loc = op->getLoc();
526  auto it = valueMapping.find(op.getVector());
527  if (it == valueMapping.end())
528  return rewriter.notifyMatchFailure(op, "no mapping");
529  Value matrix = it->second;
530 
531  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
532  nvgpu::getWarpMatrixInfo(op);
533  if (failed(warpMatrixInfo))
534  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
535  FailureOr<nvgpu::FragmentElementInfo> regInfo =
536  nvgpu::getMmaSyncRegisterType(*warpMatrixInfo);
537  if (failed(regInfo))
538  return rewriter.notifyMatchFailure(op, "not mma sync reg info");
539 
540  VectorType vectorType = getMmaSyncVectorOperandType(*regInfo);
541  Value laneId = gpu::LaneIdOp::create(rewriter, loc, /*upper_bound=*/nullptr);
542 
543  for (unsigned i = 0; i < vectorType.getShape()[0]; i++) {
544  Value logicalValueId = arith::ConstantOp::create(
545  rewriter, loc, rewriter.getIndexType(),
546  rewriter.getIndexAttr(i * regInfo->elementsPerRegister));
547  FailureOr<AffineMap> coords = nvgpu::getLaneIdAndValueIdToOperandCoord(
548  rewriter, op.getLoc(), *warpMatrixInfo);
549  if (failed(coords))
550  return rewriter.notifyMatchFailure(op, "no coords");
551 
552  Value el =
553  vector::ExtractOp::create(rewriter, loc, matrix, ArrayRef<int64_t>{i});
554  SmallVector<Value, 4> newIndices;
555  getXferIndices<vector::TransferWriteOp>(
556  rewriter, op, *coords, {laneId, logicalValueId}, newIndices);
557  vector::StoreOp::create(rewriter, loc, el, op.getBase(), newIndices);
558  }
559 
560  LDBG() << "erase: " << op;
561  rewriter.eraseOp(op);
562  return success();
563 }

565 static void populateFromInt64AttrArray(ArrayAttr arrayAttr,

571 static LogicalResult
572 convertExtractStridedSlice(RewriterBase &rewriter,
573  vector::ExtractStridedSliceOp op,
574  llvm::DenseMap<Value, Value> &valueMapping) {
575  OpBuilder::InsertionGuard g(rewriter);
576  rewriter.setInsertionPoint(op);
577 
578  Location loc = op->getLoc();
579 
580  FailureOr<nvgpu::WarpMatrixInfo> warpMatrixInfo =
581  nvgpu::getWarpMatrixInfo(op);
582  if (failed(warpMatrixInfo))
583  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
584 
585  FailureOr<nvgpu::FragmentElementInfo> mmaSyncFragmentInfo =
586  nvgpu::getMmaSyncRegisterType(*warpMatrixInfo);
587  if (failed(mmaSyncFragmentInfo))
588  return rewriter.notifyMatchFailure(op, "no mmaSyncFragmentInfo");
589 
590  // Find the vector.transer_read whose result vector is being sliced.
591  auto transferReadOp = op.getSource().getDefiningOp<vector::TransferReadOp>();
592  if (!transferReadOp)
593  return rewriter.notifyMatchFailure(op, "no transfer read");
594 
595  warpMatrixInfo = nvgpu::getWarpMatrixInfo(transferReadOp);
596  if (failed(warpMatrixInfo))
597  return rewriter.notifyMatchFailure(op, "no warpMatrixInfo");
598 
599  FailureOr<nvgpu::FragmentElementInfo> ldFragmentInfo =
600  nvgpu::getMmaSyncRegisterType(*warpMatrixInfo);
601  if (failed(ldFragmentInfo))
602  return rewriter.notifyMatchFailure(op, "no ldFragmentInfo");
603 
604  assert(
605  (mmaSyncFragmentInfo->elementsPerRegister ==
606  ldFragmentInfo->elementsPerRegister) &&
607  "Number of elements per register should be same for load and mma.sync");
608 
609  // Create vector.extract_strided_slice op for thread-owned fragments.
610  std::array<int64_t, 2> strides = {1,
611  1}; // stride for extract slice is always 1.
612  std::array<int64_t, 2> sliceShape = {
613  mmaSyncFragmentInfo->numRegistersPerFragment,
614  mmaSyncFragmentInfo->elementsPerRegister};
615  auto it = valueMapping.find(transferReadOp);
616  if (it == valueMapping.end())
617  return rewriter.notifyMatchFailure(op, "no mapping");
618  auto sourceVector = it->second;
619 
620  // offset and sizes at warp-level of onwership.
621  SmallVector<int64_t> offsets;
622  populateFromInt64AttrArray(op.getOffsets(), offsets);
623 
624  SmallVector<int64_t> sizes;
625  populateFromInt64AttrArray(op.getSizes(), sizes);
626  ArrayRef<int64_t> warpVectorShape = op.getSourceVectorType().getShape();
627 
628  // Compute offset in vector registers. Note that the mma.sync vector registers
629  // are shaped as numberOfFragments x numberOfRegistersPerfFragment. The vector
630  // registers can only be sliced along numberOfFragments, i.e., sliceOffset[0].
631  std::array<int64_t, 2> sliceOffset = {0, 0};
632 
633  if (offsets[0] && offsets[1])
634  return op->emitError() << "Slicing fragments in 2D is not supported. ";
635  if (offsets[0])
636  sliceOffset[0] = (warpVectorShape[0] / offsets[0]);
637  else if (offsets[1])
638  sliceOffset[0] = (warpVectorShape[1] / offsets[1]);
639 
640  Value newOp = vector::ExtractStridedSliceOp::create(
641  rewriter, loc, sourceVector, sliceOffset, sliceShape, strides);
642 
643  valueMapping[op] = newOp;
644  return success();
645 }

647 static LogicalResult
648 convertContractOpToMmaSync(RewriterBase &rewriter, vector::ContractionOp op,
649  llvm::DenseMap<Value, Value> &valueMapping) {
650  OpBuilder::InsertionGuard g(rewriter);
651  rewriter.setInsertionPoint(op);
652 
653  auto itA = valueMapping.find(op.getLhs());
654  auto itB = valueMapping.find(op.getRhs());
655  auto itC = valueMapping.find(op.getAcc());
656  if (itA == valueMapping.end() || itB == valueMapping.end() ||
657  itC == valueMapping.end())
658  return rewriter.notifyMatchFailure(op, "no mapping");
659  Value opA = itA->second, opB = itB->second, opC = itC->second;
660  int64_t m = cast<VectorType>(op.getLhs().getType()).getShape()[0];
661  int64_t n = cast<VectorType>(op.getRhs().getType()).getShape()[0];
662  int64_t k = cast<VectorType>(op.getLhs().getType()).getShape()[1];
663  Value matmul = nvgpu::MmaSyncOp::create(rewriter, op.getLoc(), opA, opB, opC,
664  rewriter.getI64ArrayAttr({m, n, k}));
665  valueMapping[op.getResult()] = matmul;
666  return success();
667 }

674 LogicalResult mlir::convertVectorToNVVMCompatibleMMASync(RewriterBase &rewriter,
675  Operation *rootOp) {
676  SetVector<Operation *> ops = getOpToConvert(rootOp);
677  llvm::DenseMap<Value, Value> valueMapping;
678  for (Operation *op : ops) {
679  if (llvm::TypeSwitch<Operation *, LogicalResult>(op)
680  .Case([&](vector::TransferReadOp transferReadOp) {
681  return convertTransferReadToLoads(rewriter, transferReadOp,
682  valueMapping);
683  })
684  .Case([&](vector::TransferWriteOp transferWriteOp) {
685  return convertTransferWriteToStores(rewriter, transferWriteOp,
686  valueMapping);
687  })
688  .Case([&](vector::ExtractStridedSliceOp extractStridedSliceOp) {
689  return convertExtractStridedSlice(rewriter, extractStridedSliceOp,
690  valueMapping);
691  })
692  .Case([&](vector::ContractionOp contractionOp) {
693  return convertContractOpToMmaSync(rewriter, contractionOp,
694  valueMapping);
695  })
696  .Case([&](scf::ForOp forOp) {
697  return convertForOpForMMA(rewriter, forOp, valueMapping);
698  })
699  .Case([&](scf::YieldOp yieldOp) {
700  return convertYieldOpForMMA(rewriter, yieldOp, valueMapping);
701  })
702  .Case([&](arith::ConstantOp constOp) {
703  return convertConstantOpMmaSync(rewriter, constOp, valueMapping);
704  })
705  .Default([&](Operation *op) {
706  return op->emitError() << "unhandled vector to mma type: " << *op;
707  })
708  .failed()) {
709  return op->emitOpError()
710  << "failed to convert op during vector-to-nvgpu conversion";
711  }
712  }
713  return success();
714 }
