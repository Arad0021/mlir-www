206 namespace {
207 ParseResult parseObject(AsmParser &odsParser, CompilationTarget &format,
208  StringAttr &object) {
209  std::optional<CompilationTarget> formatResult;
210  StringRef enumKeyword;
211  auto loc = odsParser.getCurrentLocation();
212  if (failed(odsParser.parseOptionalKeyword(&enumKeyword)))
213  formatResult = CompilationTarget::Fatbin;
214  if (!formatResult &&
215  (formatResult =
216  gpu::symbolizeEnum<gpu::CompilationTarget>(enumKeyword)) &&
217  odsParser.parseEqual())
218  return odsParser.emitError(loc, "expected an equal sign");
219  if (!formatResult)
220  return odsParser.emitError(loc, "expected keyword for GPU object format");
221  FailureOr<StringAttr> objectResult =
222  FieldParser<StringAttr>::parse(odsParser);
223  if (failed(objectResult))
224  return odsParser.emitError(odsParser.getCurrentLocation(),
225  "failed to parse GPU_ObjectAttr parameter "
226  "'object' which is to be a `StringAttr`");
227  format = *formatResult;
228  object = *objectResult;
229  return success();
230 }

313 KernelTableAttr KernelTableAttr::get(MLIRContext *context,
314  ArrayRef<KernelMetadataAttr> kernels,
315  bool isSorted) {
316  // Note that `is_sorted` is always only invoked once even with assertions ON.
317  assert((!isSorted || llvm::is_sorted(kernels)) &&
318  "expected a sorted kernel array");
319  // Immediately return the attribute if the array is sorted.
320  if (isSorted || llvm::is_sorted(kernels))
321  return Base::get(context, kernels);
322  // Sort the array.
323  SmallVector<KernelMetadataAttr> kernelsTmp(kernels);
324  llvm::array_pod_sort(kernelsTmp.begin(), kernelsTmp.end());
325  return Base::get(context, kernelsTmp);
326 }

448 std::pair<llvm::BumpPtrAllocator, SmallVector<const char *>>
449 TargetOptions::tokenizeCmdOptions(const std::string &cmdOptions) {
450  std::pair<llvm::BumpPtrAllocator, SmallVector<const char *>> options;
451  llvm::StringSaver stringSaver(options.first);
452  StringRef opts = cmdOptions;
453  // For a correct tokenization of the command line options `opts` must be
454  // unquoted, otherwise the tokenization function returns a single string: the
455  // unquoted `cmdOptions` -which is not the desired behavior.
456  // Remove any quotes if they are at the beginning and end of the string:
457  if (!opts.empty() && opts.front() == '"' && opts.back() == '"')
458  opts.consume_front("\""), opts.consume_back("\"");
459  if (!opts.empty() && opts.front() == '\'' && opts.back() == '\'')
460  opts.consume_front("'"), opts.consume_back("'");
461 #ifdef _WIN32
462  llvm::cl::TokenizeWindowsCommandLine(opts, stringSaver, options.second,
463  /*MarkEOLs=*/false);
464 #else
465  llvm::cl::TokenizeGNUCommandLine(opts, stringSaver, options.second,
466  /*MarkEOLs=*/false);
467 #endif // _WIN32
468  return options;
469 }

476 std::pair<llvm::BumpPtrAllocator, SmallVector<const char *>>
477 TargetOptions::tokenizeAndRemoveSuffixCmdOptions(llvm::StringRef startsWith) {
478  size_t startPos = cmdOptions.find(startsWith);
479  if (startPos == std::string::npos)
480  return {llvm::BumpPtrAllocator(), SmallVector<const char *>()};
481 
482  auto tokenized =
483  tokenizeCmdOptions(cmdOptions.substr(startPos + startsWith.size()));
484  cmdOptions.resize(startPos);
485  return tokenized;
