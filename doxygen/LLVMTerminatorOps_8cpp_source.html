33 LogicalResult BrOp::canonicalize(BrOp op, PatternRewriter &rewriter) {
34  // Merge a branch to a block that has a single predecessor, the same
35  // transform as cf::BranchOp; the successor block is inlined into the
36  // branch block and the branch is erased.
37  Block *succ = op.getDest();
38  Block *opParent = op->getBlock();
39  if (succ == opParent || !llvm::hasSingleElement(succ->getPredecessors()))
40  return failure();
41 
42  // If any branch operand is itself a block argument of the successor,
43  // merging would replace uses of that argument with itself and leave
44  // dangling uses after the successor block is erased.
45  for (Value operand : op.getDestOperands())
46  if (auto arg = dyn_cast<BlockArgument>(operand))
47  if (arg.getOwner() == succ)
48  return failure();
49 
50  // The successor must be mergeable. A landing pad must stay the first
51  // operation of its block, a block tag means the block address is taken,
52  // and merging would silently drop a loop annotation on the branch.
53  if (!succ->empty()) {
54  if (isa<LandingpadOp>(succ->front()))
55  return failure();
56  if (llvm::any_of(succ->without_terminator(),
57  [](Operation &inner) { return isa<BlockTagOp>(inner); }))
58  return failure();
59  }
60  if (op.getLoopAnnotationAttr())
61  return failure();
62 
63  SmallVector<Value> brOperands(op.getDestOperands());
64  rewriter.eraseOp(op);
65  rewriter.mergeBlocks(succ, opParent, brOperands);
66  return success();
67 }

148 /// <cases> ::= `[` (case (`,` case )* )? `]`
149 /// <case> ::= integer `:` bb-id (`(` ssa-use-and-type-list `)`)?
150 ParseResult mlir::LLVM::parseSwitchOpCases(
151  OpAsmParser &parser, Type flagType, DenseIntElementsAttr &caseValues,
152  SmallVectorImpl<Block *> &caseDestinations,
153  SmallVectorImpl<SmallVector<OpAsmParser::UnresolvedOperand>> &caseOperands,
154  SmallVectorImpl<SmallVector<Type>> &caseOperandTypes) {
155  if (failed(parser.parseLSquare()))
156  return failure();
157  if (succeeded(parser.parseOptionalRSquare()))
158  return success();
159  SmallVector<APInt> values;
160  unsigned bitWidth = flagType.getIntOrFloatBitWidth();
161  auto parseCase = [&]() {
162  int64_t value = 0;
163  if (failed(parser.parseInteger(value)))
164  return failure();
165  values.push_back(APInt(bitWidth, value, /*isSigned=*/true));
166 
167  Block *destination;
168  SmallVector<OpAsmParser::UnresolvedOperand> operands;
169  SmallVector<Type> operandTypes;
170  if (parser.parseColon() || parser.parseSuccessor(destination))
171  return failure();
172  if (!parser.parseOptionalLParen()) {
173  if (parser.parseOperandList(operands, OpAsmParser::Delimiter::None) ||
174  parser.parseColonTypeList(operandTypes) || parser.parseRParen())
175  return failure();
176  }
177  caseDestinations.push_back(destination);
178  caseOperands.emplace_back(operands);
179  caseOperandTypes.emplace_back(operandTypes);
180  return success();
181  };
182  if (failed(parser.parseCommaSeparatedList(parseCase)))
183  return failure();
184 
185  ShapedType caseValueType =
186  VectorType::get(static_cast<int64_t>(values.size()), flagType);
187  caseValues = DenseIntElementsAttr::get(caseValueType, values);
188  return parser.parseRSquare();
189 }

220 LogicalResult SwitchOp::verify() {
221  if ((!getCaseValues() && !getCaseDestinations().empty()) ||
222  (getCaseValues() &&
223  getCaseValues()->size() !=
224  static_cast<int64_t>(getCaseDestinations().size())))
225  return emitOpError("expects number of case values to match number of "
226  "case destinations");
227  if (getCaseValues() &&
228  getValue().getType() != getCaseValues()->getElementType())
229  return emitError("expects case value type to match condition value type");
230 
231  // LLVM IR rejects a switch with a repeated case value.
232  if (getCaseValues()) {
233  llvm::SmallDenseMap<APInt, unsigned> firstCase;
234  for (auto [index, value] :
235  llvm::enumerate(getCaseValues()->getValues<APInt>())) {
236  auto [it, inserted] = firstCase.try_emplace(value, index);
237  if (inserted)
238  continue;
239  // Diagnostics cannot stream APInt directly, so print it into a buffer.
240  SmallString<32> valueStr;
241  llvm::raw_svector_ostream os(valueStr);
242  value.print(os, /*isSigned=*/true);
243  if (value.isNegative()) {
244  os << " (unsigned ";
245  value.print(os, /*isSigned=*/false);
246  os << ")";
247  }
248  return emitOpError() << "case value " << valueStr
249  << " is repeated (cases " << it->second << " and "
250  << index << ")";
251  }
252  }
253  return success();
254 }

266 LogicalResult ReturnOp::verify() {
267  auto parent = (*this)->getParentOfType<LLVMFuncOp>();
268  if (!parent)
269  return success();
270 
271  Type expectedType = parent.getFunctionType().getReturnType();
272  if (llvm::isa<LLVMVoidType>(expectedType)) {
273  if (!getArg())
274  return success();
275  InFlightDiagnostic diag = emitOpError("expected no operands");
276  diag.attachNote(parent->getLoc()) << "when returning from function";
277  return diag;
278  }
279  if (!getArg()) {
280  if (llvm::isa<LLVMVoidType>(expectedType))
281  return success();
282  InFlightDiagnostic diag = emitOpError("expected 1 operand");
283  diag.attachNote(parent->getLoc()) << "when returning from function";
284  return diag;
285  }
286  if (expectedType != getArg().getType()) {
287  InFlightDiagnostic diag = emitOpError("mismatching result types");
288  diag.attachNote(parent->getLoc()) << "when returning from function";
289  return diag;
290  }
291  return success();
292 }
