38 void GpuAttachSelectObjectPass::runOnOperation() {
39  if (priority > std::numeric_limits<uint16_t>::max()) {
40  getOperation()->emitError()
41  << "Invalid priority specified: " << priority.getValue()
42  << " (expected a value in the range [0, 65535])";
43  return signalPassFailure();
44  }
45  // The default priority is the one of modules with no offloading handler.
46  if (priority == SelectObjectAttr::kDefaultPriority)
47  return;
48 
49  Builder builder(&getContext());
50  llvm::Regex matcher(moduleMatcher);
51  for (Region &region : getOperation()->getRegions()) {
52  for (Block &block : region.getBlocks()) {
53  for (GPUModuleOp module : block.getOps<GPUModuleOp>()) {
54  // Check if the name of the module matches.
55  if (!moduleMatcher.empty() && !matcher.match(module.getName()))
56  continue;
57  // Leave other offloading handlers, and handlers that already specify
58  // a priority, unchanged.
59  Attribute handler = module.getOffloadingHandlerAttr();
60  auto selectObject = dyn_cast_or_null<SelectObjectAttr>(handler);
61  if (handler && !selectObject)
62  continue;
63  if (selectObject &&
64  selectObject.getPriority() != SelectObjectAttr::kDefaultPriority)
65  continue;
66  // A null target selects the first object.
67  Attribute target =
68  selectObject ? selectObject.getTarget() : Attribute();
69  if (!target)
70  target = builder.getI64IntegerAttr(0);
71  module.setOffloadingHandlerAttr(
72  SelectObjectAttr::get(&getContext(), target, priority));
73  }
74  }
75  }
76 }
