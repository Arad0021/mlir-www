9 // Overview:
10 // ---------
11 // OpenACC compute constructs (acc.parallel, acc.serial, acc.kernels) and
12 // acc.loop can carry data clauses (acc.private, acc.firstprivate,
13 // acc.reduction) that refer to recipes (acc.private.recipe,
14 // acc.firstprivate.recipe, acc.reduction.recipe). Recipes define how to
15 // initialize, copy, combine, or destroy a particular variable. This pass clones
16 // those regions into the construct and ensures the materialized SSA values are
17 // used instead.
18 //
19 // Transforms:
20 // -----------
21 // 1. Firstprivate: Inserts acc.firstprivate_map so the initial value is
22 // available on the device, then clones the recipe init and copy regions
23 // into the construct and replaces uses with the materialized alloca.
24 // Optional destroy region is cloned before the region terminator.
25 //
26 // 2. Private: Clones the recipe init region into the construct (at the
27 // region entry or at the loop op for acc.loop private). Replaces uses
28 // of the recipe result with the materialized alloca. Optional destroy
29 // region is cloned before the region terminator.
30 //
31 // 3. Reduction: Creates acc.reduction_init (init region inlined) and
32 // acc.reduction_combine_region (combiner region inlined). Uses within
33 // the region are updated to the reduction init result.
34 //
35 // Requirements:
36 // -------------
37 // 1. OpenACCSupport: The pass uses the `acc::OpenACCSupport` analysis
38 // including emitNYI for unsupported cases.
39 //
40 //===----------------------------------------------------------------------===//

79 static void setLocation(Region &region, Location loc) {
80  // Since recipes are generated per type and not per variable, the location
81  // of the recipe operations which get inlined will not necessarily be the
82  // same as the location of the op that is being materialized. Force an update
83  // of the location of the recipe operations to the location of the op that is
84  // being materialized.
85  region.walk([&](Operation *op) { op->setLoc(loc); });
86 }

88 static void saveVarName(StringRef name, Value dst) {
89  if (name.empty())
90  return;
91  if (Operation *dstOp = dst.getDefiningOp()) {
92  if (dstOp->getDiscardableAttrOfType<acc::VarNameAttr>(
93  acc::getVarNameAttrName()))
94  return;
95  if (isa<ACC_DATA_ENTRY_OPS>(dstOp))
96  return;
97  dstOp->setDiscardableAttr(acc::getVarNameAttrName(),
98  acc::VarNameAttr::get(dstOp->getContext(), name));
99  return;
100  }
101  auto blockArg = dyn_cast<BlockArgument>(dst);
102  if (!blockArg)
103  return;
104  Block *block = blockArg.getOwner();
105  Region *region = block ? block->getParent() : nullptr;
106  if (!region || !block->isEntryBlock())
107  return;
108  Operation *parent = region->getParentOp();
109  if (!parent)
110  return;
111  auto funcOp = dyn_cast<FunctionOpInterface>(parent);
112  if (!funcOp)
113  return;
114  unsigned argIdx = blockArg.getArgNumber();
115  if (argIdx >= funcOp.getNumArguments())
116  return;
117  if (funcOp.getArgAttr(argIdx, acc::getVarNameAttrName()))
118  return;
119  funcOp.setArgAttr(argIdx, acc::getVarNameAttrName(),
120  acc::VarNameAttr::get(parent->getContext(), name));
121 }

123 static void saveVarName(Value src, Value dst) {
124  saveVarName(acc::getVariableName(src), dst);
125 }

127 static void resolveVarNamePlaceholders(Block *block, Block::iterator ip,
128  StringRef name) {
129  StringRef placeholder = acc::getVarNamePlaceholder();
130  for (auto it = block->begin(); it != std::next(ip); ++it) {
131  it->walk([&](Operation *op) {
132  auto attr = op->getDiscardableAttrOfType<acc::VarNameAttr>(
133  acc::getVarNameAttrName());
134  if (!attr || attr.getName() != placeholder)
135  return;
136  if (name.empty())
137  op->removeDiscardableAttr(acc::getVarNameAttrName());
138  else
139  op->setDiscardableAttr(acc::getVarNameAttrName(),
140  acc::VarNameAttr::get(op->getContext(), name));
141  });
142  }
143 }

145 // Clone the destroy region of the recipe before the terminator of the provided
146 // block. Values must be provided for the destroy region block arguments
147 // according to the recipe specifications.
148 template <typename RecipeOpTy>
149 static void cloneDestroy(Location loc, RecipeOpTy recipe, mlir::Block *block,
150  Block::iterator ip,
151  const llvm::SmallVector<mlir::Value> &arguments) {
152  IRMapping mapping{};
153  Region &destroyRegion = recipe.getDestroyRegion();
154  assert(destroyRegion.getBlocks().front().getNumArguments() ==
155  arguments.size() &&
156  "unexpected acc recipe destroy block arguments");
157 
158  setLocation(destroyRegion, loc);
159 
160  mapping.map(destroyRegion.getBlocks().front().getArguments(), arguments);
161  acc::cloneACCRegionInto(&destroyRegion, block, ip, mapping,
162  /*resultsToReplace=*/{});
163 }

165 class ACCRecipeMaterialization
166  : public acc::impl::ACCRecipeMaterializationBase<ACCRecipeMaterialization> {
167 public:
168  using acc::impl::ACCRecipeMaterializationBase<
169  ACCRecipeMaterialization>::ACCRecipeMaterializationBase;
170  void runOnOperation() override;
171 
172 private:
173  // When the recipe reads the original variable, its initial value needs to be
174  // available on the GPU. One way to get that value there is to map the
175  // variable through global memory.
176  // Thus, when we materialize a firstprivate, we materialize it into
177  // a mapping action first. This function ends up with doing the following:
178  // %dev = acc.firstprivate var(%var)
179  // =>
180  // %copy = acc.firstprivate_map var(%var)
181  // %dev = acc.firstprivate var(%copy)
182  // When the recipe materialization happens, the `acc.firstprivate` ends up
183  // being removed. But because of the way we chain it to the
184  // `acc.firstprivate_map`, then its result becomes live-in to the
185  // compute region and used as the variable the initial value is loaded from.
186  template <typename OpTy>
187  void handleInitialValueMapping(OpTy op) const;
188  template <typename OpTy>
189  void removeRecipe(
190  OpTy op, ModuleOp moduleOp,
191  const std::optional<llvm::DenseSet<StringAttr>> &usedSymbols) const;
192  template <typename OpTy, typename RecipeOpTy, typename AccOpTy>
193  LogicalResult materialize(OpTy op, RecipeOpTy recipe, AccOpTy accOp,
194  acc::OpenACCSupport &accSupport,
195  acc::ACCToGPUMappingPolicy &policy) const;
196  template <typename OpTy>
197  LogicalResult materializeForACCOp(OpTy accOp, acc::OpenACCSupport &accSupport,
198  acc::ACCToGPUMappingPolicy &policy) const;
199 };

201 template <typename OpTy>
202 void ACCRecipeMaterialization::handleInitialValueMapping(OpTy op) const {
203  OpBuilder builder(op);
204  auto mapInitialOp = acc::FirstprivateMapInitialOp::create(
205  builder, op.getLoc(), op.getVar(), op.getStructured(), op.getImplicit(),
206  op.getBounds());
207  mapInitialOp.setName(op.getName());
208  op.getVarMutable().assign(mapInitialOp.getAccVar());
209 }

211 // Whether a recipe region reads the variable it privatizes - a descriptor
212 // recipe loads it for the bounds, while a scalar one ignores it. Both init and
213 // destroy receive it as their first argument.
214 static bool readsVar(Region &region) {
215  if (region.empty() || region.getNumArguments() == 0)
216  return false;
217  return !region.getArgument(0).use_empty();
218 }

220 template <typename OpTy>
221 void ACCRecipeMaterialization::removeRecipe(
222  OpTy op, ModuleOp moduleOp,
223  const std::optional<llvm::DenseSet<StringAttr>> &usedSymbols) const {
224  auto recipeName = op.getNameAttr();
225  // Fall back to scanning the module when the symbol uses could not be
226  // gathered up front.
227  bool useEmpty = usedSymbols
228  ? !usedSymbols->contains(recipeName)
229  : SymbolTable::symbolKnownUseEmpty(recipeName, moduleOp);
230  if (useEmpty) {
231  LLVM_DEBUG(llvm::dbgs() << "erasing recipe: " << recipeName << "\n");
232  op.erase();
233  } else {
234  LLVM_DEBUG({
235  std::optional<SymbolTable::UseRange> symbolUses =
236  op.getSymbolUses(moduleOp);
237  if (symbolUses.has_value()) {
238  for (SymbolTable::SymbolUse symbolUse : *symbolUses) {
239  llvm::dbgs() << "symbol use: ";
240  symbolUse.getUser()->dump();
241  }
242  }
243  });
244  llvm_unreachable("expected no use of recipe symbol");
245  }
246 }

248 template <typename OpTy, typename RecipeOpTy, typename AccOpTy>
249 LogicalResult ACCRecipeMaterialization::materialize(
250  OpTy op, RecipeOpTy recipe, AccOpTy accOp, acc::OpenACCSupport &accSupport,
251  acc::ACCToGPUMappingPolicy &policy) const {
252  Region &region = accOp.getRegion();
253  Value origPtr = op.getVar();
254  Value accPtr = op.getAccVar();
255  assert(accPtr && "invalid op: null acc var");
256 
257  OpBuilder b(op);
258  SmallVector<Value> triples;
259 
260  // Clone init block into the region at the insertion point specified.
261  Region &initRegion = recipe.getInitRegion();
262  unsigned initNumArguments =
263  initRegion.getBlocks().front().getArguments().size();
264  if (initNumArguments > 1) {
265  // Code from C/C++ will most likely only provide extent arguments to the
266  // recipe arguments.
267  if ((initNumArguments - 1) % 3 != 0) {
268  (void)accSupport.emitNYI(recipe.getLoc(),
269  "privatization of array section with extents");
270  return failure();
271  }
272  // The remaining arguments must be the bounds triples
273  // (lower-bound, upper-bound, step), ...
274  unsigned argIdx = 1;
275  // Cast the given value to the type of the combiner region's argument
276  // at position argIdx, and increment argIdx.
277  auto castValueToArgType = [&](Location loc, Value v) {
278  return convertScalarToDtype(
279  b, loc, v,
280  initRegion.getBlocks().front().getArgument(argIdx++).getType(),
281  /*isUnsignedCast=*/false);
282  };
283  for (Value bound : acc::getBounds(op)) {
284  auto dataBound = bound.getDefiningOp<acc::DataBoundsOp>();
285  assert(dataBound &&
286  "acc.reduction's bound must be defined by acc.bounds");
287  // NOTE: we should probably generate get_lowerbound, get_upperbound
288  // and get_stride here, so that we can stop looking for the acc.bounds
289  // operation above, and just use the `bound` value.
290  Value lb =
291  castValueToArgType(dataBound.getLoc(), dataBound.getLowerbound());
292  Value ub =
293  castValueToArgType(dataBound.getLoc(), dataBound.getUpperbound());
294  Value step =
295  castValueToArgType(dataBound.getLoc(), dataBound.getStride());
296  triples.append({lb, ub, step});
297  }
298  assert(triples.size() + 1 == initNumArguments &&
299  "mismatch between number bounds and number of recipe init block "
300  "arguments");
301  }
302 
303  IRMapping mapping;
304  SmallVector<Value> initArgs{origPtr};
305  initArgs.append(triples);
306  mapping.map(initRegion.getBlocks().front().getArguments(), initArgs);
307 
308  Location loc = op.getLoc();
309  setLocation(initRegion, loc);
310 
311  if constexpr (std::is_same_v<OpTy, acc::PrivateOp>) {
312  // Clone the init region for a private.
313  Block *block = &region.front();
314  auto [results, ip] = acc::cloneACCRegionInto(
315  &initRegion, block, block->begin(), mapping, {accPtr});
316  assert(!results.empty() && "expected a result from init region");
317  saveVarName(op.getAccVar(), results[0]);
318  resolveVarNamePlaceholders(block, ip, acc::getVariableName(op.getAccVar()));
319  // Clone the destroy region for a private, if it exists.
320  if (!recipe.getDestroyRegion().empty()) {
321  results.insert(results.begin(), origPtr);
322  results.append(triples);
323  cloneDestroy(loc, recipe, block, std::prev(block->end()), results);
324  }
325  } else if constexpr (std::is_same_v<OpTy, acc::FirstprivateOp>) {
326  // Clone the init region for a firstprivate.
327  Block *block = &region.front();
328  auto [results, ip] = acc::cloneACCRegionInto(
329  &initRegion, block, block->begin(), mapping, {accPtr});
330  assert(!results.empty() && "expected a result from init region");
331  saveVarName(op.getAccVar(), results[0]);
332  resolveVarNamePlaceholders(block, ip, acc::getVariableName(op.getAccVar()));
333  // The copy only consumes the original and user-visible private value.
334  SmallVector<Value> copyArgs{origPtr, results.front()};
335  copyArgs.append(triples);
336  // Destruction also consumes any cleanup values yielded by init.
337  SmallVector<Value> destroyArgs{origPtr};
338  destroyArgs.append(results);
339  destroyArgs.append(triples);
340 
341  // Clone the copy region for a firstprivate
342  mapping.clear();
343  mapping.map(recipe.getCopyRegion().front().getArguments(), copyArgs);
344  // Clone the copy region for a firstprivate.
345  Region &copyRegion = recipe.getCopyRegion();
346  setLocation(copyRegion, loc);
347  acc::cloneACCRegionInto(&copyRegion, block, std::next(ip), mapping, {});
348  if (!recipe.getDestroyRegion().empty()) {
349  cloneDestroy(loc, recipe, block, std::prev(block->end()), destroyArgs);
350  }
351  } else if constexpr (std::is_same_v<OpTy, acc::ReductionOp>) {
352  auto cloneRegionIntoAccRegion = [&](Region *src, Region *dest,
353  bool hasResult) {
354  src->cloneInto(dest, mapping);
355  Block *block = &dest->front();
356  Operation *terminator = block->getTerminator();
357  b.setInsertionPoint(terminator);
358  if (hasResult)
359  acc::YieldOp::create(b, op.getLoc(), terminator->getOperands());
360  else
361  acc::YieldOp::create(b, op.getLoc(), ValueRange{});
362  terminator->erase();
363  };
364 
365  // Clone the init region into acc.reduction_init.
366  if constexpr (std::is_same_v<AccOpTy, acc::ParallelOp>)
367  b.setInsertionPointToStart(&region.front());
368  else if constexpr (std::is_same_v<AccOpTy, acc::LoopOp>)
369  b.setInsertionPoint(op);
370  else
371  llvm_unreachable("unexpected acc op with reduction recipe");
372 
373  SmallVector<Value> reductionBounds(acc::getBounds(op));
374  auto reductionOp =
375  acc::ReductionInitOp::create(b, op.getLoc(), origPtr, reductionBounds,
376  recipe.getReductionOperatorAttr());
377  saveVarName(op.getAccVar(), reductionOp.getResult());
378  cloneRegionIntoAccRegion(&initRegion, &reductionOp.getRegion(),
379  /*hasResult=*/true);
380  Block *initBlock = &reductionOp.getRegion().front();
381  resolveVarNamePlaceholders(initBlock, std::prev(initBlock->end()),
382  acc::getVariableName(op.getAccVar()));
383 
384  // Update the uses within the loop to use the reduction op result.
385  replaceAllUsesInRegionWith(accPtr, reductionOp.getResult(), region);
386 
387  // Clone the combiner region into acc.reduction_combine_region.
388  Region &combinerRegion = recipe.getCombinerRegion();
389  setLocation(combinerRegion, loc);
390 
391  Block *entryBlock = &combinerRegion.front();
392 
393  if constexpr (std::is_same_v<AccOpTy, acc::ParallelOp>)
394  b.setInsertionPoint(region.back().getTerminator());
395  else if constexpr (std::is_same_v<AccOpTy, acc::LoopOp>)
396  b.setInsertionPointAfter(accOp);
397  else
398  llvm_unreachable("unexpected acc op with reduction recipe");
399 
400  // Map the first two block arguments to the original and private
401  // reduction variables. If the recipe's combiner region has the bounds
402  // arguments, we have to map them to the corresponding operands of
403  // acc.reduction operation.
404  mapping.clear();
405  SmallVector<Value, 2> argsRemapping{origPtr, reductionOp.getResult()};
406  argsRemapping.append(triples);
407  mapping.map(entryBlock->getArguments(), argsRemapping);
408 
409  auto combineRegionOp = acc::ReductionCombineRegionOp::create(
410  b, op.getLoc(), origPtr, reductionOp.getResult());
411  cloneRegionIntoAccRegion(&combinerRegion, &combineRegionOp.getRegion(),
412  /*hasResult=*/false);
413 
414  auto *ctx = b.getContext();
415 
416  // For reductions that come from parallel constructs, explicitly set the
417  // GPU parallel dimensions attribute to blockXDim since they will always be
418  // gang private. GPU parallel dimensions cannot be determined for acc.loop
419  // at this point.
420  if constexpr (std::is_same_v<AccOpTy, acc::ParallelOp>) {
421  acc::GPUParallelDimsAttr parDimsAttr;
422  if (accOp.isEffectivelySerial()) {
423  // If acc.serial has been lowered to a parallel op that is effectively
424  // sequential
425  parDimsAttr = acc::getSeqParDimsAttr(ctx, policy);
426  } else {
427  parDimsAttr = acc::getGangDim1ParDimsAttr(ctx, policy);
428  }
429  acc::setParDimsAttr(reductionOp, parDimsAttr);
430  acc::setParDimsAttr(combineRegionOp, parDimsAttr);
431  }
432 
433  // Set sequential parallel dimensions attribute for loops in the recipe.
434  auto setSeqParDimsForRecipeLoops = [&](Region *r) {
435  r->walk([&](LoopLikeOpInterface loopLike) {
436  acc::setParDimsAttr(loopLike, acc::getSeqParDimsAttr(ctx, policy));
437  });
438  };
439  setSeqParDimsForRecipeLoops(&reductionOp.getRegion());
440  setSeqParDimsForRecipeLoops(&combineRegionOp.getRegion());
441 
442  if (!recipe.getDestroyRegion().empty()) {
443  SmallVector<Value> results{origPtr, reductionOp.getResult()};
444  results.append(triples);
445  Block::iterator ip = std::next(Block::iterator(combineRegionOp));
446  cloneDestroy(loc, recipe, combineRegionOp->getBlock(), ip, results);
447  }
448  } else {
449  llvm_unreachable("unexpected op type");
450  }
451 
452  op.erase();
453  return success();
454 }

456 template <typename OpTy>
457 LogicalResult ACCRecipeMaterialization::materializeForACCOp(
458  OpTy accOp, acc::OpenACCSupport &accSupport,
459  acc::ACCToGPUMappingPolicy &policy) const {
460  assert(isa<ACC_COMPUTE_CONSTRUCT_AND_LOOP_OPS>(accOp));
461 
462  if (!accOp.getFirstprivateOperands().empty()) {
463  // Clear the firstprivate operands list so there will be no uses after
464  // the recipe is materialized.
465  SmallVector<Value> operands(accOp.getFirstprivateOperands());
466  accOp.getFirstprivateOperandsMutable().clear();
467  for (Value operand : operands) {
468  auto firstprivateOp = cast<acc::FirstprivateOp>(operand.getDefiningOp());
469  auto symbolRef = cast<SymbolRefAttr>(firstprivateOp.getRecipeAttr());
470  auto decl = SymbolTable::lookupNearestSymbolFrom(accOp, symbolRef);
471  auto recipeOp = cast<acc::FirstprivateRecipeOp>(decl);
472  LLVM_DEBUG(llvm::dbgs() << "materializing: " << firstprivateOp << "\n"
473  << symbolRef << "\n");
474  handleInitialValueMapping(firstprivateOp);
475  if (failed(
476  materialize(firstprivateOp, recipeOp, accOp, accSupport, policy)))
477  return failure();
478  }
479  }
480 
481  if (!accOp.getPrivateOperands().empty()) {
482  // Clear the private operands list so there will be no uses after
483  // the recipe is materialized.
484  SmallVector<Value> operands(accOp.getPrivateOperands());
485  accOp.getPrivateOperandsMutable().clear();
486  for (Value operand : operands) {
487  auto privateOp = cast<acc::PrivateOp>(operand.getDefiningOp());
488  auto symbolRef = cast<SymbolRefAttr>(privateOp.getRecipeAttr());
489  auto decl = SymbolTable::lookupNearestSymbolFrom(accOp, symbolRef);
490  auto recipeOp = cast<acc::PrivateRecipeOp>(decl);
491  LLVM_DEBUG(llvm::dbgs() << "materializing: " << privateOp << "\n"
492  << symbolRef << "\n");
493  if (readsVar(recipeOp.getInitRegion()) ||
494  readsVar(recipeOp.getDestroyRegion()))
495  handleInitialValueMapping(privateOp);
496  if (failed(materialize(privateOp, recipeOp, accOp, accSupport, policy)))
497  return failure();
498  }
499  }
500 
501  if (!accOp.getReductionOperands().empty()) {
502  // Clear the reduction operands list so there will be no uses after
503  // the recipe is materialized.
504  SmallVector<Value> operands(accOp.getReductionOperands());
505  accOp.getReductionOperandsMutable().clear();
506  for (Value operand : operands) {
507  auto reductionOp = cast<acc::ReductionOp>(operand.getDefiningOp());
508  auto symbolRef = cast<SymbolRefAttr>(reductionOp.getRecipeAttr());
509  auto decl = SymbolTable::lookupNearestSymbolFrom(accOp, symbolRef);
510  auto recipeOp = cast<acc::ReductionRecipeOp>(decl);
511  LLVM_DEBUG(llvm::dbgs() << "materializing: " << reductionOp << "\n"
512  << symbolRef << "\n");
513  if (failed(materialize(reductionOp, recipeOp, accOp, accSupport, policy)))
514  return failure();
515  }
516  }
517  return success();
518 }

520 void ACCRecipeMaterialization::runOnOperation() {
521  ModuleOp moduleOp = getOperation();
522  acc::OpenACCSupport &accSupport = getAnalysis<acc::OpenACCSupport>();
523 
524  acc::DefaultACCToGPUMappingPolicy policy;
525 
526  // Materialize all recipes for all compute constructs and loop constructs.
527  bool anyFailed = false;
528  moduleOp.walk([&](Operation *op) {
529  if (anyFailed)
530  return;
531  TypeSwitch<Operation *>(op).Case<ACC_COMPUTE_CONSTRUCT_AND_LOOP_OPS>(
532  [&](auto constructOp) {
533  if (failed(materializeForACCOp(constructOp, accSupport, policy)))
534  anyFailed = true;
535  });
536  });
537  if (anyFailed) {
538  signalPassFailure();
539  return;
540  }
541 
542  // Remove all recipes. Gather the symbol uses that are left with a single
543  // walk: asking whether each recipe is still referenced walks the whole module
544  // again for every recipe, and recipes are generated per type, so there can be
545  // many of them.
546  std::optional<llvm::DenseSet<StringAttr>> usedSymbols;
547  // The module region, not the module op, is the symbol table scope: asking
548  // for the uses on the op itself would not walk into the body.
549  if (std::optional<SymbolTable::UseRange> uses =
550  SymbolTable::getSymbolUses(&moduleOp.getBodyRegion())) {
551  usedSymbols.emplace();
552  for (const SymbolTable::SymbolUse &use : *uses)
553  usedSymbols->insert(use.getSymbolRef().getLeafReference());
554  }
555 
556  moduleOp.walk([&](Operation *op) {
557  if (auto recipe = dyn_cast<acc::ReductionRecipeOp>(op))
558  removeRecipe(recipe, moduleOp, usedSymbols);
559  else if (auto recipe = dyn_cast<acc::PrivateRecipeOp>(op))
560  removeRecipe(recipe, moduleOp, usedSymbols);
561  else if (auto recipe = dyn_cast<acc::FirstprivateRecipeOp>(op))
562  removeRecipe(recipe, moduleOp, usedSymbols);
563  });
564 }
