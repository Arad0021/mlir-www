151 /// Parse an enum from the keyword, or default to the provided default value.
152 /// The return type is the enum type by default, unless overridden with the
153 /// second template argument.
154 template <typename EnumTy, typename RetTy = EnumTy>
155 RetTy parseOptionalLLVMKeyword(OpAsmParser &parser, EnumTy defaultValue) {
156  SmallVector<StringRef, 10> names;
157  for (unsigned i = 0, e = detail::EnumTraits<EnumTy>::getMaxEnumVal(); i <= e;
158  ++i)
159  names.push_back(
160  detail::EnumTraits<EnumTy>::stringify(static_cast<EnumTy>(i)));
161 
162  int index = parseOptionalKeywordAlternative(parser, names);
163  if (index == -1)
164  return static_cast<RetTy>(defaultValue);
165  return static_cast<RetTy>(index);
166 }

168 /// Verifies the operand bundles of `op`.
169 template <typename OpType>
170 LogicalResult verifyOperandBundles(OpType &op) {
171  OperandRangeRange opBundleOperands = op.getOpBundleOperands();
172  std::optional<ArrayAttr> opBundleTags = op.getOpBundleTags();
173 
174  auto isStringAttr = [](Attribute tagAttr) {
175  return isa<StringAttr>(tagAttr);
176  };
177  if (opBundleTags && !llvm::all_of(*opBundleTags, isStringAttr))
178  return op.emitError("operand bundle tag must be a StringAttr");
179 
180  size_t numOpBundles = opBundleOperands.size();
181  size_t numOpBundleTags = opBundleTags ? opBundleTags->size() : 0;
182  if (numOpBundles != numOpBundleTags)
183  return op.emitError("expected ")
184  << numOpBundles << " operand bundle tags, but actually got "
185  << numOpBundleTags;
186 
187  return success();
188 }
