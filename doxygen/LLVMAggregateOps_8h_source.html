23 /// Convert an array of integer attributes to a vector of integers that can be
24 /// used as indices in LLVM operations.
25 template <typename IntT = int64_t>
26 SmallVector<IntT> convertArrayToIndices(ArrayRef<Attribute> attrs) {
27  SmallVector<IntT> indices;
28  indices.reserve(attrs.size());
29  for (Attribute attr : attrs)
30  indices.push_back(cast<IntegerAttr>(attr).getInt());
31  return indices;
32 }
