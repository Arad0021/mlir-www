28 /// Expands CeilDivUIOp (n, m) into
29 /// n == 0 ? 0 : ((n-1) / m) + 1
30 struct CeilDivUIOpConverter : public OpRewritePattern<arith::CeilDivUIOp> {
31  using Base::Base;
32  LogicalResult matchAndRewrite(arith::CeilDivUIOp op,
33  PatternRewriter &rewriter) const final {
34  Location loc = op.getLoc();
35  Value a = op.getLhs();
36  Value b = op.getRhs();
37  Value zero = createConst(loc, a.getType(), 0, rewriter);
38  Value compare =
39  arith::CmpIOp::create(rewriter, loc, arith::CmpIPredicate::eq, a, zero);
40  Value one = createConst(loc, a.getType(), 1, rewriter);
41  Value minusOne = arith::SubIOp::create(rewriter, loc, a, one);
42  Value quotient = arith::DivUIOp::create(rewriter, loc, minusOne, b);
43  Value plusOne = arith::AddIOp::create(rewriter, loc, quotient, one);
44  rewriter.replaceOpWithNewOp<arith::SelectOp>(op, compare, zero, plusOne);
45  return success();
46  }
47 };

49 /// Expands CeilDivSIOp (a, b) into
50 /// z = a / b
51 /// if (z * b != a && (a < 0) == (b < 0)) {
52 /// return z + 1;
53 /// } else {
54 /// return z;
55 /// }
56 struct CeilDivSIOpConverter : public OpRewritePattern<arith::CeilDivSIOp> {
57  using Base::Base;
58  LogicalResult matchAndRewrite(arith::CeilDivSIOp op,
59  PatternRewriter &rewriter) const final {
60  Location loc = op.getLoc();
61  Type type = op.getType();
62  Value a = op.getLhs();
63  Value b = op.getRhs();
64 
65  Value zero = createConst(loc, type, 0, rewriter);
66  Value one = createConst(loc, type, 1, rewriter);
67 
68  Value quotient = arith::DivSIOp::create(rewriter, loc, a, b);
69  Value product = arith::MulIOp::create(rewriter, loc, quotient, b);
70  Value notEqualDivisor = arith::CmpIOp::create(
71  rewriter, loc, arith::CmpIPredicate::ne, a, product);
72 
73  Value aNeg = arith::CmpIOp::create(rewriter, loc, arith::CmpIPredicate::slt,
74  a, zero);
75  Value bNeg = arith::CmpIOp::create(rewriter, loc, arith::CmpIPredicate::slt,
76  b, zero);
77 
78  Value signEqual = arith::CmpIOp::create(
79  rewriter, loc, arith::CmpIPredicate::eq, aNeg, bNeg);
80  Value cond =
81  arith::AndIOp::create(rewriter, loc, notEqualDivisor, signEqual);
82 
83  Value quotientPlusOne = arith::AddIOp::create(rewriter, loc, quotient, one);
84 
85  rewriter.replaceOpWithNewOp<arith::SelectOp>(op, cond, quotientPlusOne,
86  quotient);
87  return success();
88  }
89 };

94 /// return z - 1;
95 /// } else {
96 /// return z;
97 /// }
