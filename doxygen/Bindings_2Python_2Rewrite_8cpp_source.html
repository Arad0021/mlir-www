87 void PyRewritePatternSet::add(nb::handle root,
88  const nb::callable &matchAndRewrite,
89  unsigned benefit) {
90  std::string opName = operationNameFromObject(root);
91  MlirStringRef rootName = mlirStringRefCreate(opName.data(), opName.size());
92 
93  MlirRewritePatternCallbacks callbacks;
94  callbacks.construct = [](void *userData) {
95  nb::handle(static_cast<PyObject *>(userData)).inc_ref();
96  };
97  callbacks.destruct = [](void *userData) {
98  nb::handle(static_cast<PyObject *>(userData)).dec_ref();
99  };
100  callbacks.matchAndRewrite = [](MlirRewritePattern, MlirOperation op,
101  MlirPatternRewriter rewriter,
102  void *userData) -> MlirLogicalResult {
103  nb::handle f(static_cast<PyObject *>(userData));
104 
105  PyMlirContextRef context =
106  PyMlirContext::forContext(mlirOperationGetContext(op));
107  nb::object opView = PyOperation::forOperation(context, op)->createOpView();
108 
109  nb::object res = f(opView, PyPatternRewriter(rewriter));
110  return logicalResultFromObject(res);
111  };
112 
113  MlirRewritePattern pattern = mlirOpRewritePatternCreate(
114  rootName, benefit, mlirRewritePatternSetGetContext(patterns), callbacks,
115  matchAndRewrite.ptr(),
116  /* nGeneratedNames */ 0,
117  /* generatedNames */ nullptr);
118  mlirRewritePatternSetAdd(patterns, pattern);
119 }

222 void PyRewritePatternSet::addConversion(nb::handle root,
223  const nb::callable &matchAndRewrite,
224  PyTypeConverter &typeConverter,
225  unsigned benefit) {
226  std::string opName = operationNameFromObject(root);
227  MlirStringRef rootName = mlirStringRefCreate(opName.data(), opName.size());
228 
229  // Value-initialize so optional callbacks (e.g. matchAndRewrite1ToN) default
230  // to null rather than an indeterminate pointer.
231  MlirConversionPatternCallbacks callbacks{};
232  callbacks.construct = [](void *userData) {
233  nb::handle(static_cast<PyObject *>(userData)).inc_ref();
234  };
235  callbacks.destruct = [](void *userData) {
236  nb::handle(static_cast<PyObject *>(userData)).dec_ref();
237  };
238  callbacks.matchAndRewrite =
239  [](MlirConversionPattern pattern, MlirOperation op, intptr_t nOperands,
240  MlirValue *operands, MlirConversionPatternRewriter rewriter,
241  void *userData) -> MlirLogicalResult {
242  nb::handle f(static_cast<PyObject *>(userData));
243 
244  PyMlirContextRef ctx =
245  PyMlirContext::forContext(mlirOperationGetContext(op));
246  nb::object opView = PyOperation::forOperation(ctx, op)->createOpView();
247 
248  std::vector<MlirValue> operandsVec(operands, operands + nOperands);
249  nb::object adaptorCls =
250  PyGlobals::get()
251  .lookupOpAdaptorClass([&] {
252  MlirStringRef ref = mlirIdentifierStr(mlirOperationGetName(op));
253  return std::string_view(ref.data, ref.length);
254  }())
255  .value_or(nb::borrow(nb::type<PyOpAdaptor>()));
256 
257  nb::object res = f(opView, adaptorCls(operandsVec, opView),
258  PyConversionPattern(pattern).getTypeConverter(),
259  PyConversionPatternRewriter(rewriter));
260  return logicalResultFromObject(res);
261  };
262  MlirConversionPattern pattern = mlirOpConversionPatternCreate(
263  rootName, benefit, mlirRewritePatternSetGetContext(patterns),
264  typeConverter.get(), callbacks, matchAndRewrite.ptr(),
265  /* nGeneratedNames */ 0,
266  /* generatedNames */ nullptr);
267  mlirRewritePatternSetAdd(patterns,
268  mlirConversionPatternAsRewritePattern(pattern));
269 }
