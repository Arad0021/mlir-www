9 // \file
10 // Xe3p uArch definition. Xe3p is a derivative of the third generation of Intel
11 // Xe GPUs and includes Crescent Island (CRI). Note that plain Xe3 exposes no
12 // XeGPU-relevant differences over Xe2 and is therefore covered by
13 // IntelGpuXe2.h; only the Xe3p derivative is modelled here.
14 //
15 // The base instruction set comes from the shared Khronos OpenCL extensions
16 // defined in uArchBase.h; on top of those Xe3p adds the scaled DPAS (MXFP)
17 // extension and relaxes the 2D block memory restrictions. Subclass and override
18 // here only when an Xe3p-specific instruction diverges from the SPIRV defaults.
19 //
20 //===----------------------------------------------------------------------===//

29 struct Xe3p : public uArch {
30  Xe3p(Kind kind, llvm::ArrayRef<const Instruction *> instructionRegistry)
31  : uArch(kind, instructionRegistry) {}
32  int getSubgroupSize() const override { return 16; }
33  unsigned getGeneralPackedFormatBitSize() const override { return 32; }
34 
35  // Restrictions Xe3p places on the 2D memory region accessed by the subgroup
36  // 2D block load / store / prefetch instructions.
37  //
38  // Xe3p relaxes every one of these relative to Xe2, which requires a 64 byte
39  // aligned base address, a base width and pitch of at least 32 bytes, and a
40  // pitch that is a multiple of 16 bytes. On Xe3p the base address, width and
41  // pitch all only need to be a multiple of 4 bytes. There is no minimum size
42  // restriction on the width or the pitch, so their minimum is just that 4 byte
43  // granularity -- a zero-sized surface would be meaningless.
44  const BlockIOMemoryRestrictions &
45  getBlockIOMemoryRestrictions() const override {
46  static constexpr BlockIOMemoryRestrictions restrictions = {
47  /*baseAddressAlignmentBytes=*/4,
48  /*minBaseWidthBytes=*/4,
49  /*baseWidthAlignmentBytes=*/4,
50  /*minBasePitchBytes=*/4,
51  /*basePitchAlignmentBytes=*/4,
52  };
53  return restrictions;
54  }
55 
56  static bool classof(const uArch *u) {
57  return u->getKind() >= Kind::Xe3p_First && u->getKind() <= Kind::Xe3p_Last;
58  }
59 };
