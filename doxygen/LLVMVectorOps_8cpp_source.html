61 LogicalResult ShuffleVectorOp::verify() {

63  llvm::any_of(getMask(), [](int32_t v) { return v != 0; }))
64  return emitOpError("expected a splat operation for scalable vectors");

68 // Folding for shufflevector op when v1 is single element 1D vector
69 // and the mask is a single zero. OpFoldResult will be v1 in this case.
70 OpFoldResult ShuffleVectorOp::fold(FoldAdaptor adaptor) {
71  // Check if operand 0 is a single element vector.
72  auto vecType = llvm::dyn_cast<VectorType>(getV1().getType());
73  if (!vecType || vecType.getRank() != 1 || vecType.getNumElements() != 1)
74  return {};
75  // Check if the mask is a single zero.
76  // Note: The mask is guaranteed to be non-empty.
77  if (getMask().size() != 1 || getMask()[0] != 0)
78  return {};
79  return getV1();
80 }
