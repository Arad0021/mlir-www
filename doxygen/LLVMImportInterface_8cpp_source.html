22 LogicalResult mlir::LLVMImportInterface::convertUnregisteredIntrinsic(
23  OpBuilder &builder, llvm::CallInst *inst,
24  LLVM::ModuleImport &moduleImport) {
25  Location loc = moduleImport.translateLoc(inst->getDebugLoc());
26  SmallVector<llvm::Value *> args(inst->args());
27  SmallVector<Value> mlirOperands;
28  SmallVector<NamedAttribute> mlirAttrs;
29  if (failed(moduleImport.convertIntrinsicArguments(
30  args, llvm::to_vector(inst->operand_bundles()),
31  /*requiresOpBundles=*/true,
32  /*immArgPositions=*/{}, /*immArgAttrNames=*/{}, mlirOperands,
33  mlirAttrs)))
34  return failure();
35 
36  CallIntrinsicOp::Properties properties{};
37  properties.setIntrin(
38  builder.getStringAttr(inst->getCalledFunction()->getName()));
39  properties.setOperandSegmentSizes(
40  {static_cast<int32_t>(args.size()),
41  static_cast<int32_t>(mlirOperands.size() - args.size())});
42  if (failed(CallIntrinsicOp::setPropertiesFromAttr(
43  properties, builder.getDictionaryAttr(mlirAttrs),
44  [&] { return emitError(loc); })))
45  return failure();
46 
47  Type resultType = moduleImport.convertType(inst->getType());
48  auto op = CallIntrinsicOp::create(
49  builder, loc,
50  isa<LLVMVoidType>(resultType) ? TypeRange{} : TypeRange{resultType},
51  mlirOperands, properties, /*discardableAttributes=*/{});
52 
53  moduleImport.setFastmathFlagsAttr(inst, op);
54  moduleImport.convertArgAndResultAttrs(inst, op);
55  op.setStrictfp(inst->getAttributes().hasFnAttr(llvm::Attribute::StrictFP));
56 
57  // Update importer tracking of results.
58  unsigned numRes = op.getNumResults();
59  if (numRes == 1)
60  moduleImport.mapValue(inst) = op.getResult(0);
61  else if (numRes == 0)
62  moduleImport.mapNoResultOp(inst);
63  else
64  return op.emitError(
65  "expected at most one result from target intrinsic call");
66 
67  return success();
68 }

70 /// Converts the LLVM intrinsic to an MLIR operation if a conversion exists.
71 /// Returns failure otherwise.
72 LogicalResult mlir::LLVMImportInterface::convertIntrinsic(
73  OpBuilder &builder, llvm::CallInst *inst,
74  LLVM::ModuleImport &moduleImport) const {
75  // Lookup the dialect interface for the given intrinsic.
76  // Verify the intrinsic identifier maps to an actual intrinsic.
77  llvm::Intrinsic::ID intrinId = inst->getIntrinsicID();
78  assert(intrinId != llvm::Intrinsic::not_intrinsic);
79 
80  // First lookup the intrinsic across different dialects for known
81  // supported conversions, examples include arm-neon, nvm-sve, etc.
82  Dialect *dialect = nullptr;
83 
84  if (!moduleImport.useUnregisteredIntrinsicsOnly())
85  dialect = intrinsicToDialect.lookup(intrinId);
86 
87  // No specialized (supported) intrinsics, attempt to generate a generic
88  // version via llvm.call_intrinsic (if available).
89  if (!dialect)
90  return convertUnregisteredIntrinsic(builder, inst, moduleImport);
91 
92  // Dispatch the conversion to the dialect interface.
93  const LLVMImportDialectInterface *iface = getInterfaceFor(dialect);
94  assert(iface && "expected to find a dialect interface");
95  return iface->convertIntrinsic(builder, inst, moduleImport);
96 }
