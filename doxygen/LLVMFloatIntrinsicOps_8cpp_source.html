26 static LogicalResult verifyHomogeneousStructResult(Operation *op,
27  Type operandType,
28  Type resultType) {
29  auto resultStructType =
30  mlir::dyn_cast<mlir::LLVM::LLVMStructType>(resultType);
31  if (!resultStructType || resultStructType.getBody().size() != 2 ||
32  resultStructType.getBody()[0] != operandType ||
33  resultStructType.getBody()[1] != operandType) {
34  return op->emitOpError(
35  "expected result type to be a homogeneous struct with "
36  "two elements matching the operand type, but got ")
37  << resultType;
38  }
39  return success();
40 }
