8 //
9 // This pass copies globals marked with the `acc.declare` attribute into the
10 // GPU module so that device code (e.g. acc routine, compute regions) can
11 // reference them.
12 //
13 // Overview:
14 // ---------
15 // Globals that have the `acc.declare` attribute (from the OpenACC declare
16 // directive or from the `ACCImplicitDeclare` pass) must be present in the
17 // GPU module for device code to use them. This pass inserts copies of those
18 // globals into the GPU module, creating the module if it does not yet exist.
19 // The host copy of each global remains in the parent module.
20 //
21 // Example:
22 // --------
23 //
24 // Before:
25 // module {
26 // memref.global @arr : memref<7xf32> = dense<0.0>
27 // {acc.declare = #acc.declare<dataClause = acc_create>}
28 // }
29 //
30 // After:
31 // module attributes {gpu.container_module} {
32 // memref.global @arr : memref<7xf32> = dense<0.0>
33 // {acc.declare = #acc.declare<dataClause = acc_create>}
34 // gpu.module @acc_gpu_module {
35 // memref.global @arr : memref<7xf32> = dense<0.0>
36 // {acc.declare = #acc.declare<dataClause = acc_create>}
37 // }
38 // }
39 //
40 // Requirements:
41 // -------------
42 // The pass uses the `acc::OpenACCSupport` for:
43 // - getOrCreateGPUModule: to obtain or create the GPU module.
44 // - emitNYI: to report failure when GPU module creation is not supported.
45 // If no custom implementation is registered, the default implementation is
46 // used (see OpenACCSupport).
47 //
48 //===----------------------------------------------------------------------===//

72 static bool hasAccDeclareGlobals(ModuleOp mod) {
73  for (Operation &op : mod.getBody()->getOperations())
74  if (op.getDiscardableAttr(acc::getDeclareAttrName()))
75  return true;
76  return false;
77 }

79 static void makeDeviceGlobalDeclaration(Operation &globalOp) {
80  globalOp.setInherentAttr(StringAttr::get(globalOp.getContext(), "initVal"),
81  {});
82  globalOp.setInherentAttr(StringAttr::get(globalOp.getContext(), "linkage"),
83  {});
84  for (Region &region : globalOp.getRegions()) {
85  region.dropAllReferences();
86  region.getBlocks().clear();
87  }
88 }

90 class ACCDeclareGPUModuleInsertion
91  : public acc::impl::ACCDeclareGPUModuleInsertionBase<
92  ACCDeclareGPUModuleInsertion> {
93 public:
94  using acc::impl::ACCDeclareGPUModuleInsertionBase<
95  ACCDeclareGPUModuleInsertion>::ACCDeclareGPUModuleInsertionBase;
96 
97  LogicalResult copyGlobalsToGPUModule(gpu::GPUModuleOp gpuMod, ModuleOp mod,
98  acc::OpenACCSupport &accSupport) const {
99  SymbolTable gpuSymTable(gpuMod);
100 
101  for (Operation &globalOp : mod.getBody()->getOperations()) {
102  if (!globalOp.getDiscardableAttr(acc::getDeclareAttrName()))
103  continue;
104 
105  auto symOp = dyn_cast<SymbolOpInterface>(&globalOp);
106  if (!symOp)
107  continue;
108 
109  StringAttr name = symOp.getNameAttr();
110  Operation *deviceGlobal = globalOp.clone();
111  auto declareAttr = globalOp.getDiscardableAttrOfType<acc::DeclareAttr>(
112  acc::getDeclareAttrName());
113  auto globalVar = dyn_cast<acc::GlobalVariableOpInterface>(&globalOp);
114  bool makeUnifiedDeclaration =
115  cudaUnified &&
116  declareAttr.getDataClause().getValue() !=
117  acc::DataClause::acc_declare_device_resident &&
118  (!globalVar || !globalVar.isConstant());
119  if (makeUnifiedDeclaration)
120  makeDeviceGlobalDeclaration(*deviceGlobal);
121 
122  if (Operation *existing = gpuSymTable.lookup(name.getValue())) {
123  // Reuse when structurally equivalent ignoring locations and discardable
124  // attrs such as `acc.declare` attributes. Only a different op type or a
125  // true definition mismatch is a conflict.
126  auto isEquivalent = [](Operation *lhs, Operation *rhs) {
127  return lhs->getName() == rhs->getName() &&
128  OperationEquivalence::isEquivalentTo(
129  lhs, rhs, OperationEquivalence::ignoreValueEquivalence,
130  /*markEquivalent=*/nullptr,
131  OperationEquivalence::IgnoreLocations |
132  OperationEquivalence::IgnoreDiscardableAttrs);
133  };
134  if (!isEquivalent(existing, deviceGlobal)) {
135  // Earlier GPU lowering can create a global in the GPU module before
136  // this pass. In unified memory, convert an equivalent pre-existing
137  // global to the declaration form expected for an OpenACC global.
138  if (makeUnifiedDeclaration) {
139  Operation *normalizedExisting = existing->clone();
140  makeDeviceGlobalDeclaration(*normalizedExisting);
141  bool canReuse = isEquivalent(normalizedExisting, deviceGlobal);
142  normalizedExisting->destroy();
143  if (canReuse)
144  makeDeviceGlobalDeclaration(*existing);
145  }
146  }
147  if (!isEquivalent(existing, deviceGlobal)) {
148  deviceGlobal->destroy();
149  accSupport.emitNYI(globalOp.getLoc(),
150  llvm::Twine("duplicate global symbol '") +
151  name.getValue() + "' in gpu module");
152  return failure();
153  }
154  // Propagate acc.declare onto the GPU copy if it was cloned before the
155  // host global was marked.
156  if (!existing->getDiscardableAttr(acc::getDeclareAttrName()))
157  if (Attribute declareAttr =
158  globalOp.getDiscardableAttr(acc::getDeclareAttrName()))
159  existing->setDiscardableAttr(acc::getDeclareAttrName(),
160  declareAttr);
161  deviceGlobal->destroy();
162  continue;
163  }
164 
165  gpuSymTable.insert(deviceGlobal);
166  }
167  return success();
168  }
169 
170  void runOnOperation() override {
171  ModuleOp mod = getOperation();
172 
173  // Check for any candidates first - do this to avoid creating the GPU module
174  // if there are no candidates.
175  if (!hasAccDeclareGlobals(mod))
176  return;
177 
178  acc::OpenACCSupport &accSupport = getAnalysis<acc::OpenACCSupport>();
179  std::optional<gpu::GPUModuleOp> gpuMod =
180  accSupport.getOrCreateGPUModule(mod);
181  if (!gpuMod) {
182  accSupport.emitNYI(mod.getLoc(), "Failed to create GPU module");
183  return;
184  }
185 
186  if (failed(copyGlobalsToGPUModule(*gpuMod, mod, accSupport)))
187  return;
188  }
189 };
