39 /// Return true if the op's own memory effects do not prevent erasure. Allowed
40 /// effects are reads and effects on results allocated by the op itself.
41 ///
42 /// If the op only has recursive effects, treat the op itself as having no
43 /// effects. Without a memory effect interface or recursive effects, return
44 /// false to conservatively assume live effects.
45 ///
46 /// The caller must also check nested effects when `hasRecursiveEffects` is
47 /// true. This function does not check uses or restrictions on erasing
48 /// terminators and symbols.
49 static bool hasOnlyRemovableEffects(Operation *op, bool hasRecursiveEffects) {
50  if (auto effectInterface = dyn_cast<MemoryEffectOpInterface>(op)) {
51  SmallVector<MemoryEffects::EffectInstance, 1> effects;
52  effectInterface.getEffects(effects);
53 
54  // Gather all results of this op that are allocated.
55  SmallPtrSet<Value, 4> allocResults;
56  for (const MemoryEffects::EffectInstance &effect : effects)
57  if (isa<MemoryEffects::Allocate>(effect.getEffect()) &&
58  effect.getValue() && effect.getValue().getDefiningOp() == op)
59  allocResults.insert(effect.getValue());
60  return llvm::all_of(effects,
61  [&](const MemoryEffects::EffectInstance &effect) {
62  // We can drop effects on values allocated by this
63  // operation and returned as results. Otherwise, the
64  // effect must be a read.
65  return allocResults.contains(effect.getValue()) ||
66  isa<MemoryEffects::Read>(effect.getEffect());
67  });
68  }
69  return hasRecursiveEffects;
70 }

72 /// Internal implementation of `mlir::wouldOpBeTriviallyDead` that also
73 /// considers terminator operations as dead if they have no side effects. This
74 /// allows for marking region operations as trivially dead without always being
75 /// conservative of terminators.
76 static bool wouldOpBeTriviallyDeadImpl(
77  Operation *root,
78  DenseMap<Operation *, DeadnessCache::State> *cache = nullptr) {
79  using State = DeadnessCache::State;
80  // Repeated queries on nested operations would otherwise scan the same
81  // subtrees. Cache each subtree after checking its effects, and skip cached
82  // subtrees on later queries. An explicit worklist also bounds C++ stack use
83  // for deeply nested IR. Uncached queries do not look up or record summaries.
84  struct WorkItem {
85  // Keep the unvisited range so a block can resume after a nested operation
86  // without adding every sibling to the worklist.
87  Block::iterator current;
88  Block::iterator end;
89  Operation *complete = nullptr;
90  };
91  SmallVector<WorkItem, 1> worklist = {
92  {Block::iterator(root), ++Block::iterator(root)}};
93  while (!worklist.empty()) {
94  WorkItem &item = worklist.back();
95  if (item.current == item.end) {
96  if (cache && item.complete)
97  (*cache)[item.complete] = State::Dead;
98  worklist.pop_back();
99  continue;
100  }
101  Operation *op = &*item.current++;
102  bool cachedAlive = false;
103  if (cache) {
104  auto cached = cache->find(op);
105  if (cached != cache->end()) {
106  if (cached->second == State::Dead)
107  continue;
108  cachedAlive = true;
109  }
110  }
111  bool recursive = op->hasTrait<OpTrait::HasRecursiveMemoryEffects>();
112  if (cachedAlive || !hasOnlyRemovableEffects(op, recursive)) {
113  // One live effect keeps every ancestor up to the root live. The
114  // remaining subtrees cannot change the answer.
115  if (cache) {
116  for (Operation *ancestor = op;; ancestor = ancestor->getParentOp()) {
117  (*cache)[ancestor] = State::Alive;
118  if (ancestor == root)
119  break;
120  }
121  }
122  return false;
123  }
124  if (!recursive) {
125  if (cache)
126  (*cache)[op] = State::Dead;
127  continue;
128  }
129  // The first pushed block is processed last. Record the parent's summary
130  // only when that block is complete and all other blocks have been checked.
131  Operation *complete = cache ? op : nullptr;
132  for (Region &region : op->getRegions()) {
133  for (Block &block : region) {
134  worklist.push_back({block.begin(), block.end(), complete});
135  complete = nullptr;
136  }
137  }
138  if (complete)
139  (*cache)[complete] = State::Dead;
140  }
141  return true;
142 }

144 bool DeadnessCache::isOpTriviallyDead(Operation *root) {
145  // Cache only effects. Uses can change without a change to the defining op.
146  // Nested operations may have internal uses or be terminators even when the
147  // whole root can be erased, so these checks apply only to the root.
148  if (!root->use_empty() || root->mightHaveTrait<OpTrait::IsTerminator>() ||
149  isa<SymbolOpInterface>(root))
150  return false;
151 
152  auto cachedRoot = cache.find(root);
153  if (cachedRoot != cache.end())
154  return cachedRoot->second == State::Dead;
155 
156  return wouldOpBeTriviallyDeadImpl(root, &cache);
157 }

164 void DeadnessCache::invalidateAncestors(Operation *op) {
165  // An ancestor may report nested effects through its own effect interface,
166  // without querying the intermediate cache entries. A missing entry cannot
167  // stop invalidation of its ancestors.
168  for (Operation *parent = op; parent; parent = parent->getParentOp())
169  cache.erase(parent);
170 }

380 bool mlir::isMemoryEffectFree(Operation *op) {
381  if (auto memInterface = dyn_cast<MemoryEffectOpInterface>(op)) {
382  if (!memInterface.hasNoEffect())
383  return false;
384  // If the op does not have recursive side effects, then it is memory effect
385  // free.
386  if (!op->hasTrait<OpTrait::HasRecursiveMemoryEffects>())
387  return true;
388  } else if (!op->hasTrait<OpTrait::HasRecursiveMemoryEffects>()) {
389  // Otherwise, if the op does not implement the memory effect interface and
390  // it does not have recursive side effects, then it cannot be known that the
391  // op is moveable.
392  return false;
393  }
394 
395  // Recurse into the regions and ensure that all nested ops are memory effect
396  // free.
397  for (Region &region : op->getRegions())
398  for (Operation &op : region.getOps())
399  if (!isMemoryEffectFree(&op))
400  return false;
401  return true;
402 }

404 // the returned vector may contain duplicate effects
405 std::optional<llvm::SmallVector<MemoryEffects::EffectInstance>>
406 mlir::getEffectsRecursively(Operation *rootOp) {
407  SmallVector<MemoryEffects::EffectInstance> effects;
408  SmallVector<Operation *> effectingOps(1, rootOp);
409  while (!effectingOps.empty()) {
410  Operation *op = effectingOps.pop_back_val();
411 
412  // If the operation has recursive effects, push all of the nested
413  // operations on to the stack to consider.
414  bool hasRecursiveEffects =
415  op->hasTrait<OpTrait::HasRecursiveMemoryEffects>();
416  if (hasRecursiveEffects) {
417  for (Region &region : op->getRegions()) {
418  for (Block &block : region) {
419  for (Operation &nestedOp : block) {
420  effectingOps.push_back(&nestedOp);
421  }
422  }
423  }
424  }
425 
426  if (auto effectInterface = dyn_cast<MemoryEffectOpInterface>(op)) {
427  effectInterface.getEffects(effects);
428  } else if (!hasRecursiveEffects) {
429  // the operation does not have recursive memory effects or implement
430  // the memory effect op interface. Its effects are unknown.
431  return std::nullopt;
432  }
433  }
434  return effects;
435 }

437 bool mlir::isSpeculatable(Operation *op) {
438  auto conditionallySpeculatable = dyn_cast<ConditionallySpeculatable>(op);
439  if (!conditionallySpeculatable)
440  return false;
441 
442  switch (conditionallySpeculatable.getSpeculatability()) {
443  case Speculation::RecursivelySpeculatable:
444  for (Region &region : op->getRegions()) {
445  for (Operation &op : region.getOps())
446  if (!isSpeculatable(&op))
447  return false;
448  }
449  return true;
450 
451  case Speculation::Speculatable:
452  return true;
453 
454  case Speculation::NotSpeculatable:
