522 TosaProfileCompliance::getOperatorMatchedEntries(Operation *op) {
523  const std::string opName = op->getName().getStringRef().str();
524  const auto complianceMap = getProfileComplianceMap<T>();
525  const auto it = complianceMap.find(opName);

527  return {};

532 template <typename T>
533 LogicalResult TosaProfileCompliance::checkProfileOrExtension(
534  Operation *op, const tosa::TargetEnv &targetEnv) {
535  const auto maybeOpEntries = getOperatorMatchedEntries<T>(op);
536  if (failed(maybeOpEntries))
537  return success();
538 
539  const auto opEntries = maybeOpEntries.value();
540  if (opEntries.size() == 0) {
541  // No matched restriction found.
542  return success();
543  }
544 
545  // Check the profile/extension requirement according to the current target
546  // profiles/extensions.
547  const auto isModeAllowed = [&](const OpComplianceInfo<T> &info) -> bool {
548  if (info.condition == CheckCondition::allOf)
549  return targetEnv.allowsAllOf(info.mode);
550  return targetEnv.allowsAnyOf(info.mode);
551  };
552 
553  // Check the matched op compliance version does not exceed the target
554  // specification version.
555  const TosaSpecificationVersion targetVersion{targetEnv.getSpecVersion()};
556  const auto isVersionCompatible =
557  [&targetVersion](const OpComplianceInfo<T> &info) -> bool {
558  // Assume compatible if there are no versioned type constraints
559  if (info.operandTypeInfoSet.empty())
560  return true;
561  const TosaSpecificationVersion complianceVersion{
562  info.operandTypeInfoSet.front().second};
563  return targetVersion.isBackwardsCompatibleWith(complianceVersion);
564  };
565 
566  for (const auto &info : opEntries) {
567  // Type constraints exist, check if they are compatible with the current
568  // target definition, including: profile, extension and specification
569  // version
570  if (isModeAllowed(info) && isVersionCompatible(info))
571  return success();
572  }
573 
574  // No valid entry was found, now emit appropriate error message and return
575  // failure
576  std::string message;

582  bool mismatchedVersion = false;

584  mismatchedVersion = true;

586  << stringifyVersion(info.operandTypeInfoSet.front().second) << " (got "
587  << stringifyVersion(targetVersion) << ") ";
588  }
589

596  << "] profiles/extensions ";

619 LogicalResult TosaProfileCompliance::checkInvalid(Operation *op) {
620  const auto maybeProfEntries = getOperatorMatchedEntries<Profile>(op);
621  const auto maybeExtEntries = getOperatorMatchedEntries<Extension>(op);
622  if (failed(maybeProfEntries) && failed(maybeExtEntries))
623  return success();
624 
625  const bool hasEntry =
626  (succeeded(maybeProfEntries) && !maybeProfEntries.value().empty()) ||
627  (succeeded(maybeExtEntries) && !maybeExtEntries.value().empty());
628 
629  if (!hasEntry) {
630  std::string message;
631  llvm::raw_string_ostream os(message);
632  os << "illegal: operation operand/result data types did not align with any "
633  "profile or extension, got (";
634 
635  ProfileInfoDepot depot(op);
636  SmallVector<TypeInfo> current = depot.getInfo();
637  for (const auto &typeInfo : llvm::drop_end(current))
638  os << stringifyTypeInfo(typeInfo) << ",";
639  os << stringifyTypeInfo(current.back()) << ")";
640 
641  // avoid polluting the error message output by outputting only
642  // the best match
643  const std::string opName = op->getName().getStringRef().str();
644  int maxMatches = -1;
645  SmallVector<TypeInfo> bestTypeInfo;
646  const auto searchBestMatch = [&](auto map) {
647  for (const auto &complianceInfos : map[opName]) {
648  for (const auto &versionedTypeInfos :
649  complianceInfos.operandTypeInfoSet) {
650  const SmallVector<TypeInfo> typeInfos = versionedTypeInfos.first;
651  if (current.size() != typeInfos.size())
652  continue;
653  const int matches = llvm::count_if(
654  llvm::zip_equal(current, typeInfos), [&](const auto zipType) {
655  return isSameTypeInfo(std::get<0>(zipType),
656  std::get<1>(zipType));
657  });
658  if (matches > maxMatches) {
659  maxMatches = matches;
660  bestTypeInfo = typeInfos;
661  }
662  }
663  }
664  };
665  searchBestMatch(getProfileComplianceMap<Profile>());
666  searchBestMatch(getProfileComplianceMap<Extension>());
667 
668  os << ", did you mean (";
669  for (const auto &typeInfo : llvm::drop_end(bestTypeInfo))
670  os << stringifyTypeInfo(typeInfo) << ",";
671  os << stringifyTypeInfo(bestTypeInfo.back()) << ")? ";
672  os << "Otherwise, please refer to the 'supported data types' for '"
673  << opName << "' in the specification.";
674  op->emitOpError(message);
675  return failure();
676  }
677 
678  return success();
679 }

681 // Find the profiles or extensions requirement according to the signature of
682 // type of the operand list.
683 template <typename T>
684 SmallVector<OpComplianceInfo<T>> TosaProfileCompliance::findMatchedEntries(
685  Operation *op, SmallVector<OpComplianceInfo<T>> compInfo) {
686  assert(compInfo.size() != 0 &&
687  "profile-based compliance information is empty");
688 
689  // If there are no type constraints for the given operation, return the
690  // matched entries directly so required profile/extension information can
691  // be checked
692  SmallVector<OpComplianceInfo<T>> matchedInfos;
693  if (llvm::all_of(compInfo, [&matchedInfos](const OpComplianceInfo<T> &info) {
694  if (info.operandTypeInfoSet.empty()) {
695  matchedInfos.push_back(info);
696  return true;
697  }
698  return false;
699  }))
700  return matchedInfos;
701 
702  // Get the type information of the operator we're trying to match
703  ProfileInfoDepot depot(op);
704  SmallVector<TypeInfo> present = depot.getInfo();
705  if (present.size() == 0)
706  return {};
707 
708  // Compare 'present' against known type signatures in the compliance metadata
709  for (size_t i = 0; i < compInfo.size(); i++) {
710  SmallVector<VersionedTypeInfo> sets = compInfo[i].operandTypeInfoSet;
711  for (const auto &set : sets) {
712  SmallVector<TypeInfo> expected = set.first;
713  // Tensor-list operators can legitimately have multiple valid signatures
714  // with different operand/result counts, e.g. data-only and data+scale
715  // forms. Treat those as non-matches instead of asserting.
716  if (present.size() != expected.size())
717  continue;
718 
719  bool isFound = true;
720  // Compare the type signature between the given operation and the
721  // compliance metadata.
722  for (size_t j = 0; j < expected.size(); j++) {
723  if (!isSameTypeInfo(present[j], expected[j])) {
724  // Verify the next mode set from the list.
725  isFound = false;
726  break;
727  }
728  }
729 
730  if (isFound == true) {
731  SmallVector<VersionedTypeInfo> typeInfoSet{set};
732  OpComplianceInfo<T> info{compInfo[i].mode, typeInfoSet,
733  compInfo[i].condition};
734  matchedInfos.push_back(info);
735  }
736  }
737  }
738 
739  return matchedInfos;
740 }
