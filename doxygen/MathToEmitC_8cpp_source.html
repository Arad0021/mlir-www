53 template <typename OpType>
54 LogicalResult LowerToEmitCCallOpaque<OpType>::matchAndRewrite(
55  OpType op, PatternRewriter &rewriter) const {
56  if (!llvm::all_of(op->getOperandTypes(),
57  llvm::IsaPred<Float32Type, Float64Type>) ||
58  !llvm::all_of(op->getResultTypes(), [](Type type) {
59  return isa<Float32Type, Float64Type>(type) || type.isInteger(1);
60  }))
61  return rewriter.notifyMatchFailure(
62  op.getLoc(), "expected all operands to be of type f32 or f64, and "
63  "all results to be of type f32, f64 or i1");
64  if (auto roundingModeOp =
65  dyn_cast<arith::ArithRoundingModeInterface>(op.getOperation());
66  roundingModeOp && roundingModeOp.getRoundingModeAttr())
67  return rewriter.notifyMatchFailure(op.getLoc(),
68  "static rounding mode not supported");
69  std::string modifiedCalleeStr = calleeStr;
70  if (languageTarget == emitc::LanguageTarget::cpp11) {
71  modifiedCalleeStr = "std::" + calleeStr;
72  } else if (languageTarget == emitc::LanguageTarget::c99 && typeSuffix) {
73  auto operandType = op->getOperandTypes()[0];
74  if (operandType.isF32())
75  modifiedCalleeStr = calleeStr + "f";
76  }
77  rewriter.replaceOpWithNewOp<emitc::CallOpaqueOp>(
78  op, op.getType(), modifiedCalleeStr, op->getOperands());
79  return success();
80 }
