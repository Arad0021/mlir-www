9 // This file implements Mem2Reg-related interfaces that let a statically-shaped
10 // memref be promoted into a single SSA value, provided every access to
11 // the buffer is a whole-buffer read or write (or a whole-sub-region access of
12 // such a buffer via a subview). With these models, Mem2Reg replaces the memory
13 // slot with a vector value (*), threading it as the reaching definition:
14 //
15 // * a static, same-rank `memref.subview` is exposed as a promotable sub-slice
16 // alias of the buffer's slot (via `PromotableAliaserInterface`): a read of
17 // the subview projects out of the vector value with
18 // `vector.extract_strided_slice`, and a write into it composes back into
19 // the value with `vector.insert_strided_slice`. This lets a buffer that is
20 // only ever accessed through static subviews promote as well, with partial
21 // and overlapping sub-writes composing in program order.
22 //
23 // Accesses that are not whole-(sub-)buffer -- dynamic offsets, rank-reducing or
24 // non-unit-stride subviews -- are left untouched, so the buffer is not
25 // promoted.
26 //
27 // (*) Addging support for scalar values is left as a TODO.
28 //===----------------------------------------------------------------------===//

45 /// Returns the offsets of `subView` as a static, contiguous, same-rank slice of
46 /// its source, or nullopt if the subview is not promotable as a whole-buffer
47 /// sub-slice. Promotion projects the parent buffer's vector value through
48 /// `vector.extract_strided_slice` / `insert_strided_slice`, which require:
49 /// * fully static offsets and sizes,
50 /// * unit strides,
51 /// * no rank reduction (result rank == source rank),
52 /// so a dropped or dynamic dimension disqualifies the subview.
53 static std::optional<SmallVector<int64_t>>
54 getPromotableSubViewOffsets(memref::SubViewOp subView) {
55  auto srcType = dyn_cast<MemRefType>(subView.getSource().getType());
56  auto resType = dyn_cast<MemRefType>(subView.getResult().getType());
57  if (!srcType || !resType || !srcType.hasStaticShape() ||
58  !resType.hasStaticShape())
59  return std::nullopt;
60 
61  // No rank reduction: extract/insert_strided_slice operate at a single rank.
62  if (srcType.getRank() != resType.getRank())
63  return std::nullopt;
64 
65  // Unit strides only.
66  for (OpFoldResult stride : subView.getMixedStrides()) {
67  std::optional<int64_t> s = getConstantIntValue(stride);
68  if (!s || *s != 1)
69  return std::nullopt;
70  }
71 
72  // Static offsets.
73  SmallVector<int64_t> offsets;
74  for (OpFoldResult offset : subView.getMixedOffsets()) {
75  std::optional<int64_t> o = getConstantIntValue(offset);
76  if (!o)
77  return std::nullopt;
78  offsets.push_back(*o);
79  }
80 
81  // Static sizes (already implied by the result's static shape, but the sizes
82  // must match the result shape so the slice covers exactly the subview).
83  for (auto [size, dim] :
84  llvm::zip_equal(subView.getMixedSizes(), resType.getShape())) {
85  std::optional<int64_t> s = getConstantIntValue(size);
86  if (!s || *s != dim)
87  return std::nullopt;
88  }
89  return offsets;
90 }

92 /// Exposes a static, same-rank `memref.subview` as a sub-slice alias of a
93 /// whole-buffer vector slot. Reads of the subview become
94 /// `vector.extract_strided_slice` of the parent value; writes become
95 /// `vector.insert_strided_slice` into the current reaching definition.
96 struct SubViewOpAliasModel
97  : public PromotableAliaserInterface::ExternalModel<SubViewOpAliasModel,
98  memref::SubViewOp> {
99  void getPromotableSlotAliases(Operation *op,
100 
101  OpOperand &aliasedSlotPointerOperand,
102  const MemorySlot &parentSlot,
103  SmallVectorImpl<MemorySlot> &newSlots) const {
104  auto subView = cast<memref::SubViewOp>(op);
105  if (aliasedSlotPointerOperand.get() != subView.getSource())
106  return;
107 
108  // The parent slot must promote to a vector (whole-buffer promotion). A
109  // scalar (single-element) parent slot cannot be sliced.
110  auto parentVecType = dyn_cast<VectorType>(parentSlot.valueType);
111  if (!parentVecType)
112  return;
113 
114  if (!getPromotableSubViewOffsets(subView))
115  return;
116 
117  // The alias's value type is the sub-vector matching the subview's shape.
118  auto resType = cast<MemRefType>(subView.getResult().getType());
119  if (!VectorType::isValidElementType(resType.getElementType()))
120  return;
121  VectorType aliasVecType =
122  VectorType::get(resType.getShape(), resType.getElementType());
123  newSlots.push_back(MemorySlot{subView.getResult(), aliasVecType});
124  }

126  Value projectSlotValueToAliasValue(Operation *op,
127  OpOperand & /*aliasedSlotPointerOperand*/,
128  const MemorySlot & /*parentSlot*/,
129  const MemorySlot &aliasSlot,
130  Value slotValue,
131  OpBuilder &builder) const {
132  auto subView = cast<memref::SubViewOp>(op);
133  SmallVector<int64_t> offsets = *getPromotableSubViewOffsets(subView);
134  auto aliasVecType = cast<VectorType>(aliasSlot.valueType);
135  SmallVector<int64_t> strides(offsets.size(), 1);
136  return vector::ExtractStridedSliceOp::create(
137  builder, op->getLoc(), slotValue, offsets,
138  aliasVecType.getShape(), strides)
139  .getResult();
140  }

142  Value projectAliasValueToSlotValue(Operation *op,
143  OpOperand & /*aliasedSlotPointerOperand*/,
144  const MemorySlot & /*parentSlot*/,
145  const MemorySlot & /*aliasSlot*/,
146  Value aliasValue, Value reachingDef,
147  OpBuilder &builder) const {
148  auto subView = cast<memref::SubViewOp>(op);
149  SmallVector<int64_t> offsets = *getPromotableSubViewOffsets(subView);
150  SmallVector<int64_t> strides(offsets.size(), 1);
151  return vector::InsertStridedSliceOp::create(
152  builder, op->getLoc(), aliasValue, reachingDef, offsets, strides)
153  .getResult();
154  }

157 /// Companion `PromotableOpInterface` model: once the slot is promoted, the
158 /// subview has no remaining memory uses and is erased.
159 struct SubViewOpPromotableModel
160  : public PromotableOpInterface::ExternalModel<SubViewOpPromotableModel,
161  memref::SubViewOp> {
162  bool canUsesBeRemoved(Operation *op,
163  const SmallPtrSetImpl<OpOperand *> &blockingUses,
164  SmallVectorImpl<OpOperand *> &newBlockingUses,
165  const DataLayout &dataLayout) const {
166  // The subview result is itself a blocking use of the parent slot; its own
167  // users (the transfers) are resolved through the alias projections.
168  for (OpOperand &use : op->getResult(0).getUses())
169  newBlockingUses.push_back(&use);
170  return true;
171  }
