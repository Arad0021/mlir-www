82 /// Enumeration used to encode the execution mapping on a loop construct.
83 /// They refer directly to the OpenACC 3.3 standard:
84 /// 2.9.2. gang
85 /// 2.9.3. worker
86 /// 2.9.4. vector
87 ///
88 /// Value can be combined bitwise to reflect the mapping applied to the
89 /// construct. e.g. `acc.loop gang vector`, the `gang` and `vector` could be
90 /// combined and the final mapping value would be 5 (4 | 1).
91 enum OpenACCExecMapping { NONE = 0, VECTOR = 1, WORKER = 2, GANG = 4 };

242 template <typename ComputeOpT>
243 static bool isGangWorkerVectorAllOne(ComputeOpT op) {
244  // Strip index_cast operations from a value before checking for a constant.
245  auto stripIndexCasts = [](Value val) -> Value {
246  while (auto castOp = val.getDefiningOp<arith::IndexCastOp>())
247  val = castOp.getIn();
248  return val;
249  };
250 
251  auto numGangs = op.getNumGangsValues();
252  if (numGangs.empty())
253  return false;
254  for (Value gangSize : numGangs) {
255  if (!isConstantIntValue(stripIndexCasts(gangSize), 1))
256  return false;
257  }
258  Value numWorkers = op.getNumWorkersValue();
259  if (!numWorkers)
260  return false;
261  Value vectorLength = op.getVectorLengthValue();
262  if (!vectorLength)
263  return false;
264  return isConstantIntValue(stripIndexCasts(numWorkers), 1) &&
265  isConstantIntValue(stripIndexCasts(vectorLength), 1);
266 }
