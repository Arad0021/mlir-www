41 template <typename PredicateAttr, typename Predicate>
42 static ParseResult parseCmpPredicateImpl(
43  OpAsmParser &parser, PredicateAttr &predicate,
44  function_ref<std::optional<Predicate>(StringRef)> symbolize) {
45  std::string spelling;
46  SMLoc loc = parser.getCurrentLocation();
47  if (parser.parseString(&spelling))
48  return failure();
49  std::optional<Predicate> value = symbolize(spelling);
50  if (!value)
51  return parser.emitError(loc)
52  << "'" << spelling
53  << "' is an incorrect value of the 'predicate' attribute";
54  predicate = PredicateAttr::get(parser.getContext(), *value);
55  return success();
56 }

91 OpFoldResult ICmpOp::fold(FoldAdaptor adaptor) {
92  if (getPredicate() != ICmpPredicate::eq &&
93  getPredicate() != ICmpPredicate::ne)
94  return {};
95 
96  // cmpi(eq/ne, x, x) -> true/false
97  if (getLhs() == getRhs())
98  return getBoolAttribute(getType(), getContext(),
99  getPredicate() == ICmpPredicate::eq);
100 
101  // cmpi(eq/ne, alloca, null) -> false/true
102  if (getLhs().getDefiningOp<AllocaOp>() && getRhs().getDefiningOp<ZeroOp>())
103  return getBoolAttribute(getType(), getContext(),
104  getPredicate() == ICmpPredicate::ne);
105 
106  // cmpi(eq/ne, null, alloca) -> cmpi(eq/ne, alloca, null)
107  if (getLhs().getDefiningOp<ZeroOp>() && getRhs().getDefiningOp<AllocaOp>()) {
108  Value lhs = getLhs();
109  Value rhs = getRhs();
110  getLhsMutable().assign(rhs);
111  getRhsMutable().assign(lhs);
112  return getResult();
113  }
114 
115  return {};
116 }

122 OpFoldResult LLVM::ShlOp::fold(FoldAdaptor adaptor) {
123  auto rhs = dyn_cast_or_null<IntegerAttr>(adaptor.getRhs());
124  if (!rhs)
125  return {};
126 
127  if (rhs.getValue().uge(getLhs().getType().getIntOrFloatBitWidth()))
128  return {}; // TODO: Fold into poison.
129 
130  auto lhs = dyn_cast_or_null<IntegerAttr>(adaptor.getLhs());
131  if (!lhs)
132  return {};
133 
134  return IntegerAttr::get(getType(), lhs.getValue().shl(rhs.getValue()));
135 }

172  // INT_MIN / -1 => UB
