51 /// A helper struct that performs various "expensive checks" to detect broken
52 /// rewrite patterns use the rewriter API incorrectly. A rewrite pattern is
53 /// broken if:
54 /// * IR does not verify after pattern application / folding.
55 /// * Pattern returns "failure" but the IR has changed.
56 /// * Pattern returns "success" but the IR has not changed.
57 ///
58 /// This struct stores finger prints of ops to determine whether the IR has
59 /// changed or not.
60 struct ExpensiveChecks : public RewriterBase::ForwardingListener {
61  ExpensiveChecks(RewriterBase::Listener *listener, Operation *topLevel)
62  : RewriterBase::ForwardingListener(listener), topLevel(topLevel) {}

64  /// Compute finger prints of the given op and its nested ops.
65  void computeFingerPrints(Operation *topLevel) {
66  this->topLevel = topLevel;
67  this->topLevelFingerPrint.emplace(topLevel);
68  topLevel->walk([&](Operation *op) {
69  fingerprints.try_emplace(op, op, /*includeNested=*/false);
70  });
71  }
72 
73  /// Clear all finger prints.
74  void clear() {
75  topLevel = nullptr;
76  topLevelFingerPrint.reset();
77  fingerprints.clear();
78  }

80  void notifyRewriteSuccess() {
81  if (!topLevel)
82  return;
83 
84  // Make sure that the IR still verifies.
85  if (failed(verify(topLevel)))
86  llvm::report_fatal_error("IR failed to verify after pattern application");
87 
88  // Pattern application success => IR must have changed.
89  OperationFingerPrint afterFingerPrint(topLevel);
90  if (*topLevelFingerPrint == afterFingerPrint) {
91  // Note: Run "mlir-opt -debug" to see which pattern is broken.
92  llvm::report_fatal_error(
93  "pattern returned success but IR did not change");
94  }
95  for (const auto &it : fingerprints) {
96  // Skip top-level op, its finger print is never invalidated.
97  if (it.first == topLevel)
98  continue;
99  // Note: Finger print computation may crash when an op was erased
100  // without notifying the rewriter. (Run with ASAN to see where the op was
101  // erased; the op was probably erased directly, bypassing the rewriter
102  // API.) Finger print computation does may not crash if a new op was
103  // created at the same memory location. (But then the finger print should
104  // have changed.)
105  if (it.second !=
106  OperationFingerPrint(it.first, /*includeNested=*/false)) {
107  // Note: Run "mlir-opt -debug" to see which pattern is broken.
108  llvm::report_fatal_error("operation finger print changed");
109  }
110  }
111  }
112 
113  void notifyRewriteFailure() {
114  if (!topLevel)
115  return;
116 
117  // Pattern application failure => IR must not have changed.
118  OperationFingerPrint afterFingerPrint(topLevel);
119  if (*topLevelFingerPrint != afterFingerPrint) {
120  // Note: Run "mlir-opt -debug" to see which pattern is broken.
121  llvm::report_fatal_error("pattern returned failure but IR did change");
122  }
123  }
124 
125  void notifyFoldingSuccess() {
126  if (!topLevel)
127  return;
128 
129  // Make sure that the IR still verifies.
130  if (failed(verify(topLevel)))
131  llvm::report_fatal_error("IR failed to verify after folding");
132  }
133 
134 protected:
135  /// Invalidate the finger print of the given op, i.e., remove it from the map.
136  void invalidateFingerPrint(Operation *op) { fingerprints.erase(op); }
137 
138  void notifyBlockInserted(Block *block, Region *previous,
139  Region::iterator previousIt) override {
140  RewriterBase::ForwardingListener::notifyBlockInserted(block, previous,
141  previousIt);
142 
143  // Inserting or moving a block changes the region structure of both the
144  // new and previous owners. Either region may be detached.
145  invalidateFingerPrint(block->getParentOp());

147  invalidateFingerPrint(previous->getParentOp());
148  }
149 
150  void notifyBlockErased(Block *block) override {
151  RewriterBase::ForwardingListener::notifyBlockErased(block);
152 
153  // The block structure (number of blocks, types of block arguments, etc.)
154  // is part of the fingerprint of the parent op.
155  // TODO: The parent op fingerprint should also be invalidated when modifying
156  // the block arguments of a block, but we do not have a
157  // `notifyBlockModified` callback yet.
158  invalidateFingerPrint(block->getParentOp());
159  }

161  void notifyOperationInserted(Operation *op,
162  OpBuilder::InsertPoint previous) override {
163  RewriterBase::ForwardingListener::notifyOperationInserted(op, previous);
164  invalidateFingerPrint(op->getParentOp());
165  }
166 
167  void notifyOperationModified(Operation *op) override {
168  RewriterBase::ForwardingListener::notifyOperationModified(op);
169  invalidateFingerPrint(op);
170  }
171 
172  void notifyOperationErased(Operation *op) override {
173  RewriterBase::ForwardingListener::notifyOperationErased(op);
174  op->walk([this](Operation *op) { invalidateFingerPrint(op); });
175  }

178  /// against these finger prints after pattern application to detect cases
179  /// where IR was modified directly, bypassing the rewriter API.
180  DenseMap<Operation *, OperationFingerPrint> fingerprints;
181 
182  /// Top-level operation of the current greedy rewrite.
183  Operation *topLevel = nullptr;
184 
185  /// Finger print of the top-level operation.
186  std::optional<OperationFingerPrint> topLevelFingerPrint;

259 void Worklist::push(Operation *op) {
260  assert(op && "cannot push nullptr to worklist");
261  // Check to see if the worklist already contains this op.
262  if (!map.insert({op, list.size()}).second)
263  return;
264  list.push_back(op);
265 }
266 
267 Operation *Worklist::pop() {
268  assert(!empty() && "cannot pop from empty worklist");
269  // Skip and remove all trailing nullptr.
270  while (!list.back())
271  list.pop_back();
272  Operation *op = list.back();
273  list.pop_back();
274  map.erase(op);
275  // Cleanup: Remove all trailing nullptr.
276  while (!list.empty() && !list.back())
277  list.pop_back();
278  return op;
279 }

281 void Worklist::remove(Operation *op) {
282  assert(op && "cannot remove nullptr from worklist");
283  auto it = map.find(op);
284  if (it != map.end()) {
285  assert(list[it->second] == op && "malformed worklist data structure");
286  list[it->second] = nullptr;
287  map.erase(it);
288  }
289 }
290 
291 void Worklist::reverse() {
292  std::reverse(list.begin(), list.end());
293  for (size_t i = 0, e = list.size(); i != e; ++i)
294  map[list[i]] = i;
295 }

332 /// Tracks which blocks are reachable from their region's entry block while
333 /// patterns rewrite the CFG, so that the driver can skip operations in blocks
334 /// that a rewrite disconnected. Notifications only record what changed; the
335 /// queries, which the driver issues between rewrites, bring the cache up to
336 /// date. `Block::isReachable` is not used because it traverses the region on
337 /// every call, while the driver queries once per processed operation. All
338 /// notifications are forwarded to the wrapped listener.
339 class ReachabilityListener : public RewriterBase::ForwardingListener {
340 public:
341  ReachabilityListener(RewriterBase::Listener *listener, Region *scope)
342  : RewriterBase::ForwardingListener(listener), scope(scope) {}
343 
344  /// Return whether `op` is reachable: its block is reachable from the entry
345  /// block of its region, and so is every containing block up to `scope`.
346  /// Must be called between rewrites, when the IR is in a consistent state.
347  bool isReachable(Operation *op);
348 
349  /// Drop all cached reachability information.
350  void clear() { regionStates.clear(); }
351 
352  void notifyBlockInserted(Block *block, Region *previous,
353  Region::iterator previousIt) override;
354  void notifyBlockErased(Block *block) override;
355  void notifyOperationInserted(Operation *op,
356  OpBuilder::InsertPoint previous) override;
357  void notifyOperationModified(Operation *op) override;
358  void notifyOperationErased(Operation *op) override;
359 
360 private:
361  /// The reachability cache of one region.
362  struct RegionState {
363  /// The entry block at the time of the traversal. The traversal starts
364  /// from the entry block, so a different entry block invalidates the cache.
365  Block *entry = nullptr;
366  /// The reachable blocks, mapped to their successors within the region as
367  /// of the last query. The snapshots let the next query tell lost edges
368  /// from added ones.
369  DenseMap<Block *, SmallVector<Block *, 2>> reachable;
370  /// Blocks of `reachable` whose reachability could not be decided
371  /// incrementally. A query for one of them traverses the region. Marking
372  /// and traversing both cost O(blocks), so a lost edge into a loop whose
373  /// target keeps other predecessors costs O(blocks) per rewrite if one of
374  /// the marked blocks is queried before the next change. Bottom-up
375  /// processing has usually popped the operations of these blocks already;
376  /// top-down processing pops them next.
377  llvm::SmallDenseSet<Block *, 4> uncertain;
378  /// Blocks whose terminator changed since the last query, in notification
379  /// order, so that the cache evolves deterministically.
380  llvm::SmallSetVector<Block *, 4> changedBlocks;
381  /// Blocks erased from or moved out of the region since the last query.
382  /// Their pointers are only compared, never dereferenced.
383  llvm::SmallDenseSet<Block *, 4> removedBlocks;
384 
385  /// Rebuild the cache from the entry block of `region`.
386  void traverse(Region *region);
387  /// Add `roots` and every block reachable from them to `reachable`. The
388  /// added blocks are uncertain if `isUncertain` is set.
389  void addReachable(SmallVector<Block *> roots, bool isUncertain);
390  /// Mark `block` and every block behind it in the snapshots uncertain.
391  void markUncertain(Block *block);
392  /// Apply the pending changes, if any. On failure, the state is
393  /// inconsistent and the region must be traversed again.
394  LogicalResult applyChanges();
395  };

398  /// record changes, so no pointer to a block of an untracked region is kept.

408  auto it = regionStates.find(region);
409  if (it != regionStates.end())

412  bool isBlockReachable(Block *block);
413 
414  /// The tracked regions. A region without an entry is traversed by the next

416  DenseMap<Region *, RegionState> regionStates;
417 
418  Region *const scope;

611 void ReachabilityListener::RegionState::addReachable(SmallVector<Block *> roots,
612  bool isUncertain) {
613  while (!roots.empty()) {
614  auto [it, inserted] = reachable.try_emplace(roots.pop_back_val());
615  if (!inserted)
616  continue;
617  auto successors = getRegionSuccessors(it->first);
618  it->second.append(successors.begin(), successors.end());

620  uncertain.insert(it->first);
621  roots.append(it->second);
622  }

636 LogicalResult ReachabilityListener::RegionState::applyChanges() {
637  if (changedBlocks.empty() && removedBlocks.empty())
638  return success();
639  // Blocks are dropped after the loop, so that the snapshot of a lost block
640  // remains available to every changed block that lost an edge to it.
641  SmallVector<Block *> unreachable;
642  for (Block *changed : changedBlocks) {
643  // A removed block is no longer part of the region, and a change in an
644  // uncached block cannot make any block reachable.
645  if (removedBlocks.contains(changed))
646  continue;
647  auto it = reachable.find(changed);
648  if (it == reachable.end())
649  continue;
650  // Keep even a disconnected block's snapshot up to date: a later rewrite
651  // may reconnect it without querying the block itself. Its added successors
652  // must remain uncertain until a traversal establishes their reachability.
653  if (changed != entry && changed->hasNoPredecessors())
654  markUncertain(changed);
655  // Keep the snapshot available while marking lost successors uncertain.
656  // A lost self-edge can make that traversal visit `changed` itself.
657  const auto &old = it->second;
658  auto current = getRegionSuccessors(changed);
659  for (Block *lost : old) {
660  if (current.contains(lost))
661  continue;
662  // The successors of a reachable block are reachable, so `lost` has a
663  // snapshot unless the cache is inconsistent. This is not expected to
664  // happen; the failure is a safety net that forces a traversal.
665  auto lostIt = reachable.find(lost);
666  if (lostIt == reachable.end())
667  return failure();
668  // `lost` and every block behind it may have become unreachable. When
669  // `lost` is gone or has no predecessors left, it is unreachable; if all
670  // of its cached successors are still successors of `changed`, nothing
671  // else changed. This is the shape left by folding a constant conditional
672  // branch and by merging a block into its predecessor. Any other lost
673  // edge leaves the blocks behind `lost` uncertain until a query for one
674  // of them traverses the region.
675  bool isGone = removedBlocks.contains(lost) || lost->hasNoPredecessors();
676  if (isGone && llvm::all_of(lostIt->second, [&](Block *successor) {
677  return current.contains(successor);
678  })) {
679  unreachable.push_back(lost);
680  continue;
681  }
682  markUncertain(lost);
683  }
684  // Added edges can only make more blocks reachable. Blocks reached from an
685  // uncertain block are uncertain as well.
686  SmallVector<Block *> roots;
687  for (Block *successor : current)
688  if (!reachable.contains(successor))
689  roots.push_back(successor);
690  // Nothing was inserted into `reachable` since `it` was looked up.
691  it->second.assign(current.begin(), current.end());
692  addReachable(std::move(roots), uncertain.contains(changed));
693  }
694  // Drop every block that left the region, so that no stale key can match a
695  // block inserted later at the same address.
696  unreachable.append(removedBlocks.begin(), removedBlocks.end());
697  for (Block *block : unreachable) {
698  reachable.erase(block);
699  uncertain.erase(block);
700  }
701  changedBlocks.clear();
702  removedBlocks.clear();
703  return success();
704 }

706 bool ReachabilityListener::isBlockReachable(Block *block) {
707  Region *region = block->getParent();
708  // A detached block is treated as reachable, as is the entry block.
709  if (!region || block == &region->front())
710  return true;
711  if (block->hasNoPredecessors())
712  return false;
713  auto [it, inserted] = regionStates.try_emplace(region);
714  RegionState &state = it->second;
715  bool isStale = inserted || state.entry != &region->front();
716  if (isStale || failed(state.applyChanges()) ||
717  state.uncertain.contains(block))
718  state.traverse(region);
719  bool result = state.reachable.contains(block);
720 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
721  assert(result == region->front().isReachable(block) &&
722  "reachability cache disagrees with a traversal");
723 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
724  return result;
725 }

727 bool ReachabilityListener::isReachable(Operation *op) {
728  // A nested region can be locally reachable while its owning operation is in
729  // an unreachable block, so check every containing block up to the scope.
730  while (Block *block = op->getBlock()) {
731  if (!isBlockReachable(block))
732  return false;
733  Region *region = block->getParent();
734  if (!region || region == scope)
735  break;
736  op = region->getParentOp();
737  }
738  return true;
739 }

741 void ReachabilityListener::notifyBlockInserted(Block *block, Region *previous,
742  Region::iterator previousIt) {
743  Region *region = block->getParent();
744  // A block moved within its region keeps its reachability, unless it becomes
745  // the entry block, which the next query detects. A block that is new to the
746  // region is unreachable until a recorded terminator change adds an edge to
747  // it, but a stale entry from an earlier removal cannot be trusted.
748  if (previous != region) {
749  if (previous)
750  markRemoved(previous, block);
751  auto it = regionStates.find(region);
752  if (it != regionStates.end() && (it->second.removedBlocks.contains(block) ||
753  it->second.reachable.contains(block)))
754  regionStates.erase(it);
755  }
756  RewriterBase::ForwardingListener::notifyBlockInserted(block, previous,
757  previousIt);
758 }

818 bool GreedyPatternRewriteDriver::processWorklist() {
819  // The cache is only maintained while this worklist pass runs. Between
820  // passes, the IR changes without notifications, e.g. in region
821  // simplification and in the unreachable-block sweep.
822  llvm::scope_exit clearReachability([&] { reachabilityListener.clear(); });
823 
824 #ifndef NDEBUG
825  const char *logLineComment =
826  "//===-------------------------------------------===//\n";
827 
828  /// A utility function to log a process result for the given reason.
829  auto logResult = [&](StringRef result, const llvm::Twine &msg = {}) {
830  logger.unindent();
831  logger.startLine() << "} -> " << result;
832  if (!msg.isTriviallyEmpty())
833  logger.getOStream() << " : " << msg;
834  logger.getOStream() << "\n";
835  };
836  auto logResultWithLine = [&](StringRef result, const llvm::Twine &msg = {}) {
837  logResult(result, msg);
838  logger.startLine() << logLineComment;
839  };
840 #endif
841 
842  bool changed = false;
843  int64_t numRewrites = 0;
844  while (!worklist.empty() &&
845  (numRewrites < config.getMaxNumRewrites() ||
846  config.getMaxNumRewrites() == GreedyRewriteConfig::kNoLimit)) {
847  auto *op = worklist.pop();
848  if (!reachabilityListener.isReachable(op)) {
849  LLVM_DEBUG(logger.startLine() << "Skipping unreachable operation : '"
850  << op->getName() << "'(" << op << ")\n");
851  continue;
852  }
853 
854  LLVM_DEBUG({
855  logger.getOStream() << "\n";
856  logger.startLine() << logLineComment;
857  logger.startLine() << "Processing operation : '" << op->getName() << "'("
858  << op << ") {\n";
859  logger.indent();
860 
861  // If the operation has no regions, just print it here.
862  if (op->getNumRegions() == 0) {
863  op->print(
864  logger.startLine(),
865  OpPrintingFlags().printGenericOpForm().elideLargeElementsAttrs());
866  logger.getOStream() << "\n\n";
867  }
868  });
869 
870  // If the operation is trivially dead - remove it.
871  if (deadnessListener.isOpTriviallyDead(op)) {
872  rewriter.eraseOp(op);
873  changed = true;
874 
875  LLVM_DEBUG(logResultWithLine("success", "operation is trivially dead"));
876  continue;
877  }
878 
879  // Try to fold this op. Do not fold constant ops. That would lead to an
880  // infinite folding loop, as every constant op would be folded to an
881  // Attribute and then immediately be rematerialized as a constant op, which
882  // is then put on the worklist.
883  if (config.isFoldingEnabled() && !op->hasTrait<OpTrait::ConstantLike>()) {
884  SmallVector<OpFoldResult> foldResults;
885  if (succeeded(op->fold(foldResults))) {
886  LLVM_DEBUG(logResultWithLine("success", "operation was folded"));
887 #ifndef NDEBUG
888  Operation *dumpRootOp = getDumpRootOp(op);
889 #endif // NDEBUG
890  if (foldResults.empty()) {
891  // Folding modified the op without using the rewriter. Notify the
892  // cache listeners, which forward the modification to the driver.
893  deadnessListener.notifyOperationModified(op);
894  changed = true;
895  LLVM_DEBUG(logSuccessfulFolding(dumpRootOp));
896 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
897  expensiveChecks.notifyFoldingSuccess();
898 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
899  continue;
900  }
901 
902  // Op results can be replaced with `foldResults`.
903  assert(foldResults.size() == op->getNumResults() &&
904  "folder produced incorrect number of results");
905  OpBuilder::InsertionGuard g(rewriter);
906  rewriter.setInsertionPoint(op);
907  SmallVector<Value> replacements;
908  bool materializationSucceeded = true;
909  for (auto [ofr, resultType] :
910  llvm::zip_equal(foldResults, op->getResultTypes())) {
911  if (auto value = dyn_cast<Value>(ofr)) {
912  assert(value.getType() == resultType &&
913  "folder produced value of incorrect type");
914  replacements.push_back(value);
915  continue;
916  }
917  // Materialize Attributes as SSA values.
918  Operation *constOp = op->getDialect()->materializeConstant(
919  rewriter, cast<Attribute>(ofr), resultType, op->getLoc());
920 
921  if (!constOp) {
922  // If materialization fails, cleanup any operations generated for
923  // the previous results.
924  llvm::SmallDenseSet<Operation *> replacementOps;
925  for (Value replacement : replacements) {
926  assert(replacement.use_empty() &&
927  "folder reused existing op for one result but constant "
928  "materialization failed for another result");
929  replacementOps.insert(replacement.getDefiningOp());
930  }
931  for (Operation *op : replacementOps) {
932  rewriter.eraseOp(op);
933  }
934 
935  materializationSucceeded = false;
936  break;
937  }
938 
939  assert(constOp->hasTrait<OpTrait::ConstantLike>() &&
940  "materializeConstant produced op that is not a ConstantLike");
941  assert(constOp->getResultTypes()[0] == resultType &&
942  "materializeConstant produced incorrect result type");
943  replacements.push_back(constOp->getResult(0));
944  }
945 
946  if (materializationSucceeded) {
947  rewriter.replaceOp(op, replacements);
948  changed = true;
949  LLVM_DEBUG(logSuccessfulFolding(dumpRootOp));
950 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
951  expensiveChecks.notifyFoldingSuccess();
952 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
953  continue;
954  }
955  }
956  }
957 
958  // Try to match one of the patterns. The rewriter is automatically
959  // notified of any necessary changes, so there is nothing else to do
960  // here.
961  auto canApplyCallback = [&](const Pattern &pattern) {
962  LLVM_DEBUG({
963  logger.getOStream() << "\n";
964  logger.startLine() << "* Pattern " << pattern.getDebugName() << " : '"
965  << op->getName() << " -> (";
966  llvm::interleaveComma(pattern.getGeneratedOps(), logger.getOStream());
967  logger.getOStream() << ")' {\n";
968  logger.indent();
969  });
970  if (RewriterBase::Listener *listener = config.getListener())
971  listener->notifyPatternBegin(pattern, op);
972  return true;
973  };
974  function_ref<bool(const Pattern &)> canApply = canApplyCallback;
975  auto onFailureCallback = [&](const Pattern &pattern) {
976  LLVM_DEBUG(logResult("failure", "pattern failed to match"));
977  if (RewriterBase::Listener *listener = config.getListener())
978  listener->notifyPatternEnd(pattern, failure());
979  };
980  function_ref<void(const Pattern &)> onFailure = onFailureCallback;
981  auto onSuccessCallback = [&](const Pattern &pattern) {
982  LLVM_DEBUG(logResult("success", "pattern applied successfully"));
983  if (RewriterBase::Listener *listener = config.getListener())
984  listener->notifyPatternEnd(pattern, success());
985  return success();
986  };
987  function_ref<LogicalResult(const Pattern &)> onSuccess = onSuccessCallback;
988 
989 #ifdef NDEBUG
990  // Optimization: PatternApplicator callbacks are not needed when running in
991  // optimized mode and without a listener.
992  if (!config.getListener()) {
993  canApply = nullptr;
994  onFailure = nullptr;
995  onSuccess = nullptr;
996  }
997 #endif // NDEBUG
998 
999 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1000  if (config.getScope()) {
1001  expensiveChecks.computeFingerPrints(config.getScope()->getParentOp());
1002  }
1003  llvm::scope_exit clearFingerprints([&]() { expensiveChecks.clear(); });
1004 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1005 
1006  LogicalResult matchResult =
1007  matcher.matchAndRewrite(op, rewriter, canApply, onFailure, onSuccess);
1008 
1009  if (succeeded(matchResult)) {
1010  LLVM_DEBUG(logResultWithLine("success", "at least one pattern matched"));
1011 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1012  expensiveChecks.notifyRewriteSuccess();
1013 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1014  changed = true;
1015  ++numRewrites;
1016  } else {
1017  LLVM_DEBUG(logResultWithLine("failure", "all patterns failed to match"));
1018 #if MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1019  expensiveChecks.notifyRewriteFailure();
1020 #endif // MLIR_ENABLE_EXPENSIVE_PATTERN_API_CHECKS
1021  }
1022  }
1023 
1024  return changed;
1025 }

1027 void GreedyPatternRewriteDriver::addToWorklist(Operation *op) {
1028  assert(op && "expected valid op");
1029  // Gather potential ancestors while looking for a "scope" parent region.
1030  SmallVector<Operation *, 8> ancestors;
1031  Region *region = nullptr;
1032  do {
1033  ancestors.push_back(op);
1034  region = op->getParentRegion();
1035  if (config.getScope() == region) {
1036  // Scope (can be `nullptr`) was reached. Stop traveral and enqueue ops.
1037  for (Operation *op : ancestors)
1038  addSingleOpToWorklist(op);
1039  return;
1040  }
1041  if (region == nullptr)
1042  return;
1043  } while ((op = region->getParentOp()));
1044 }

1086 void GreedyPatternRewriteDriver::addOperandsToWorklist(Operation *op) {
1087  for (Value operand : op->getOperands()) {
1088  // If this operand currently has at most 2 users, add its defining op to the
1089  // worklist. Indeed, after the op is deleted, then the operand will have at
1090  // most 1 user left. If it has 0 users left, it can be deleted too,
1091  // and if it has 1 user left, there may be further canonicalization
1092  // opportunities.
1093  if (!operand)
1094  continue;
1095 
1096  auto *defOp = operand.getDefiningOp();
1097  if (!defOp)
1098  continue;
1099 
1100  Operation *otherUser = nullptr;
1101  bool hasMoreThanTwoUses = false;
1102  for (auto *user : operand.getUsers()) {
1103  if (user == op || user == otherUser)
1104  continue;
1105  if (!otherUser) {
1106  otherUser = user;
1107  continue;
1108  }
1109  hasMoreThanTwoUses = true;
1110  break;
1111  }
1112  if (hasMoreThanTwoUses)
1113  continue;
1114 
1115  addToWorklist(defOp);
1116  }
1117 }

1228 LogicalResult RegionPatternRewriteDriver::simplify(bool *changed) && {
1229  bool continueRewrites = false;
1230  bool anyChanged = false;
1231  int64_t iteration = 0;
1232  MLIRContext *ctx = rewriter.getContext();
1233  do {
1234  // Check if the iteration limit was reached.
1235  if (++iteration > config.getMaxIterations() &&
1236  config.getMaxIterations() != GreedyRewriteConfig::kNoLimit)
1237  break;
1238 
1239  // New iteration: start with an empty worklist.
1240  worklist.clear();
1241 
1242  // `OperationFolder` CSE's constant ops (and may move them into parents
1243  // regions to enable more aggressive CSE'ing).
1244  OperationFolder folder(ctx, rewriter.getListener());
1245  auto insertKnownConstant = [&](Operation *op) {
1246  // Check for existing constants when populating the worklist. This avoids
1247  // accidentally reversing the constant order during processing.
1248  Attribute constValue;
1249  if (!matchPattern(op, m_Constant(&constValue)))
1250  return false;
1251  auto result = folder.insertKnownConstant(op, constValue);
1252  anyChanged |=
1253  result != OperationFolder::ConstantInsertionResult::Unchanged;
1254  return result == OperationFolder::ConstantInsertionResult::Replaced;
1255  };
1256 
1257  if (!config.getUseTopDownTraversal()) {
1258  // Add operations to the worklist in postorder.
1259  region.walk([&](Operation *op) {

1428 /// Find the region that is the closest common ancestor of all given ops.
1429 ///
1430 /// Note: This function returns `nullptr` if there is a top-level op among the
1431 /// given list of ops.
1432 static Region *findCommonAncestor(ArrayRef<Operation *> ops) {
1433  assert(!ops.empty() && "expected at least one op");
1434  // Fast path in case there is only one op.
1435  if (ops.size() == 1)
1436  return ops.front()->getParentRegion();
1437 
1438  Region *region = ops.front()->getParentRegion();
1439  ops = ops.drop_front();
1440  int sz = ops.size();
1441  llvm::BitVector remainingOps(sz, true);
1442  while (region) {
1443  int pos = -1;
1444  // Iterate over all remaining ops.
1445  while ((pos = remainingOps.find_first_in(pos + 1, sz)) != -1) {
1446  // Is this op contained in `region`?
1447  if (region->findAncestorOpInRegion(*ops[pos]))
1448  remainingOps.reset(pos);
1449  }
1450  if (remainingOps.none())
1451  break;
1452  region = region->getParentRegion();
1453  }
