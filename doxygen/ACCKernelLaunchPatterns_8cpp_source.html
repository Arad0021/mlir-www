37 static bool isDevicePtrMapping(Value operand) {
38  Operation *mapOp = operand.getDefiningOp();
39  if (!mapOp)
40  return false;
41  if (std::optional<MapFlags> flags = getMapFlags(mapOp))
42  return bitEnumContainsAny(*flags, MapFlags::devptr);
43  if (std::optional<DataClause> clause = getDataClause(mapOp))
44  return *clause == DataClause::acc_deviceptr;
45  return false;
46 }

54 /// The flags of a data clause operation that maps a kernel argument. A kernel
55 /// argument only looks up the mapping its clause established, so the clauses
56 /// that establish or tear down something else contribute no flag for it.
57 static MapFlags getKernelDataClauseMapFlags(Operation *entryOp) {
58  bool ptrAndObj = hasAttachPoint(entryOp);
59  MapFlags ptrAndObjFlag = ptrAndObj ? MapFlags::ptr_and_obj : MapFlags::none;
60  std::optional<DataClause> clause = getDataClause(entryOp);
61  if (clause == DataClause::acc_declare_device_resident)
62  return MapFlags::present | ptrAndObjFlag;
63  if (clause == DataClause::acc_attach || clause == DataClause::acc_detach)
64  return ptrAndObjFlag;
65  return bitEnumClear(computeDataClauseMapFlags(entryOp, ptrAndObj),
66  MapFlags::device_resident);
67 }

69 /// The flags a kernel argument is passed with. An operand the environment
70 /// already mapped is passed as the device address its entry call returned. An
71 /// operand that no clause maps is passed by value.
72 static MapFlags getKernelArgMapFlags(Value operand, bool alreadyMapped,
73  const ACCRuntimeCallConfig &config) {
74  Operation *mapOp = operand.getDefiningOp();
75  if (!isDataEntryOp(mapOp))
76  return mapOp ? config.postProcessMapFlags(mapOp, MapFlags::literal)
77  : MapFlags::literal;
78 
79  MapFlags flags;
80  if (alreadyMapped)
81  flags = MapFlags::devptr | MapFlags::literal;
82  else if (auto mapInfo = dyn_cast<MapInfoOp>(mapOp))
83  flags = mapInfo.getMapFlags();
84  else
85  flags = getKernelDataClauseMapFlags(mapOp);
86  if (getImplicitFlag(mapOp))
87  flags = flags | MapFlags::implicit;
88  return config.postProcessMapFlags(mapOp, flags);
89 }

91 /// The descriptor of a kernel argument. The device address the environment
92 /// returned is passed as is, since the runtime would map an address that
93 /// carries a descriptor again.
94 static FailureOr<Value>
95 getKernelArgDescriptor(Location loc, Value operand, Value converted,
96  bool alreadyMapped, OpenACCSupport &accSupport,
97  ConversionPatternRewriter &rewriter) {
98  Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
99  Operation *mapOp = operand.getDefiningOp();
100  if (alreadyMapped)
101  return Value(LLVM::ZeroOp::create(rewriter, loc, ptrTy));
102  if (isDataEntryOp(mapOp))
103  return createACCArgumentDescriptor(mapOp, converted, accSupport, rewriter);
104  if (auto memrefType = dyn_cast<MemRefType>(operand.getType())) {
105  ModuleOp module = rewriter.getInsertionBlock()
106  ->getParentOp()
107  ->getParentOfType<ModuleOp>();
108  auto sizeAndAlignment = acc::getTypeSizeAndAlignment(
109  memrefType.getElementType(), module, &accSupport);
110  if (!sizeAndAlignment) {
111  std::string message;
112  llvm::raw_string_ostream os(message);
113  os << "unhandled type size computation for type: "
114  << memrefType.getElementType();
115  (void)accSupport.emitNYI(loc, message);
116  return failure();
117  }
118  return createACCMemRefDescriptorWrapperArg(
119  loc, memrefType, converted, /*bounds=*/{},
120  constantI64(loc, sizeAndAlignment->first.getFixedValue(), rewriter),
121  rewriter);
122  }
123  return Value(LLVM::ZeroOp::create(rewriter, loc, ptrTy));
124 }

126 /// Emits the argument arrays of the kernel launched by \p launch, one entry per
127 /// kernel operand. Each operand is one parameter of the kernel.
128 static FailureOr<ACCDataRuntimeArgs> emitKernelRuntimeArgs(
129  gpu::LaunchFuncOp launch, gpu::LaunchFuncOpAdaptor adaptor,
130  ValueRange alreadyMappedOperands, ConversionPatternRewriter &rewriter,
131  OpenACCSupport &accSupport, Region &globalSymbolRegion,
132  SymbolTable &symbolTable, const ACCRuntimeCallConfig &config) {
133  Location loc = launch.getLoc();
134  SmallVector<ACCMappedObjectArgs> objects;
135  for (auto [operand, converted] : llvm::zip_equal(
136  launch.getKernelOperands(), adaptor.getKernelOperands())) {
137  bool alreadyMapped = llvm::is_contained(alreadyMappedOperands, operand);
138  MapFlags mapFlags = getKernelArgMapFlags(operand, alreadyMapped, config);
139  Operation *mapOp = operand.getDefiningOp();
140 
141  ACCMappedObjectArgs args;
142  // A kernel argument states the base of the mapping it looks up, which for
143  // an object the declare directive made resident is that of its entry.
144  args.base = createACCMappedObjectBase(
145  loc, isDataEntryOp(mapOp) ? mapOp : nullptr, mapFlags, rewriter);
146  args.pointer = createACCMappedObjectPointer(loc, operand, converted,
147  mapFlags, rewriter);
148  FailureOr<Value> size = createACCMappedObjectSize(
149  loc, operand, converted, mapFlags, accSupport, rewriter);
150  if (failed(size))
151  return failure();
152  args.size = *size;
153  args.type =
154  constantI64(loc, config.getMapFlagsRuntimeValue(mapFlags), rewriter);
155  args.name = createACCMappedObjectName(
156  loc, operand, accSupport, globalSymbolRegion, &symbolTable, rewriter);
157  FailureOr<Value> descriptor = getKernelArgDescriptor(
158  loc, operand, converted, alreadyMapped, accSupport, rewriter);
159  if (failed(descriptor))
160  return failure();
161  args.descriptor = *descriptor;
162  objects.push_back(args);
163  }
164 
165  ACCDataRuntimeArgs runtimeArgs;
166  createACCMappedObjectArrays(loc, objects, runtimeArgs, rewriter);
167  return runtimeArgs;
168 }

170 static FailureOr<Value> createKernelArgumentStruct(
171  gpu::LaunchFuncOp launch, gpu::LaunchFuncOpAdaptor adaptor,
172  ValueRange alreadyMappedOperands, ConversionPatternRewriter &rewriter,
173  OpenACCSupport &accSupport, Region &globalSymbolRegion,
174  SymbolTable &symbolTable, const ACCRuntimeCallConfig &config) {
175  FailureOr<ACCDataRuntimeArgs> emitted = emitKernelRuntimeArgs(
176  launch, adaptor, alreadyMappedOperands, rewriter, accSupport,
177  globalSymbolRegion, symbolTable, config);
178  if (failed(emitted))
179  return failure();
180  ACCDataRuntimeArgs runtimeArgs = *emitted;
181 
182  Location loc = launch.getLoc();
183  MLIRContext *context = rewriter.getContext();
184  Type i32Ty = rewriter.getI32Type();
185  Type ptrTy = LLVM::LLVMPointerType::get(context);
186  Type argsType = getKernelArgumentType(context);
187  Value args = LLVM::ZeroOp::create(rewriter, loc, argsType);
188  auto insert = [&](Value value, KernelArgumentField field) {
189  args = LLVM::InsertValueOp::create(
190  rewriter, loc, argsType, args, value,
191  ArrayRef<int64_t>{getKernelArgumentFieldIndex(field)});
192  };
193  insert(runtimeArgs.argNum, KernelArgumentField::ArgNum);
194  insert(runtimeArgs.argBasePtrs, KernelArgumentField::ArgBasePtrs);
195  insert(runtimeArgs.argPtrs, KernelArgumentField::ArgPtrs);
196  insert(runtimeArgs.argSizes, KernelArgumentField::ArgSizes);
197  insert(runtimeArgs.argTypes, KernelArgumentField::ArgTypes);
198  insert(runtimeArgs.argNames, KernelArgumentField::ArgNames);
199  insert(runtimeArgs.argMappers, KernelArgumentField::ArgMappers);
200  insert(runtimeArgs.argDescs, KernelArgumentField::ArgDescs);
201 
202  // The vector length is the first block dimension. The worker count is the
203  // product of the remaining ones.
204  Value blockY = castToI64(loc, adaptor.getBlockSizeY(), rewriter);
205  Value blockZ = castToI64(loc, adaptor.getBlockSizeZ(), rewriter);
206  insert(LLVM::MulOp::create(rewriter, loc, blockY, blockZ),
207  KernelArgumentField::NumWorkers);
208  insert(castToI64(loc, adaptor.getBlockSizeX(), rewriter),
209  KernelArgumentField::VectorLength);
210  Value dynamicSharedMemorySize =
211  adaptor.getDynamicSharedMemorySize()
212  ? castToI64(loc, adaptor.getDynamicSharedMemorySize(), rewriter)
213  : constantI64(loc, 0, rewriter);
214  insert(dynamicSharedMemorySize, KernelArgumentField::SmemSize);
215 
216  // The grid sizes are the entries of the array in the NumGangs field.
217  int64_t numGangsField =
218  getKernelArgumentFieldIndex(KernelArgumentField::NumGangs);
219  std::array<Value, 3> gridSizes = {
220  adaptor.getGridSizeX(), adaptor.getGridSizeY(), adaptor.getGridSizeZ()};
221  for (auto [dimension, gridSize] : llvm::enumerate(gridSizes))
222  args = LLVM::InsertValueOp::create(
223  rewriter, loc, argsType, args, castToI64(loc, gridSize, rewriter),
224  ArrayRef<int64_t>{numGangsField, static_cast<int64_t>(dimension)});
225 
226  Value one = LLVM::ConstantOp::create(rewriter, loc, i32Ty, 1);
227  Value argsPtr = LLVM::AllocaOp::create(rewriter, loc, ptrTy, argsType, one);
228  LLVM::StoreOp::create(rewriter, loc, args, argsPtr);
229  return argsPtr;
230 }

232 static FailureOr<Value> getKernelPointer(gpu::LaunchFuncOp launch,
233  ConversionPatternRewriter &rewriter,
234  Region &globalSymbolRegion,
235  SymbolTable &symbolTable) {
236  Location loc = launch.getLoc();
237  Type ptrTy = LLVM::LLVMPointerType::get(rewriter.getContext());
238  StringRef kernelName = launch.getKernel().getLeafReference().getValue();
239  if (auto function = symbolTable.lookup<LLVM::LLVMFuncOp>(kernelName))
240  return LLVM::AddressOfOp::create(rewriter, loc, ptrTy, function.getName())
241  .getResult();
242  if (auto function = symbolTable.lookup<func::FuncOp>(kernelName))
243  return LLVM::AddressOfOp::create(rewriter, loc, ptrTy, function.getName())
244  .getResult();
245 
246  auto gpuFunction = SymbolTable::lookupNearestSymbolFrom<gpu::GPUFuncOp>(
247  launch, launch.getKernel());
248  if (!gpuFunction)
249  return failure();
250 
251  {
252  OpBuilder::InsertionGuard guard(rewriter);
253  rewriter.setInsertionPointToStart(&globalSymbolRegion.front());
254  auto functionType = LLVM::LLVMFunctionType::get(
255  LLVM::LLVMVoidType::get(rewriter.getContext()), {});
256  auto declaration =
257  LLVM::LLVMFuncOp::create(rewriter, loc, kernelName, functionType);
258  symbolTable.insert(declaration);
259  }
260  return LLVM::AddressOfOp::create(rewriter, loc, ptrTy, kernelName)
261  .getResult();
262 }

275 /// The kernel entry point reports a status other than success when the kernel
276 /// could not run on the device. The program is then meant to run the region on
277 /// the host, which is not implemented, so the failure is reported and the
278 /// program stops.
279 static void emitHostFallbackNotImplemented(Location loc, Value status,
280  ConversionPatternRewriter &rewriter,
281  Region &globalSymbolRegion,
282  SymbolTable &symbolTable) {
283  MLIRContext *context = rewriter.getContext();
284  Type i32Ty = rewriter.getI32Type();
285  Type ptrTy = LLVM::LLVMPointerType::get(context);
286  Type voidTy = LLVM::LLVMVoidType::get(context);
287 
288  Block *preBlock = rewriter.getInsertionBlock();
289  Block *continueBlock =
290  rewriter.splitBlock(preBlock, rewriter.getInsertionPoint());
291  Block *errorBlock =
292  rewriter.createBlock(preBlock->getParent(), continueBlock->getIterator());
293  rewriter.setInsertionPointToEnd(preBlock);
294  Value success = LLVM::ConstantOp::create(
295  rewriter, loc, i32Ty, static_cast<int32_t>(RuntimeResult::Success));
296  Value failed = LLVM::ICmpOp::create(rewriter, loc, LLVM::ICmpPredicate::ne,
297  status, success);
298  LLVM::CondBrOp::create(rewriter, loc, failed, errorBlock, ValueRange{},
299  continueBlock, ValueRange{});
300 
301  // TODO: Runtime API should provide a mechanism to report this error.
302  rewriter.setInsertionPointToStart(errorBlock);
303  declareLibc(symbolTable, globalSymbolRegion, loc, "puts", i32Ty, {ptrTy});
304  declareLibc(symbolTable, globalSymbolRegion, loc, "fflush", i32Ty, {ptrTy});
305  declareLibc(symbolTable, globalSymbolRegion, loc, "abort", voidTy, {});
306  Value message = getOrCreateGlobalString(
307  loc, rewriter, getInternalGlobalName("message", "host_fallback_nyi"),
308  "host fallback execution is not implemented\n", globalSymbolRegion,
309  &symbolTable);
310  LLVM::CallOp::create(rewriter, loc, TypeRange{i32Ty},
311  SymbolRefAttr::get(context, "puts"),
312  ValueRange{message});
313  Value nullFile = LLVM::ZeroOp::create(rewriter, loc, ptrTy);
314  LLVM::CallOp::create(rewriter, loc, TypeRange{i32Ty},
315  SymbolRefAttr::get(context, "fflush"),
316  ValueRange{nullFile});
317  LLVM::CallOp::create(rewriter, loc, TypeRange{},
318  SymbolRefAttr::get(context, "abort"), ValueRange{});
319  LLVM::UnreachableOp::create(rewriter, loc);
320  rewriter.setInsertionPointToStart(continueBlock);
321 }

323 static LogicalResult
324 emitKernelCall(gpu::LaunchFuncOp launch, gpu::LaunchFuncOpAdaptor adaptor,
325  Value asyncQueue, ValueRange alreadyMappedOperands,
326  ConversionPatternRewriter &rewriter, OpenACCSupport &accSupport,
327  Region &globalSymbolRegion, SymbolTable &symbolTable,
328  const ACCRuntimeCallConfig &config) {
329  Location loc = launch.getLoc();
330  FailureOr<Value> kernelPointer =
331  getKernelPointer(launch, rewriter, globalSymbolRegion, symbolTable);
332  if (failed(kernelPointer)) {
333  (void)accSupport.emitNYI(loc, "undefined kernel");
334  return failure();
335  }
336  FailureOr<Value> arguments = createKernelArgumentStruct(
337  launch, adaptor, alreadyMappedOperands, rewriter, accSupport,
338  globalSymbolRegion, symbolTable, config);
339  if (failed(arguments))
340  return failure();
341 
342  StringRef kernelName = launch.getKernel().getLeafReference().getValue();
343  Value kernelNamePtr = getOrCreateGlobalString(
344  loc, rewriter, getInternalGlobalName("kernel_name", kernelName),
345  kernelName, globalSymbolRegion, &symbolTable);
346  Value ident = createIdent(loc, getParentFunctionName(launch), rewriter,
347  globalSymbolRegion, config, &symbolTable);
348  // The kernel call uses the explicit data-region flag.
349  Value flags = constantI64(loc, static_cast<int64_t>(ACCDataRegionFlags::None),
350  rewriter);
351  // The launch is made for the default device type, as the data calls of the
352  // environment are.
353  Value deviceType = constantI64(
354  loc, config.getDeviceTypeRuntimeValue(DeviceType::None), rewriter);
355  Value binaryDescriptor = config.createKernelBinaryDescriptor(loc, rewriter);
356  FailureOr<LLVM::CallOp> call =
357  createRuntimeCall(loc, rewriter, globalSymbolRegion, symbolTable,
358  RuntimeFunction::ACCRTL_tgt_acc_kernel, config,
359  {ident, *kernelPointer, flags, deviceType, *arguments,
360  asyncQueue, kernelNamePtr, binaryDescriptor});
361  if (failed(call))
362  return failure();
363  emitHostFallbackNotImplemented(loc, call->getResult(), rewriter,
364  globalSymbolRegion, symbolTable);
365  return success();
366 }

369 struct KernelEnvironmentLowering
370  : public ConvertOpToLLVMPattern<KernelEnvironmentOp> {
371  KernelEnvironmentLowering(const LLVMTypeConverter &converter,
372  OpenACCSupport &accSupport,
373  Region &globalSymbolRegion,
374  SymbolTable &symbolTable,
375  const ACCRuntimeCallConfig &config,
376  DeviceType clauseDeviceType,
377  PatternBenefit benefit = 1)
378  : ConvertOpToLLVMPattern<KernelEnvironmentOp>(converter, benefit),
379  accSupport(accSupport), globalSymbolRegion(globalSymbolRegion),
380  symbolTable(symbolTable), config(config),
381  clauseDeviceType(clauseDeviceType) {}
382 
383  LogicalResult
384  matchAndRewrite(KernelEnvironmentOp environment, OpAdaptor adaptor,
385  ConversionPatternRewriter &rewriter) const override {
386  ValueRange mappingOperands = environment.getDataClauseOperands();
387  ValueRange convertedMappingOperands = adaptor.getDataClauseOperands();
388  Location beginLoc = mappingOperands.empty()
389  ? environment.getLoc()
390  : mappingOperands.front().getLoc();
391  Location endLoc = getMappingExitLoc(mappingOperands).value_or(beginLoc);
392  Value asyncQueue =
393  getAsyncQueue(environment, clauseDeviceType, rewriter, config);
394  if (failed(emitWaitClause(environment, clauseDeviceType, asyncQueue,
395  rewriter, accSupport, globalSymbolRegion,
396  symbolTable, config)))
397  return failure();
398 
399  SmallVector<Value> alreadyMappedOperands;
400  for (Value operand : mappingOperands)
401  if (!isDevicePtrMapping(operand))
402  alreadyMappedOperands.push_back(operand);
403 
404  ACCDataCallOptions beginOptions;
405  beginOptions.flags = static_cast<int64_t>(ACCDataRegionFlags::Implicit);
406  beginOptions.returnAddressOperands = alreadyMappedOperands;
407  FailureOr<SmallVector<Value>> devicePointers = emitACCDataRuntimeCall(
408  beginLoc, RuntimeFunction::ACCRTL_tgt_acc_data_begin, mappingOperands,
409  convertedMappingOperands, rewriter, accSupport, globalSymbolRegion,
410  symbolTable, config, {asyncQueue}, beginOptions);
411  if (failed(devicePointers))
412  return failure();
413 
414  llvm::DenseMap<Value, Value> deviceAddress;
415  for (auto [operand, pointer] :
416  llvm::zip_equal(alreadyMappedOperands, *devicePointers))
417  deviceAddress[operand] = pointer;
418 
419  SmallVector<gpu::LaunchFuncOp> launches;
420  environment.walk(
421  [&](gpu::LaunchFuncOp launch) { launches.push_back(launch); });
422  for (gpu::LaunchFuncOp launch : launches) {
423  SmallVector<Value> converted;
424  converted.reserve(launch->getNumOperands());
425  for (Value operand : launch->getOperands()) {
426  Value value;
427  // A data entry in the region is mapped by this kernel, so it is passed
428  // as the variable of its clause.
429  if (isDataEntryInsideEnvironment(environment, operand))
430  value = remapOrSelf(getVar(operand.getDefiningOp()), rewriter);
431  else
432  value = deviceAddress.lookup(operand);
433  if (!value)
434  value = rewriter.getRemappedValue(operand);
435  if (!value)
436  return rewriter.notifyMatchFailure(launch,
437  "unconverted kernel operand");
438  converted.push_back(value);
439  }
440 
441  gpu::LaunchFuncOpAdaptor launchAdaptor(converted, launch);
442  Value queue = launchAdaptor.getAsyncObject();
443  queue = queue ? castToI64(launch.getLoc(), queue, rewriter) : asyncQueue;
444  rewriter.setInsertionPoint(launch);
445  if (failed(emitKernelCall(launch, launchAdaptor, queue,
446  alreadyMappedOperands, rewriter, accSupport,
447  globalSymbolRegion, symbolTable, config)))
448  return failure();
449  rewriter.eraseOp(launch);
450  }
451 
452  rewriter.setInsertionPointAfter(environment);
453  ACCDataCallOptions endOptions;
454  endOptions.callKind = ACCDataCallKind::DataExit;
455  endOptions.flags = static_cast<int64_t>(ACCDataRegionFlags::Implicit);
456  if (failed(emitACCDataRuntimeCall(
457  endLoc, RuntimeFunction::ACCRTL_tgt_acc_data_end, mappingOperands,
458  convertedMappingOperands, rewriter, accSupport, globalSymbolRegion,
459  symbolTable, config, {asyncQueue}, endOptions)))
460  return failure();
461 
462  spliceConstructRegion(environment, environment.getRegion(), rewriter);
463  rewriter.eraseOp(environment);
464  return success();
465  }
466 
467  OpenACCSupport &accSupport;
468  Region &globalSymbolRegion;
469  SymbolTable &symbolTable;
470  ACCRuntimeCallConfig config;
471  DeviceType clauseDeviceType;
472 };
