36 /// The arguments every mapping entry point of the OpenACC runtime takes, in the
37 /// order they are passed. An entry point that maps `argNum` objects reads one
38 /// element from each of the arrays below per object; an entry point may also
39 /// take further arguments of its own, which are not part of this set.
40 struct ACCDataRuntimeArgs {
41  /// Source position and enclosing function name of the directive.
42  Value ident;
43  /// Region flags, in the encoding of `ACCDataRegionFlags`.
44  Value flags;
45  /// Device type the directive applies to, in the runtime encoding.
46  Value deviceType;
47  /// Number of mapped objects, that is, the length of each array below.
48  Value argNum;
49  /// Array of pointer slots to attach a mapped object to, null when the object
50  /// is not attached to anything.
51  Value argBasePtrs;
52  /// Array of addresses of the mapped objects themselves.
53  Value argPtrs;
54  /// Array of object sizes in bytes; zero where the descriptor or the bounds
55  /// state the extent instead.
56  Value argSizes;
57  /// Array of map-type flags, in the runtime encoding of `acc::MapFlags`.
58  Value argTypes;
59  /// Array of variable names for runtime diagnostics, null where unknown.
60  Value argNames;
61  /// Array of user-defined mappers. OpenACC has none, so this is always null.
62  Value argMappers;
63  /// Array of descriptors stating layout and bounds, null for whole objects of
64  /// a known size.
65  Value argDescs;
66 
67  /// The `argBasePtrs` slot the entry point writes the device address of each
68  /// requested operand into, in the order the operands were requested.
69  SmallVector<unsigned> returnAddressSlots;
70 
71  /// Returns the fields above in the order the entry points take them, so that
72  /// a caller only appends the arguments specific to the one it calls.
73  SmallVector<Value> getCallArgs() const;
74 };

82 /// Flags of `__tgt_acc_data_begin` and `__tgt_acc_data_end`. They describe the
83 /// data region.
84 enum class ACCDataRegionFlags : int64_t {
85  /// An explicit data construct.
86  None = 0,
87  /// The implicit data region of a compute construct.
88  Implicit = 1,
89  /// The region of a structured declare directive in a local scope, which
90  /// lasts until that scope ends.
91  Declare = 2,
92 };

179 /// Emit the OpenACC data runtime arguments for data-clause operands. The
