99 /// Returns the length of the given memory intrinsic in bytes if it can be known
100 /// at compile-time on a best-effort basis, nothing otherwise.
101 template <class MemIntr>
102 std::optional<uint64_t> getStaticMemIntrLen(MemIntr op) {
103  APInt memIntrLen;
104  if (!matchPattern(op.getLen(), m_ConstantInt(&memIntrLen)))
105  return {};
106  if (memIntrLen.getBitWidth() > 64)
107  return {};
108  return memIntrLen.getZExtValue();
109 }

193 /// Returns whether one can be sure the memory intrinsic does not write outside
194 /// of the bounds of the given slot, on a best-effort basis.
195 template <class MemIntr>
196 static bool definitelyWritesOnlyWithinSlot(MemIntr op, const MemorySlot &slot,
197  const DataLayout &dataLayout) {
198  if (!isa<LLVM::LLVMPointerType>(slot.ptr.getType()) ||
199  op.getDst() != slot.ptr)
200  return false;
201 
202  std::optional<uint64_t> memIntrLen = getStaticMemIntrLen(op);
203  return memIntrLen && *memIntrLen <= dataLayout.getTypeSize(slot.valueType);
204 }

206 /// Checks whether all indices are i32. This is used to check GEPs can index
207 /// into them.
208 static bool areAllIndicesI32(const DestructurableMemorySlot &slot) {
209  Type i32 = IntegerType::get(slot.ptr.getContext(), 32);
210  return llvm::all_of(llvm::make_first_range(slot.subelementTypes),
211  [&](Attribute index) {
212  auto intIndex = dyn_cast<IntegerAttr>(index);
213  return intIndex && intIndex.getType() == i32;

221 template <class MemsetIntr>
222 static bool memsetCanRewire(MemsetIntr op, const DestructurableMemorySlot &slot,
223  SmallPtrSetImpl<Attribute> &usedIndices,
224  SmallVectorImpl<MemorySlot> &mustBeSafelyUsed,
225  const DataLayout &dataLayout) {
226  if (&slot.valueType.getDialect() != op.getOperation()->getDialect())
227  return false;
228 
229  if (op.getIsVolatile())
230  return false;
231 
232  if (!cast<DestructurableTypeInterface>(slot.valueType)
233  .getSubelementIndexMap())
234  return false;
235 
236  if (!areAllIndicesI32(slot))
237  return false;
238 
239  return definitelyWritesOnlyWithinSlot(op, slot, dataLayout);
240 }

242 template <class MemsetIntr>
243 static Value memsetGetStored(MemsetIntr op, const MemorySlot &slot,
244  OpBuilder &builder) {
245  /// Returns an integer value that is `width` bits wide representing the value
246  /// assigned to the slot by memset.
247  auto buildMemsetValue = [&](unsigned width) -> Value {
248  assert(width % 8 == 0);
249  auto intType = IntegerType::get(op.getContext(), width);
250 
251  // If we know the pattern at compile time, we can compute and assign a
252  // constant directly.
253  IntegerAttr constantPattern;
254  if (matchPattern(op.getVal(), m_Constant(&constantPattern))) {
255  assert(constantPattern.getValue().getBitWidth() == 8);
256  APInt memsetVal(/*numBits=*/width, /*val=*/0);
257  for (unsigned loBit = 0; loBit < width; loBit += 8)
258  memsetVal.insertBits(constantPattern.getValue(), loBit);
259  return LLVM::ConstantOp::create(builder, op.getLoc(),
260  IntegerAttr::get(intType, memsetVal));
261  }
262 
263  // If the output is a single byte, we can return the pattern directly.
264  if (width == 8)
265  return op.getVal();
266 
267  // Otherwise build the memset integer at runtime by repeatedly shifting the
268  // value and or-ing it with the previous value.
269  uint64_t coveredBits = 8;
270  Value currentValue =
271  LLVM::ZExtOp::create(builder, op.getLoc(), intType, op.getVal());
272  while (coveredBits < width) {
273  Value shiftBy =
274  LLVM::ConstantOp::create(builder, op.getLoc(), intType, coveredBits);
275  Value shifted =
276  LLVM::ShlOp::create(builder, op.getLoc(), currentValue, shiftBy);
277  currentValue =
278  LLVM::OrOp::create(builder, op.getLoc(), currentValue, shifted);
279  coveredBits *= 2;
280  }
281 
282  return currentValue;
283  };
284  return TypeSwitch<Type, Value>(slot.valueType)
285  .Case([&](IntegerType type) -> Value {
286  return buildMemsetValue(type.getWidth());
287  })
288  .Case([&](FloatType type) -> Value {
289  Value intVal = buildMemsetValue(type.getWidth());
290  return LLVM::BitcastOp::create(builder, op.getLoc(), type, intVal);
291  })
292  .DefaultUnreachable(
293  "getStored should not be called on memset to unsupported type");
294 }

296 template <class MemsetIntr>
297 static bool
298 memsetCanUsesBeRemoved(MemsetIntr op, const MemorySlot &slot,
299  const SmallPtrSetImpl<OpOperand *> &blockingUses,
300  SmallVectorImpl<OpOperand *> &newBlockingUses,
301  const DataLayout &dataLayout) {
302  bool canConvertType =
303  TypeSwitch<Type, bool>(slot.valueType)
304  .Case<IntegerType, FloatType>([](auto type) {
305  return type.getWidth() % 8 == 0 && type.getWidth() > 0;
306  })
307  .Default(false);
308  if (!canConvertType)
309  return false;
310 
311  if (op.getIsVolatile())
312  return false;
313 
314  return getStaticMemIntrLen(op) == dataLayout.getTypeSize(slot.valueType);
315 }

317 template <class MemsetIntr>
318 static DeletionKind
319 memsetRewire(MemsetIntr op, const DestructurableMemorySlot &slot,
320  DenseMap<Attribute, MemorySlot> &subslots, OpBuilder &builder,
321  const DataLayout &dataLayout) {
322 
323  std::optional<DenseMap<Attribute, Type>> types =
324  cast<DestructurableTypeInterface>(slot.valueType).getSubelementIndexMap();
325 
326  IntegerAttr memsetLenAttr = createMemsetLenAttr(op);
327 
328  auto structType = dyn_cast<LLVM::LLVMStructType>(slot.valueType);
329 
330  Type i32 = IntegerType::get(op.getContext(), 32);
331  uint64_t memsetLen = memsetLenAttr.getValue().getZExtValue();
332  for (size_t i = 0; i < types->size(); i++) {
333  // Create indices on the fly to get elements in the right order.
334  Attribute index = IntegerAttr::get(i32, i);
335  Type elemType = types->at(index);
336  uint64_t typeSize = dataLayout.getTypeSize(elemType);
337  uint64_t offset = structType ? structType.getElementOffset(dataLayout, i)
338  : i * dataLayout.getTypeAllocSize(elemType);
339  if (offset >= memsetLen)
340  break;
341 
342  // If this subslot is used, apply a new memset to it.
343  if (subslots.contains(index)) {
344  uint64_t newMemsetSize = std::min(memsetLen - offset, typeSize);
345  createMemsetIntr(builder, op, memsetLenAttr, newMemsetSize, subslots,
346  index);
347  }
348  }
349 
350  return DeletionKind::Delete;
351 }

511 template <class MemcpyLike>
512 static bool memcpyCanRewire(MemcpyLike op, const DestructurableMemorySlot &slot,
513  SmallPtrSetImpl<Attribute> &usedIndices,
514  SmallVectorImpl<MemorySlot> &mustBeSafelyUsed,
515  const DataLayout &dataLayout) {
516  if (op.getIsVolatile())
517  return false;

521  return false;

532 
533  return true;

561 /// Rewires a memcpy-like operation. Only copies to or from the full slot are
562 /// supported.
563 template <class MemcpyLike>
564 static DeletionKind
565 memcpyRewire(MemcpyLike op, const DestructurableMemorySlot &slot,
566  DenseMap<Attribute, MemorySlot> &subslots, OpBuilder &builder,
567  const DataLayout &dataLayout) {
568  if (subslots.empty())
569  return DeletionKind::Delete;
570 
571  assert((slot.ptr == op.getDst()) != (slot.ptr == op.getSrc()));
572  bool isDst = slot.ptr == op.getDst();
573 
574 #ifndef NDEBUG
575  size_t slotsTreated = 0;
576 #endif
577 
578  // It was previously checked that index types are consistent, so this type can
579  // be fetched now.
580  Type indexType = cast<IntegerAttr>(subslots.begin()->first).getType();
581  for (size_t i = 0, e = slot.subelementTypes.size(); i != e; i++) {
582  Attribute index = IntegerAttr::get(indexType, i);
583  if (!subslots.contains(index))
584  continue;
585  const MemorySlot &subslot = subslots.at(index);
586 
587 #ifndef NDEBUG
588  slotsTreated++;
589 #endif
590 
591  // First get a pointer to the equivalent of this subslot from the source
592  // pointer.
593  SmallVector<LLVM::GEPArg> gepIndices{
594  0, static_cast<int32_t>(
595  cast<IntegerAttr>(index).getValue().getZExtValue())};
596  Value subslotPtrInOther = LLVM::GEPOp::create(
597  builder, op.getLoc(), LLVM::LLVMPointerType::get(op.getContext()),
598  slot.valueType, isDst ? op.getSrc() : op.getDst(), gepIndices);
599 
600  // Then create a new memcpy out of this source pointer.
601  createMemcpyLikeToReplace(builder, dataLayout, op,
602  isDst ? subslot.ptr : subslotPtrInOther,
603  isDst ? subslotPtrInOther : subslot.ptr,
604  subslot.valueType, op.getIsVolatile());
605  }
606 
607  assert(subslots.size() == slotsTreated);
608 
609  return DeletionKind::Delete;
610 }
