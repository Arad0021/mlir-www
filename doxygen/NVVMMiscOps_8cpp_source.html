35 LogicalResult PMEventOp::verify() {
36  auto eventId = getEventId();
37  auto maskedEventId = getMaskedEventId();
38  if (!maskedEventId && !eventId) {
39  return emitOpError() << "either `id` or `mask` must be set";
40  }
41 
42  if (maskedEventId && eventId) {
43  return emitOpError() << "`id` and `mask` cannot be set at the same time";
44  }
45 
46  if (eventId) {
47  if (eventId < 0 || eventId > 15) {
48  return emitOpError() << "`id` must be between 0 and 15";
49  }
50  }
51 
52  return llvm::success();
53 }

67 mlir::NVVM::IDArgPair
68 PMEventOp::getIntrinsicIDAndArgs(Operation &op, LLVM::ModuleTranslation &mt,
69  llvm::IRBuilderBase &builder) {
70  auto thisOp = cast<NVVM::PMEventOp>(op);
71  llvm::Type *i16Ty = llvm::Type::getInt16Ty(mt.getLLVMContext());
72 
73  // With event-id, mask is generated as (1 << event-id)
74  llvm::Value *maskVal;
75  if (auto eventAttr = thisOp.getEventIdAttr()) {
76  uint16_t mask = static_cast<uint16_t>(1u << eventAttr.getInt());
77  maskVal = llvm::ConstantInt::get(i16Ty, mask);
78  } else {
79  maskVal =
80  llvm::ConstantInt::get(i16Ty, thisOp.getMaskedEventIdAttr().getValue());
81  }
82 
83  return {llvm::Intrinsic::nvvm_pm_event_mask, {maskVal}};
84 }
